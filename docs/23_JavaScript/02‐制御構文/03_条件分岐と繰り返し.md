# 03. 条件分岐と繰り返し

---

## この章で学ぶこと

- `if / else if / else` で条件によって処理を変える
- `switch` で複数の値を分岐する
- `for / while / do-while` で繰り返し処理をする
- `break / continue` でループを制御する
- 早期returnで読みやすいコードを書く

---

## 03-1. if / else if / else

### 基本の if

```javascript
let age = 20;

if (age >= 18) {
  console.log("成人です");
}
// → 成人です
```

`if (条件)` の条件が `true` のとき、`{ }` の中が実行されます。

### if ... else

```javascript
let age = 15;

if (age >= 18) {
  console.log("成人です");
} else {
  console.log("未成年です");
}
// → 未成年です
```

### if ... else if ... else

```javascript
let score = 75;

if (score >= 90) {
  console.log("優");
} else if (score >= 70) {
  console.log("良");
} else if (score >= 50) {
  console.log("可");
} else {
  console.log("不可");
}
// → 良
```

**上から順に判定される**ので、最初に `true` になった分岐だけが実行されます。

### 条件の組み合わせ

```javascript
let age = 25;
let isMember = true;

// AND: 両方満たす
if (age >= 20 && isMember) {
  console.log("会員割引で飲み放題");
}

// OR: どちらか満たす
if (age < 12 || age >= 65) {
  console.log("割引対象");
}

// NOT: 否定
if (!isMember) {
  console.log("会員ではありません");
}
```

### 三項演算子（短い if-else）

```javascript
let age = 20;

// 三項演算子：条件 ? trueのとき : falseのとき
let status = age >= 18 ? "成人" : "未成年";
console.log(status);  // → "成人"

// これは以下と同じ意味
let status2;
if (age >= 18) {
  status2 = "成人";
} else {
  status2 = "未成年";
}
```

> 三項演算子は**短い条件分岐**に便利ですが、複雑な条件には使わないでください。読みにくくなります。

## 03-2. switch（使いどころ）

### 基本の switch

一つの値に対して複数の分岐をする場合に使います。

```javascript
let fruit = "りんご";

switch (fruit) {
  case "りんご":
    console.log("赤い果物");
    break;
  case "バナナ":
    console.log("黄色い果物");
    break;
  case "ぶどう":
    console.log("紫の果物");
    break;
  default:
    console.log("不明な果物");
    break;
}
// → 赤い果物
```

### break を忘れるとフォールスルーする

```javascript
let day = "月";

switch (day) {
  case "月":
    console.log("月曜日");
    // break を忘れると...
  case "火":
    console.log("火曜日");  // ← これも実行される！
    break;
  case "水":
    console.log("水曜日");
    break;
}
// → 月曜日
// → 火曜日（意図しない！）
```

> **ルール：各 case の最後には必ず `break` を書く。** フォールスルーを意図的に使う場合はコメントで明記する。

### switch vs if-else の使い分け

```javascript
// switch が向いている：一つの値を複数の固定値と比較
switch (status) {
  case "active": ...
  case "inactive": ...
  case "deleted": ...
}

// if-else が向いている：範囲の比較や複雑な条件
if (score >= 90) { ... }
else if (score >= 70) { ... }
```

## 03-3. for / while / do-while

### for ループ（回数が決まっているとき）

```javascript
// 0 から 4 まで繰り返す
for (let i = 0; i < 5; i++) {
  console.log(i);
}
// → 0, 1, 2, 3, 4
```

構文：`for (初期化; 条件; 更新) { 処理 }`

```
for (let i = 0; i < 5; i++) {
      ↑          ↑       ↑
      |          |       毎回のループ後に実行
      |          ループを続ける条件
      最初に一度だけ実行
```

```javascript
// 実際の使用例：配列の要素を表示
let fruits = ["りんご", "バナナ", "みかん"];

for (let i = 0; i < fruits.length; i++) {
  console.log(`${i + 1}番目: ${fruits[i]}`);
}
// → 1番目: りんご
// → 2番目: バナナ
// → 3番目: みかん
```

### while ループ（条件が満たされる間繰り返す）

```javascript
let count = 0;

while (count < 3) {
  console.log(count);
  count++;
}
// → 0, 1, 2
```

```javascript
// 実際の使用例：ランダムに一致するまで繰り返す
let target = 7;
let guess;
let attempts = 0;

while (guess !== target) {
  guess = Math.floor(Math.random() * 10);  // 0-9のランダムな整数
  attempts++;
}
console.log(`${attempts}回目で当たり！`);
```

> ⚠️ `while` は条件を間違えると**無限ループ**になります。必ずループが終わる条件を確認しましょう。

### do-while ループ（最低1回は実行する）

```javascript
let input;

do {
  input = prompt("パスワードを入力してください");
} while (input !== "secret");

console.log("正解！");
```

`do-while` は**先に処理を実行してから**条件を判定するので、必ず1回は実行されます。

### for ループのパターン

```javascript
// カウントダウン
for (let i = 5; i >= 1; i--) {
  console.log(i);
}
// → 5, 4, 3, 2, 1

// 2つずつスキップ
for (let i = 0; i < 10; i += 2) {
  console.log(i);
}
// → 0, 2, 4, 6, 8

// ネスト（九九の表）
for (let i = 1; i <= 9; i++) {
  let row = "";
  for (let j = 1; j <= 9; j++) {
    row += `${i * j}\t`;
  }
  console.log(row);
}
```

## 03-4. break / continue

### break：ループを途中で抜ける

```javascript
for (let i = 0; i < 10; i++) {
  if (i === 5) {
    break;  // i が 5 になったらループを抜ける
  }
  console.log(i);
}
// → 0, 1, 2, 3, 4
```

### continue：今回のループをスキップして次へ

```javascript
for (let i = 0; i < 6; i++) {
  if (i === 3) {
    continue;  // i が 3 のときだけスキップ
  }
  console.log(i);
}
// → 0, 1, 2, 4, 5（3がスキップされた）
```

```javascript
// 実際の使用例：偶数だけ表示
for (let i = 0; i < 10; i++) {
  if (i % 2 !== 0) continue;  // 奇数はスキップ
  console.log(i);
}
// → 0, 2, 4, 6, 8
```

## 03-5. 早期returnで読みやすくする

### ネストが深いコードの問題

```javascript
// ❌ ネストが深くて読みにくい
function canDrink(user) {
  if (user !== null) {
    if (user.age !== undefined) {
      if (user.age >= 20) {
        return true;
      } else {
        return false;
      }
    } else {
      return false;
    }
  } else {
    return false;
  }
}
```

### 早期return（ガード節）で改善

```javascript
// ✅ 早期returnで読みやすい
function canDrink(user) {
  if (user === null) return false;
  if (user.age === undefined) return false;
  if (user.age < 20) return false;

  return true;
}
```

**早期return のコツ**：
1. 「ダメな条件」を先にチェックして `return` する
2. 「ダメな条件」をすべて通過したら「正常な処理」を書く
3. ネストが1段になるので読みやすい

```javascript
// もう一つの例
// ❌ ネストが深い
function processOrder(order) {
  if (order) {
    if (order.items.length > 0) {
      if (order.payment) {
        // 本来の処理...
        return "注文を処理しました";
      }
    }
  }
  return "注文を処理できません";
}

// ✅ 早期returnで改善
function processOrder(order) {
  if (!order) return "注文を処理できません";
  if (order.items.length === 0) return "注文を処理できません";
  if (!order.payment) return "注文を処理できません";

  // 本来の処理...
  return "注文を処理しました";
}
```

## よくある間違い

```javascript
// ❌ for文の条件で <= を使って1回多くループする
let arr = [1, 2, 3];
for (let i = 0; i <= arr.length; i++) {  // i <= 3 → 0,1,2,3
  console.log(arr[i]);  // arr[3] は undefined！
}
// ✅ < を使う
for (let i = 0; i < arr.length; i++) { ... }

// ❌ if文の条件で = を使ってしまう（代入になる）
let x = 5;
if (x = 10) {  // 比較ではなく代入！常にtrueになる
  console.log("10です");
}
// ✅ === を使う
if (x === 10) { ... }

// ❌ switch の break 忘れ
switch (value) {
  case 1:
    doSomething();
    // break忘れ → 次のcaseも実行される
  case 2:
    doOtherThing();
    break;
}

// ❌ 無限ループ
let i = 0;
while (i < 10) {
  console.log(i);
  // i++ を忘れている → 永遠にループ
}
```

## 重要ポイント

1. `if / else if / else` は**上から順に**判定され、最初に `true` になった分岐だけ実行
2. `switch` は**一つの値を複数の固定値と比較**するとき便利。`break` を忘れずに
3. `for` は**回数が決まっている**とき、`while` は**条件次第**のとき
4. `break` はループ終了、`continue` は今回スキップ
5. **早期return**（ガード節）でネストを浅くして読みやすいコードを書く
6. ループの条件では `<` と `<=` の違いに注意（off-by-one エラー）
