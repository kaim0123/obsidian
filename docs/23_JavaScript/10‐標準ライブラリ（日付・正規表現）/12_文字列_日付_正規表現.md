## 12-1. 文字列処理の頻出（split / join / replace）

### split：文字列を分割して配列に

```javascript
"a,b,c".split(",");           // → ["a", "b", "c"]
"Hello World".split(" ");     // → ["Hello", "World"]
"abc".split("");              // → ["a", "b", "c"]

// CSVの処理
const csv = "太郎,20,東京";
const [name, age, city] = csv.split(",");
console.log(name);  // → "太郎"
```

### join：配列を結合して文字列に

```javascript
["a", "b", "c"].join(",");    // → "a,b,c"
["Hello", "World"].join(" "); // → "Hello World"
[2024, 1, 15].join("-");      // → "2024-1-15"
["a", "b", "c"].join("");     // → "abc"
```

### replace / replaceAll：置換

```javascript
// 最初の1つだけ置換
"hello world hello".replace("hello", "hi");
// → "hi world hello"

// すべて置換
"hello world hello".replaceAll("hello", "hi");
// → "hi world hi"

// 正規表現で置換
"hello world".replace(/[aeiou]/g, "*");
// → "h*ll* w*rld"
```

### その他の頻出メソッド

```javascript
const str = "  Hello, World!  ";

// 空白除去
str.trim();          // → "Hello, World!"
str.trimStart();     // → "Hello, World!  "
str.trimEnd();       // → "  Hello, World!"

// 大文字・小文字
"hello".toUpperCase();   // → "HELLO"
"HELLO".toLowerCase();   // → "hello"

// 検索
"Hello".includes("ell");    // → true
"Hello".startsWith("He");   // → true
"Hello".endsWith("lo");     // → true
"Hello".indexOf("l");       // → 2（位置）

// 切り出し
"Hello, World".slice(0, 5);   // → "Hello"
"Hello, World".slice(7);      // → "World"
"Hello, World".slice(-5);     // → "World"（後ろから）

// 繰り返し
"ha".repeat(3);              // → "hahaha"

// パディング（桁合わせ）
"5".padStart(3, "0");        // → "005"
"42".padStart(5, " ");       // → "   42"
"hello".padEnd(10, ".");     // → "hello....."
```

## 12-2. 日付（Dateの注意点と扱い方）

### Dateオブジェクトの作成

```javascript
// 現在日時
const now = new Date();
console.log(now);  // → 2024-01-15T06:30:00.000Z（例）

// 文字列から（ISO 8601形式が安全）
const date = new Date("2024-03-15T10:30:00");

// 数値から（月は0始まり！）
const date2 = new Date(2024, 2, 15);  // 2024年3月15日（2 = 3月）
//                      ↑    ↑
//                      年  月(0-11)

// タイムスタンプから（ミリ秒）
const date3 = new Date(1700000000000);
```

> ⚠️ **月は0始まり**（0=1月、1=2月、...、11=12月）。これは非常にハマりやすい罠です。

### 日時の取得

```javascript
const now = new Date();

now.getFullYear();   // → 2024
now.getMonth();      // → 0-11（0始まり！）
now.getDate();       // → 1-31（日）
now.getDay();        // → 0-6（0=日曜、1=月曜、...、6=土曜）
now.getHours();      // → 0-23
now.getMinutes();    // → 0-59
now.getSeconds();    // → 0-59
now.getTime();       // → タイムスタンプ（ミリ秒）
```

### 日時の計算

```javascript
const now = new Date();

// 3日後
const threeDaysLater = new Date(now);
threeDaysLater.setDate(threeDaysLater.getDate() + 3);

// 1ヶ月後
const nextMonth = new Date(now);
nextMonth.setMonth(nextMonth.getMonth() + 1);

// 日付の差（ミリ秒→日に変換）
const start = new Date("2024-01-01");
const end = new Date("2024-03-15");
const diffMs = end - start;
const diffDays = diffMs / (1000 * 60 * 60 * 24);
console.log(diffDays);  // → 74
```

### 日付の比較

```javascript
const date1 = new Date("2024-01-01");
const date2 = new Date("2024-12-31");

// 比較演算子はそのまま使える
console.log(date1 < date2);   // → true
console.log(date1 > date2);   // → false

// 等価比較は getTime() を使う
const date3 = new Date("2024-01-01");
console.log(date1 === date3);              // → false（オブジェクトなので）
console.log(date1.getTime() === date3.getTime());  // → true
```

## 12-3. 正規表現の最小セット（検索 / 置換）

### 正規表現とは

文字列のパターンを表現する仕組み。検索・置換・バリデーションに使います。

### 基本の書き方

```javascript
// リテラル（/ で囲む）
const pattern = /hello/;

// コンストラクタ
const pattern2 = new RegExp("hello");
```

### よく使うパターン

```
.       任意の1文字
\d      数字（0-9）
\w      英数字とアンダースコア
\s      空白文字
^       先頭
$       末尾
+       1回以上の繰り返し
*       0回以上の繰り返し
?       0回または1回
{n}     ちょうどn回
{n,m}   n回以上m回以下
[abc]   a, b, c のいずれか
[a-z]   a〜zのいずれか
(...)   グループ化
|       または
```

### test：マッチするかチェック

```javascript
/\d+/.test("abc123");      // → true（数字がある）
/^\d+$/.test("123");       // → true（数字だけ）
/^\d+$/.test("123abc");    // → false（数字だけではない）
/^[a-zA-Z]+$/.test("Hello"); // → true（英字だけ）
```

### match：マッチした部分を取得

```javascript
"Hello 123 World 456".match(/\d+/);   // → ["123"]（最初の1つ）
"Hello 123 World 456".match(/\d+/g);  // → ["123", "456"]（gフラグで全部）
```

### replace with 正規表現

```javascript
// 数字を全部除去
"abc123def456".replace(/\d+/g, "");  // → "abcdef"

// グループを使った置換
"2024-01-15".replace(/(\d{4})-(\d{2})-(\d{2})/, "$1年$2月$3日");
// → "2024年01月15日"
```

### 実用的な例

```javascript
// メールアドレスの簡易チェック
const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
emailPattern.test("user@example.com");   // → true
emailPattern.test("invalid-email");      // → false

// 電話番号（日本）
const phonePattern = /^0\d{1,4}-\d{1,4}-\d{4}$/;
phonePattern.test("090-1234-5678");      // → true

// 郵便番号
const zipPattern = /^\d{3}-\d{4}$/;
zipPattern.test("100-0001");             // → true
```

## 12-4. 数値とMath（丸め、乱数、浮動小数点の注意）

### Math の便利メソッド

```javascript
// 丸め
Math.round(4.5);    // → 5（四捨五入）
Math.floor(4.9);    // → 4（切り捨て）
Math.ceil(4.1);     // → 5（切り上げ）
Math.trunc(4.9);    // → 4（小数部分を除去）
Math.trunc(-4.9);   // → -4

// 絶対値
Math.abs(-5);       // → 5

// 最大・最小
Math.max(1, 5, 3);  // → 5
Math.min(1, 5, 3);  // → 1
Math.max(...[1, 5, 3]);  // 配列の場合はスプレッド

// べき乗・平方根
Math.pow(2, 3);     // → 8（2の3乗）
2 ** 3;             // → 8（同じ）
Math.sqrt(9);       // → 3

// 乱数
Math.random();                          // → 0以上1未満のランダム
Math.floor(Math.random() * 10);         // → 0〜9のランダム整数
Math.floor(Math.random() * 10) + 1;     // → 1〜10のランダム整数

// ランダムな範囲の整数を生成する関数
function randomInt(min, max) {
  return Math.floor(Math.random() * (max - min + 1)) + min;
}
randomInt(1, 100);  // → 1〜100のランダム整数
```

### 浮動小数点の罠

```javascript
console.log(0.1 + 0.2);  // → 0.30000000000000004 ← ⚠️

// なぜ？コンピュータは10進数の小数を2進数で正確に表現できないから
// 0.1 や 0.2 は2進数では無限小数になる

// 対処法1: 整数に変換して計算
const price = 0.1 * 100 + 0.2 * 100;  // 10 + 20 = 30
const result = price / 100;             // → 0.3

// 対処法2: toFixed で丸める（文字列が返る）
(0.1 + 0.2).toFixed(1);  // → "0.3"

// 対処法3: 比較のときは誤差を許容する
Math.abs(0.1 + 0.2 - 0.3) < Number.EPSILON;  // → true
```

### 数値のフォーマット

```javascript
// 桁区切り
(1234567).toLocaleString();  // → "1,234,567"

// 小数点以下の桁数
(3.14159).toFixed(2);  // → "3.14"（文字列が返る）

// 通貨フォーマット
(1234).toLocaleString("ja-JP", { style: "currency", currency: "JPY" });
// → "￥1,234"
```

## 12-5. 日付フォーマット（表示用に整形する）

### 手動でフォーマット

```javascript
function formatDate(date) {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, "0");
  const d = String(date.getDate()).padStart(2, "0");
  return `${y}-${m}-${d}`;
}

console.log(formatDate(new Date()));  // → "2024-01-15"
```

### toLocaleString

```javascript
const date = new Date("2024-03-15T14:30:00");

// 日本語フォーマット
date.toLocaleDateString("ja-JP");
// → "2024/3/15"

date.toLocaleTimeString("ja-JP");
// → "14:30:00"

date.toLocaleString("ja-JP");
// → "2024/3/15 14:30:00"

// オプションで細かく指定
date.toLocaleDateString("ja-JP", {
  year: "numeric",
  month: "long",
  day: "numeric",
  weekday: "long",
});
// → "2024年3月15日金曜日"
```

### Intl.DateTimeFormat

```javascript
const formatter = new Intl.DateTimeFormat("ja-JP", {
  year: "numeric",
  month: "2-digit",
  day: "2-digit",
  hour: "2-digit",
  minute: "2-digit",
});

console.log(formatter.format(new Date()));
// → "2024/01/15 14:30"
```

### ISO文字列（APIで送受信するとき）

```javascript
const date = new Date();

// ISO 8601形式（UTC）
date.toISOString();
// → "2024-01-15T05:30:00.000Z"

// ISO文字列から復元
const restored = new Date("2024-01-15T05:30:00.000Z");
```

## よくある間違い

```javascript
// ❌ Date の月が0始まりだと忘れる
new Date(2024, 3, 1);  // → 4月1日（3月ではない！）

// ❌ 0.1 + 0.2 === 0.3 が true だと思う
console.log(0.1 + 0.2 === 0.3);  // → false

// ❌ toFixed が数値を返すと思う
typeof (3.14).toFixed(1);  // → "string"（文字列！）

// ❌ replace が元の文字列を変えると思う
let str = "hello";
str.replace("hello", "world");
console.log(str);  // → "hello"（文字列はイミュータブル！）
// ✅ 結果を受け取る
str = str.replace("hello", "world");
```

---

## 重要ポイント

1. **`split / join`** で文字列⇔配列を変換
2. **`replace / replaceAll`** で文字列を置換。元の文字列は変わらない（戻り値を使う）
3. **Dateの月は0始まり**（0=1月）。これは絶対に覚える
4. **正規表現**は `/パターン/フラグ` で書く。`test()`でマッチ確認、`match()`で取得
5. **浮動小数点の誤差**に注意。金額計算は整数（円単位）で行う
6. 日付のフォーマットは **`toLocaleDateString`** が手軽
