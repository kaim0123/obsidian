# サーバーレスアプリケーション実装例

## 概要
このガイドでは、実際のサーバーレスアプリケーションの完全な実装例を通じて、これまで学んだ技術を統合します。

## プロジェクト: タスク管理アプリケーション

### システム概要
```
機能:
- ユーザー認証（Cognito）
- タスクCRUD操作（DynamoDB）
- ファイル添付（S3署名付きURL）
- メール通知（SNS）
- リアルタイム更新（WebSocket API）
- 定期レポート（EventBridge）
```

### アーキテクチャ
```
フロントエンド（React）
    ↓ HTTPS
CloudFront + S3（静的ホスティング）
    ↓
API Gateway (HTTP API)
    ↓ Cognito認証
Lambda関数群
    ├→ DynamoDB（タスクデータ）
    ├→ S3（ファイル添付）
    ├→ SNS（通知）
    └→ EventBridge（スケジュール）
```

## Lambda関数の実装

### 共通ライブラリ（Layer）
```python
# layers/common/python/lib/utils.py

import json
import logging
import os
from decimal import Decimal
from typing import Dict, Any

logger = logging.getLogger()
logger.setLevel(os.environ.get('LOG_LEVEL', 'INFO'))

class DecimalEncoder(json.JSONEncoder):
    def default(self, obj):
        if isinstance(obj, Decimal):
            return float(obj)
        return super(DecimalEncoder, self).default(obj)

def create_response(status_code: int, body: Any, headers: Dict = None) -> Dict[str, Any]:
    return {
        'statusCode': status_code,
        'headers': {
            'Content-Type': 'application/json',
            'Access-Control-Allow-Origin': '*',
            **(headers or {})
        },
        'body': json.dumps(body, cls=DecimalEncoder)
    }

def get_user_id(event: Dict[str, Any]) -> str:
    """認証済みユーザーIDを取得"""
    jwt_claims = event['requestContext']['authorizer']['jwt']['claims']
    return jwt_claims['sub']
```

### タスク一覧取得
```python
# functions/list_tasks/app.py

import boto3
import os
from lib.utils import create_response, get_user_id, logger

dynamodb = boto3.resource('dynamodb')
table = dynamodb.Table(os.environ['TASKS_TABLE'])

def lambda_handler(event, context):
    """タスク一覧を取得"""
    try:
        user_id = get_user_id(event)
        query_params = event.get('queryStringParameters', {}) or {}
        status = query_params.get('status')
        
        logger.info(f"Listing tasks for user {user_id}, status={status}")
        
        if status:
            # GSIを使用してステータスでフィルタ
            response = table.query(
                IndexName='UserStatusIndex',
                KeyConditionExpression='userId = :userId AND #status = :status',
                ExpressionAttributeNames={'#status': 'status'},
                ExpressionAttributeValues={
                    ':userId': user_id,
                    ':status': status
                }
            )
        else:
            # 全タスクを取得
            response = table.query(
                KeyConditionExpression='userId = :userId',
                ExpressionAttributeValues={':userId': user_id}
            )
        
        tasks = response['Items']
        logger.info(f"Found {len(tasks)} tasks")
        
        return create_response(200, {
            'tasks': tasks,
            'count': len(tasks)
        })
        
    except Exception as e:
        logger.error(f"Error listing tasks: {str(e)}", exc_info=True)
        return create_response(500, {'error': 'Internal server error'})
```

### タスク作成
```python
# functions/create_task/app.py

import boto3
import json
import os
import uuid
from datetime import datetime
from lib.utils import create_response, get_user_id, logger

dynamodb = boto3.resource('dynamodb')
sns = boto3.client('sns')

table = dynamodb.Table(os.environ['TASKS_TABLE'])
TOPIC_ARN = os.environ['NOTIFICATION_TOPIC']

def lambda_handler(event, context):
    """新しいタスクを作成"""
    try:
        user_id = get_user_id(event)
        body = json.loads(event['body'])
        
        # タスクデータを作成
        task_id = str(uuid.uuid4())
        timestamp = int(datetime.now().timestamp())
        
        task = {
            'userId': user_id,
            'taskId': task_id,
            'title': body['title'],
            'description': body.get('description', ''),
            'status': 'pending',
            'priority': body.get('priority', 'medium'),
            'dueDate': body.get('dueDate'),
            'attachments': body.get('attachments', []),
            'createdAt': timestamp,
            'updatedAt': timestamp
        }
        
        # DynamoDBに保存
        table.put_item(Item=task)
        
        logger.info(f"Created task {task_id} for user {user_id}")
        
        # 通知を送信
        try:
            sns.publish(
                TopicArn=TOPIC_ARN,
                Subject='New Task Created',
                Message=json.dumps({
                    'type': 'task_created',
                    'taskId': task_id,
                    'userId': user_id,
                    'title': task['title']
                })
            )
        except Exception as e:
            logger.warning(f"Failed to send notification: {str(e)}")
        
        return create_response(201, task)
        
    except KeyError as e:
        return create_response(400, {'error': f'Missing required field: {str(e)}'})
    except Exception as e:
        logger.error(f"Error creating task: {str(e)}", exc_info=True)
        return create_response(500, {'error': 'Internal server error'})
```

### ファイルアップロード用署名付きURL生成
```python
# functions/generate_upload_url/app.py

import boto3
import json
import os
import uuid
from lib.utils import create_response, get_user_id, logger

s3 = boto3.client('s3')
BUCKET_NAME = os.environ['FILES_BUCKET']

def lambda_handler(event, context):
    """ファイルアップロード用の署名付きURLを生成"""
    try:
        user_id = get_user_id(event)
        body = json.loads(event['body'])
        
        file_name = body['fileName']
        content_type = body.get('contentType', 'application/octet-stream')
        task_id = body.get('taskId')
        
        # S3キーを生成
        file_id = str(uuid.uuid4())
        s3_key = f"users/{user_id}/tasks/{task_id}/{file_id}/{file_name}"
        
        # 署名付きURL生成
        upload_url = s3.generate_presigned_url(
            'put_object',
            Params={
                'Bucket': BUCKET_NAME,
                'Key': s3_key,
                'ContentType': content_type
            },
            ExpiresIn=3600  # 1時間
        )
        
        logger.info(f"Generated upload URL for {s3_key}")
        
        return create_response(200, {
            'uploadUrl': upload_url,
            'fileKey': s3_key,
            'expiresIn': 3600
        })
        
    except Exception as e:
        logger.error(f"Error generating upload URL: {str(e)}", exc_info=True)
        return create_response(500, {'error': 'Internal server error'})
```

### Stream処理（通知送信）
```python
# functions/stream_processor/app.py

import boto3
import json
import os
from lib.utils import logger

sns = boto3.client('sns')
TOPIC_ARN = os.environ['NOTIFICATION_TOPIC']

def lambda_handler(event, context):
    """DynamoDB Streamイベントを処理して通知送信"""
    try:
        for record in event['Records']:
            event_name = record['eventName']
            
            if event_name == 'INSERT':
                # 新規タスク作成
                new_image = record['dynamodb']['NewImage']
                handle_task_created(new_image)
            
            elif event_name == 'MODIFY':
                # タスク更新
                old_image = record['dynamodb']['OldImage']
                new_image = record['dynamodb']['NewImage']
                handle_task_updated(old_image, new_image)
        
        return {'statusCode': 200}
        
    except Exception as e:
        logger.error(f"Error processing stream: {str(e)}", exc_info=True)
        raise

def handle_task_created(task):
    """タスク作成通知"""
    title = task['title']['S']
    user_id = task['userId']['S']
    
    logger.info(f"Sending notification for new task: {title}")
    
    sns.publish(
        TopicArn=TOPIC_ARN,
        Subject='New Task Created',
        Message=f'A new task has been created: {title}',
        MessageAttributes={
            'eventType': {'DataType': 'String', 'StringValue': 'task_created'},
            'userId': {'DataType': 'String', 'StringValue': user_id}
        }
    )

def handle_task_updated(old_task, new_task):
    """タスク更新通知"""
    old_status = old_task.get('status', {}).get('S')
    new_status = new_task.get('status', {}).get('S')
    
    if old_status != new_status and new_status == 'completed':
        # 完了通知
        title = new_task['title']['S']
        logger.info(f"Sending completion notification for: {title}")
        
        sns.publish(
            TopicArn=TOPIC_ARN,
            Subject='Task Completed',
            Message=f'Task completed: {title}',
            MessageAttributes={
                'eventType': {'DataType': 'String', 'StringValue': 'task_completed'}
            }
        )
```

### タスクサービス
```typescript
// src/api/taskService.ts

import apiClient from './client';

export interface Task {
  userId: string;
  taskId: string;
  title: string;
  description?: string;
  status: 'pending' | 'in_progress' | 'completed';
  priority: 'low' | 'medium' | 'high';
  dueDate?: string;
  attachments?: string[];
  createdAt: number;
  updatedAt: number;
}

export const taskService = {
  async getTasks(status?: string): Promise<Task[]> {
    const params = status ? { status } : {};
    const response = await apiClient.get('/tasks', { params });
    return response.data.tasks;
  },
  
  async getTask(taskId: string): Promise<Task> {
    const response = await apiClient.get(`/tasks/${taskId}`);
    return response.data;
  },
  
  async createTask(task: Partial<Task>): Promise<Task> {
    const response = await apiClient.post('/tasks', task);
    return response.data;
  },
  
  async updateTask(taskId: string, updates: Partial<Task>): Promise<Task> {
    const response = await apiClient.put(`/tasks/${taskId}`, updates);
    return response.data;
  },
  
  async deleteTask(taskId: string): Promise<void> {
    await apiClient.delete(`/tasks/${taskId}`);
  }
};
```

### Reactコンポーネント
```tsx
// src/components/TaskList.tsx

import React, { useState, useEffect } from 'react';
import { taskService, Task } from '../api/taskService';

export const TaskList: React.FC = () => {
  const [tasks, setTasks] = useState<Task[]>([]);
  const [loading, setLoading] = useState(true);
  const [filter, setFilter] = useState<string>('');
  
  useEffect(() => {
    fetchTasks();
  }, [filter]);
  
  const fetchTasks = async () => {
    try {
      setLoading(true);
      const data = await taskService.getTasks(filter);
      setTasks(data);
    } catch (error) {
      console.error('Error fetching tasks:', error);
    } finally {
      setLoading(false);
    }
  };
  
  const handleStatusChange = async (taskId: string, newStatus: string) => {
    try {
      await taskService.updateTask(taskId, { status: newStatus as any });
      fetchTasks();
    } catch (error) {
      console.error('Error updating task:', error);
    }
  };
  
  if (loading) return <div>Loading...</div>;
  
  return (
    <div className="task-list">
      <div className="filters">
        <button onClick={() => setFilter('')}>All</button>
        <button onClick={() => setFilter('pending')}>Pending</button>
        <button onClick={() => setFilter('in_progress')}>In Progress</button>
        <button onClick={() => setFilter('completed')}>Completed</button>
      </div>
      
      {tasks.map(task => (
        <div key={task.taskId} className="task-item">
          <h3>{task.title}</h3>
          <p>{task.description}</p>
          <span className={`priority priority-${task.priority}`}>
            {task.priority}
          </span>
          <select
            value={task.status}
            onChange={(e) => handleStatusChange(task.taskId, e.target.value)}
          >
            <option value="pending">Pending</option>
            <option value="in_progress">In Progress</option>
            <option value="completed">Completed</option>
          </select>
        </div>
      ))}
    </div>
  );
};
```
