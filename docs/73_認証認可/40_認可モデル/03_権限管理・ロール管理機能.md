# 権限管理・ロール管理機能の実装方法とベストプラクティス

## 概要

権限管理・ロール管理機能は、ユーザーがシステム内でどの操作を実行できるかを制御する重要なセキュリティ機能です。適切な権限設計により、最小権限の原則に基づいた安全なシステムを構築できます。

## 権限管理の基本概念

### 認証（Authentication）vs 認可（Authorization）
- **認証**: ユーザーが誰であるかを確認
- **認可**: ユーザーが何を実行できるかを決定

### 権限モデル
1. **RBAC（Role-Based Access Control）** - ロールベース
2. **ABAC（Attribute-Based Access Control）** - 属性ベース
3. **ACL（Access Control List）** - アクセス制御リスト
4. **PBAC（Policy-Based Access Control）** - ポリシーベース

// 権限チェックミドルウェア
const checkPermission = (requiredPermission) => {
  return async (req, res, next) => {
    try {
      const user = req.user;
      
      if (!user) {
        return res.status(401).json({ error: 'Authentication required' });
      }
      
      // ユーザーの権限を取得
      const userPermissions = await getUserPermissions(user.id);
      
      // 権限チェック
      if (!userPermissions.includes(requiredPermission)) {
        return res.status(403).json({ 
          error: 'Insufficient permissions',
          required: requiredPermission,
          userPermissions: userPermissions
        });
      }
      
      next();
    } catch (error) {
      console.error('Permission check error:', error);
      res.status(500).json({ error: 'Permission check failed' });
    }
  };
};

// ユーザーの権限取得
const getUserPermissions = async (userId) => {
  try {
    const user = await User.findById(userId)
      .populate({
        path: 'roles',
        populate: {
          path: 'roleId',
          model: 'Role'
        }
      });
    
    if (!user) {
      return [];
    }
    
    const permissions = new Set();
    
    // ロールから権限を取得
    user.roles.forEach(userRole => {
      if (userRole.isActive && (!userRole.expiresAt || userRole.expiresAt > new Date())) {
        const role = userRole.roleId;
        if (role && role.isActive) {
          role.permissions.forEach(permission => {
            permissions.add(permission);
          });
        }
      }
    });
    
    // 直接権限を追加（オーバーライド）
    user.permissions.forEach(permission => {
      permissions.add(permission);
    });
    
    return Array.from(permissions);
  } catch (error) {
    console.error('Get user permissions error:', error);
    return [];
  }
};

app.post('/api/roles', checkPermission('roles:create'), async (req, res) => {
  try {
    const { name, description, permissions } = req.body;
    
    // 入力値検証
    if (!name || !permissions || !Array.isArray(permissions)) {
      return res.status(400).json({ error: 'Name and permissions are required' });
    }
    
    // 重複チェック
    const existingRole = await Role.findOne({ name });
    if (existingRole) {
      return res.status(409).json({ error: 'Role already exists' });
    }
    
    // 権限の有効性チェック
    const validPermissions = await Permission.find({ 
      name: { $in: permissions },
      isActive: true 
    });
    
    if (validPermissions.length !== permissions.length) {
      return res.status(400).json({ error: 'Invalid permissions provided' });
    }
    
    const role = await Role.create({
      name,
      description,
      permissions
    });
    
    res.status(201).json({ 
      success: true, 
      role: {
        id: role.id,
        name: role.name,
        description: role.description,
        permissions: role.permissions
      }
    });
    
  } catch (error) {
    console.error('Role creation error:', error);
    res.status(500).json({ error: 'Role creation failed' });
  }
});

    const { roleId } = req.params;

    const role = await Role.findById(roleId);

        return res.status(400).json({ error: 'Invalid permissions provided' });

    // ロール更新
    role.name = name || role.name;
    role.description = description !== undefined ? description : role.description;
    role.permissions = permissions || role.permissions;
    role.updatedAt = new Date();
    
    await role.save();
    
    res.json({

        id: role.id,
        name: role.name,
        description: role.description,
        permissions: role.permissions

    role.isActive = false;
    role.updatedAt = new Date();
    await role.save();
    
    res.json({ success: true, message: 'Role deleted successfully' });

      userId,
      roleId,
      assignedBy,
      expiresAt: expiresAt ? new Date(expiresAt) : undefined

        role: {
          id: role.id,
          name: role.name,
          description: role.description
        },

// ロール作成フォーム
const CreateRoleForm = ({ permissions, onSubmit, onCancel, isLoading }) => {
  const [formData, setFormData] = useState({
    name: '',
    description: '',
    permissions: []
  });

  const handleInputChange = (e) => {
    const { name, value } = e.target;
    setFormData(prev => ({
      ...prev,
      [name]: value
    }));
  };

  const handlePermissionChange = (permission, checked) => {
    setFormData(prev => ({
      ...prev,
      permissions: checked
        ? [...prev.permissions, permission]
        : prev.permissions.filter(p => p !== permission)
    }));
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    onSubmit(formData);
  };

  // 権限をリソース別にグループ化
  const groupedPermissions = permissions.reduce((groups, permission) => {
    const resource = permission.resource;
    if (!groups[resource]) {
      groups[resource] = [];
    }
    groups[resource].push(permission);
    return groups;
  }, {});

  return (
    <form onSubmit={handleSubmit} className="create-role-form">
      <h3>Create New Role</h3>
      
      <div className="form-group">
        <label htmlFor="name">Role Name</label>
        <input
          type="text"
          id="name"
          name="name"
          value={formData.name}
          onChange={handleInputChange}
          required
        />
      </div>
      
      <div className="form-group">
        <label htmlFor="description">Description</label>
        <textarea
          id="description"
          name="description"
          value={formData.description}
          onChange={handleInputChange}
          rows="3"
        />
      </div>
      
      <div className="form-group">
        <label>Permissions</label>
        <div className="permissions-grid">
          {Object.entries(groupedPermissions).map(([resource, resourcePermissions]) => (
            <div key={resource} className="permission-group">
              <h4>{resource}</h4>
              {resourcePermissions.map(permission => (
                <label key={permission.id} className="permission-item">

## 2. 高度な権限管理機能

### リソースベース権限

```javascript
// リソースベース権限チェック
const checkResourcePermission = (resource, action, resourceId = null) => {
  return async (req, res, next) => {
    try {
      const user = req.user;
      const permission = `${resource}:${action}`;
      
      // 基本権限チェック
      const userPermissions = await getUserPermissions(user.id);
      if (!userPermissions.includes(permission)) {
        return res.status(403).json({ error: 'Insufficient permissions' });
      }
      
      // リソース固有の権限チェック
      if (resourceId) {
        const hasResourceAccess = await checkResourceAccess(user.id, resource, resourceId);
        if (!hasResourceAccess) {
          return res.status(403).json({ error: 'Access denied to this resource' });
        }
      }
      
      next();
    } catch (error) {
      console.error('Resource permission check error:', error);
      res.status(500).json({ error: 'Permission check failed' });
    }
  };
};

// リソースアクセスチェック
const checkResourceAccess = async (userId, resource, resourceId) => {
  switch (resource) {
    case 'posts':
      const post = await Post.findById(resourceId);
      return post && (post.authorId.toString() === userId || await hasAdminRole(userId));
    
    case 'users':
      return resourceId === userId || await hasAdminRole(userId);
    
    default:
      return true;
  }
};
```

### 監査ログ

```javascript
// 権限変更の監査ログ
const auditLog = async (action, details) => {
  await AuditLog.create({
    action,
    details,
    timestamp: new Date(),
    ipAddress: req.ip,
    userAgent: req.get('User-Agent')
  });
};

// ロール変更時の監査ログ
app.post('/api/roles', checkPermission('roles:create'), async (req, res) => {
  // ... ロール作成処理 ...
  
  await auditLog('role_created', {
    roleId: role.id,
    roleName: role.name,
    permissions: role.permissions,
    createdBy: req.user.userId
  });
  
  res.json({ success: true, role });
});
```

## 実装の選択指針

| システム規模 | 推奨権限モデル | 理由 |
|-------------|---------------|------|
| **小規模** | シンプルRBAC | 実装の簡素化 |
| **中規模** | RBAC + リソースベース | 柔軟性とセキュリティのバランス |
| **大規模** | ABAC + ポリシーエンジン | 複雑な要件への対応 |
| **エンタープライズ** | 複数モデルの組み合わせ | 包括的なセキュリティ |
