## 1. JWT（JSON Web Token）

### 概要
JSON Web Tokenを使用したステートレスな認証方式

### 実装例

```javascript
// フロントエンド（React）
import jwt from 'jsonwebtoken';

// ログイン処理
const login = async (email, password) => {
  try {
    const response = await fetch('/api/auth/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email, password })
    });
    
    const { token, refreshToken } = await response.json();
    
    // トークンをlocalStorageに保存
    localStorage.setItem('accessToken', token);
    localStorage.setItem('refreshToken', refreshToken);
    
    return { success: true };
  } catch (error) {
    return { success: false, error: error.message };
  }
};

// 認証が必要なAPIリクエスト
const authenticatedRequest = async (url, options = {}) => {
  const token = localStorage.getItem('accessToken');
  
  const response = await fetch(url, {
    ...options,
    headers: {
      ...options.headers,
      'Authorization': `Bearer ${token}`
    }
  });
  
  // トークンが期限切れの場合
  if (response.status === 401) {
    const refreshed = await refreshAccessToken();
    if (refreshed) {
      // 再試行
      return authenticatedRequest(url, options);
    } else {
      // ログイン画面にリダイレクト
      window.location.href = '/login';
    }
  }
  
  return response;
};

// リフレッシュトークンでアクセストークンを更新
const refreshAccessToken = async () => {
  const refreshToken = localStorage.getItem('refreshToken');
  
  try {
    const response = await fetch('/api/auth/refresh', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ refreshToken })
    });
    
    if (response.ok) {
      const { token } = await response.json();
      localStorage.setItem('accessToken', token);
      return true;
    }
  } catch (error) {
    console.error('Token refresh failed:', error);
  }
  
  // リフレッシュに失敗した場合、ログアウト
  localStorage.removeItem('accessToken');
  localStorage.removeItem('refreshToken');
  return false;
};
```

```javascript
// バックエンド（Node.js + Express）
const jwt = require('jsonwebtoken');
const bcrypt = require('bcrypt');

// ログイン処理
app.post('/api/auth/login', async (req, res) => {
  try {
    const { email, password } = req.body;
    
    // ユーザー検索
    const user = await User.findOne({ email });
    if (!user) {
      return res.status(401).json({ error: 'Invalid credentials' });
    }
    
    // パスワード検証
    const isValidPassword = await bcrypt.compare(password, user.password);
    if (!isValidPassword) {
      return res.status(401).json({ error: 'Invalid credentials' });
    }
    
    // JWTトークン生成
    const accessToken = jwt.sign(
      { userId: user.id, email: user.email },
      process.env.JWT_SECRET,
      { expiresIn: '15m' }
    );
    
    const refreshToken = jwt.sign(
      { userId: user.id },
      process.env.JWT_REFRESH_SECRET,
      { expiresIn: '7d' }
    );
    
    res.json({ accessToken, refreshToken });
  } catch (error) {
    res.status(500).json({ error: 'Internal server error' });
  }
});

// ミドルウェア：JWT認証
const authenticateToken = (req, res, next) => {
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1];
  
  if (!token) {
    return res.status(401).json({ error: 'Access token required' });
  }
  
  jwt.verify(token, process.env.JWT_SECRET, (err, user) => {
    if (err) {
      return res.status(403).json({ error: 'Invalid or expired token' });
    }
    req.user = user;
    next();
  });
};

// 保護されたルート
app.get('/api/profile', authenticateToken, (req, res) => {
  res.json({ user: req.user });
});
```

### メリット
- ✅ **ステートレス**: サーバー側でセッション情報を保持不要
- ✅ **スケーラブル**: 複数のサーバーで共有可能
- ✅ **クロスドメイン対応**: CORS設定で異なるドメイン間で使用可能
- ✅ **モバイル対応**: ネイティブアプリでも簡単に実装
- ✅ **マイクロサービス対応**: サービス間認証に適している

### デメリット
- ❌ **トークンサイズ**: セッションIDより大きい
- ❌ **取り消しが困難**: トークンが有効期限まで有効
- ❌ **セキュリティリスク**: トークンが漏洩した場合の影響が大きい
- ❌ **実装の複雑さ**: リフレッシュトークンの管理が必要

### 使用場面
- 📱 **SPA（Single Page Application）**: React、Vue.js アプリ
- 📲 **モバイルアプリ**: React Native、Flutter アプリ
- 🔗 **マイクロサービス**: サービス間認証
- 🌐 **API サービス**: RESTful API の認証

## 2. セッション管理

### 概要
サーバー側でセッション情報を管理する従来の認証方式

### 実装例

```javascript
// バックエンド（Node.js + Express + express-session）
const session = require('express-session');
const MongoStore = require('connect-mongo');

// セッション設定
app.use(session({
  secret: process.env.SESSION_SECRET,
  resave: false,
  saveUninitialized: false,
  store: MongoStore.create({
    mongoUrl: process.env.MONGODB_URI,
    ttl: 14 * 24 * 60 * 60 // 14日
  }),
  cookie: {
    secure: process.env.NODE_ENV === 'production',
    httpOnly: true,
    maxAge: 14 * 24 * 60 * 60 * 1000 // 14日
  }
}));

// ログイン処理
app.post('/api/auth/login', async (req, res) => {
  try {
    const { email, password } = req.body;
    
    const user = await User.findOne({ email });
    if (!user) {
      return res.status(401).json({ error: 'Invalid credentials' });
    }
    
    const isValidPassword = await bcrypt.compare(password, user.password);
    if (!isValidPassword) {
      return res.status(401).json({ error: 'Invalid credentials' });
    }
    
    // セッションにユーザー情報を保存
    req.session.userId = user.id;
    req.session.email = user.email;
    req.session.isAuthenticated = true;
    
    res.json({ success: true, user: { id: user.id, email: user.email } });
  } catch (error) {
    res.status(500).json({ error: 'Internal server error' });
  }
});

// ログアウト処理
app.post('/api/auth/logout', (req, res) => {
  req.session.destroy((err) => {
    if (err) {
      return res.status(500).json({ error: 'Logout failed' });
    }
    res.clearCookie('connect.sid');
    res.json({ success: true });
  });
});

// ミドルウェア：認証チェック
const requireAuth = (req, res, next) => {
  if (req.session && req.session.isAuthenticated) {
    return next();
  } else {
    return res.status(401).json({ error: 'Authentication required' });
  }
};

// 保護されたルート
app.get('/api/profile', requireAuth, async (req, res) => {
  try {
    const user = await User.findById(req.session.userId);
    res.json({ user: { id: user.id, email: user.email } });
  } catch (error) {
    res.status(500).json({ error: 'Internal server error' });
  }
});
```

```javascript
// フロントエンド（React）
// セッション管理は自動的にクッキーで処理される

  }

const logout = async () => {
  try {
    await fetch('/api/auth/logout', {
      method: 'POST',
      credentials: 'include'

    // ログイン画面にリダイレクト
    window.location.href = '/login';
  } catch (error) {
    console.error('Logout failed:', error);
  }

// 認証状態の確認
const checkAuthStatus = async () => {
  try {
    const response = await fetch('/api/auth/status', {
      credentials: 'include'
    });

      return { isAuthenticated: true, user: data.user };

      return { isAuthenticated: false };
    }
  } catch (error) {
    return { isAuthenticated: false };
  }
};
```

### メリット
- ✅ **セキュリティが高い**: サーバー側でセッション管理
- ✅ **取り消しが容易**: セッションを即座に無効化可能
- ✅ **実装が直感的**: 従来のWebアプリケーションの標準
- ✅ **セッション情報の管理**: 複雑なユーザー状態を管理可能
- ✅ **CSRF対策**: CSRFトークンと組み合わせやすい

### デメリット
- ❌ **ステートフル**: サーバー側でセッション情報を保持
- ❌ **スケーラビリティ**: 複数サーバーでの共有が困難
- ❌ **メモリ使用量**: セッション情報の保存にメモリが必要
- ❌ **クロスドメイン対応**: 異なるドメイン間での使用が困難
- ❌ **モバイル対応**: ネイティブアプリでの実装が複雑

### 使用場面
- 🌐 **従来のWebアプリケーション**: PHP、Ruby on Rails、Django
- 🏢 **企業内システム**: イントラネットアプリケーション
- 🛒 **ECサイト**: ショッピングカート機能付きサイト
- 📊 **管理画面**: 複雑な権限管理が必要なシステム

## 3. OAuth 2.0 / OpenID Connect

### 概要
サードパーティ認証プロバイダーを使用した認証方式

### 実装例

```javascript
// フロントエンド（React）
import { GoogleLogin } from '@react-oauth/google';

const OAuthLogin = () => {
  const handleGoogleSuccess = async (credentialResponse) => {
    try {
      const response = await fetch('/api/auth/google', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ token: credentialResponse.credential })
      });
      
      if (response.ok) {
        const data = await response.json();
        // ログイン成功処理
        console.log('Login successful:', data.user);
      }
    } catch (error) {
      console.error('OAuth login failed:', error);
    }
  };

  const handleGoogleError = () => {
    console.error('Google login failed');
  };

  return (
    <div>
      <GoogleLogin
        onSuccess={handleGoogleSuccess}
        onError={handleGoogleError}
        useOneTap
      />
    </div>
  );
};
```

```javascript
// バックエンド（Node.js + Express）
const { OAuth2Client } = require('google-auth-library');
const client = new OAuth2Client(process.env.GOOGLE_CLIENT_ID);

// Google OAuth認証
app.post('/api/auth/google', async (req, res) => {
  try {
    const { token } = req.body;
    
    // Googleトークンの検証
    const ticket = await client.verifyIdToken({
      idToken: token,
      audience: process.env.GOOGLE_CLIENT_ID
    });
    
    const payload = ticket.getPayload();
    const { sub: googleId, email, name, picture } = payload;
    
    // ユーザーを検索または作成
    let user = await User.findOne({ googleId });
    
    if (!user) {
      user = await User.create({
        googleId,
        email,
        name,
        picture,
        provider: 'google'
      });
    }
    
    // セッションまたはJWTトークンを作成
    req.session.userId = user.id;
    req.session.isAuthenticated = true;
    
    res.json({ 
      success: true, 
      user: { id: user.id, email: user.email, name: user.name } 
    });
  } catch (error) {
    res.status(500).json({ error: 'OAuth authentication failed' });
  }
});

// GitHub OAuth認証
const passport = require('passport');
const GitHubStrategy = require('passport-github2').Strategy;

passport.use(new GitHubStrategy({
  clientID: process.env.GITHUB_CLIENT_ID,
  clientSecret: process.env.GITHUB_CLIENT_SECRET,
  callbackURL: "/api/auth/github/callback"
}, async (accessToken, refreshToken, profile, done) => {
  try {
    let user = await User.findOne({ githubId: profile.id });
    
    if (!user) {
      user = await User.create({
        githubId: profile.id,
        username: profile.username,
        email: profile.emails[0].value,
        avatar: profile.photos[0].value,
        provider: 'github'
      });
    }
    
    return done(null, user);
  } catch (error) {
    return done(error, null);
  }
}));

app.get('/api/auth/github', passport.authenticate('github'));
app.get('/api/auth/github/callback', 
  passport.authenticate('github', { failureRedirect: '/login' }),
  (req, res) => {
    res.redirect('/dashboard');
  }
);
```

### メリット
- ✅ **ユーザー体験**: パスワード入力不要
- ✅ **セキュリティ**: パスワード管理の負担を軽減
- ✅ **信頼性**: 大手プロバイダーのセキュリティを活用
- ✅ **実装の簡素化**: 認証ロジックを外部に委譲
- ✅ **多様なプロバイダー**: Google、GitHub、Facebook等に対応

### デメリット
- ❌ **外部依存**: プロバイダーのサービスに依存
- ❌ **プライバシー**: ユーザー情報が外部に提供される
- ❌ **カスタマイズ制限**: プロバイダーの制約を受ける
- ❌ **実装の複雑さ**: 複数のプロバイダー対応が複雑
- ❌ **オフライン対応**: インターネット接続が必要

### 使用場面
- 🔐 **ソーシャルログイン**: Google、GitHub、Facebook ログイン

## 4. API Key認証

### 概要
APIキーを使用したシンプルな認証方式

### 実装例

```javascript
// バックエンド（Node.js + Express）
const crypto = require('crypto');

// APIキー生成
const generateApiKey = () => {
  return crypto.randomBytes(32).toString('hex');
};

// APIキー認証ミドルウェア
const authenticateApiKey = async (req, res, next) => {
  const apiKey = req.headers['x-api-key'] || req.query.api_key;
  
  if (!apiKey) {
    return res.status(401).json({ error: 'API key required' });
  }
  
  try {
    const key = await ApiKey.findOne({ key: apiKey, isActive: true });
    
    if (!key) {
      return res.status(401).json({ error: 'Invalid API key' });
    }
    
    // 使用回数の更新
    key.usageCount += 1;
    key.lastUsed = new Date();
    await key.save();
    
    req.apiKey = key;
    req.user = await User.findById(key.userId);
    next();
  } catch (error) {
    res.status(500).json({ error: 'Authentication failed' });
  }
};

// APIキー管理
app.post('/api/keys', requireAuth, async (req, res) => {
  try {
    const { name, permissions } = req.body;
    const apiKey = generateApiKey();
    
    const key = await ApiKey.create({
      key: apiKey,
      name,
      userId: req.user.id,
      permissions: permissions || ['read'],
      isActive: true
    });
    
    res.json({ 
      success: true, 
      apiKey: key.key, // 初回のみ表示
      keyId: key.id 
    });
  } catch (error) {
    res.status(500).json({ error: 'Failed to create API key' });
  }
});

// 保護されたAPI
app.get('/api/data', authenticateApiKey, (req, res) => {
  // 権限チェック
  if (!req.apiKey.permissions.includes('read')) {
    return res.status(403).json({ error: 'Insufficient permissions' });
  }
  
  res.json({ data: 'Protected data' });
});

app.post('/api/data', authenticateApiKey, (req, res) => {
  if (!req.apiKey.permissions.includes('write')) {
    return res.status(403).json({ error: 'Insufficient permissions' });
  }
  
  res.json({ success: true });
});
```


### メリット
- ✅ **実装が簡単**: シンプルな認証方式
- ✅ **パフォーマンス**: 軽量で高速
- ✅ **デバッグが容易**: トラブルシューティングが簡単
- ✅ **B2B連携**: 企業間API連携に適している
- ✅ **レート制限**: 使用量制限の実装が容易

### デメリット
- ❌ **セキュリティリスク**: キーの漏洩リスクが高い
- ❌ **取り消しが困難**: キーの無効化が複雑
- ❌ **ユーザー体験**: キーの管理が煩雑
- ❌ **権限管理**: 細かい権限制御が困難
- ❌ **ログイン機能なし**: ユーザー認証機能がない

### 使用場面
- 🔗 **API サービス**: 外部向けAPI
- 🤖 **Webhook**: システム間連携
- 📊 **データ連携**: B2Bデータ交換

## 5. 多要素認証（MFA）

### 概要
複数の認証要素を組み合わせた高セキュリティ認証方式

### 実装例

```javascript
// バックエンド（Node.js + Express + speakeasy）
const speakeasy = require('speakeasy');
const QRCode = require('qrcode');

// TOTP（Time-based One-Time Password）設定
app.post('/api/auth/setup-mfa', requireAuth, async (req, res) => {
  try {
    const user = req.user;
    
    // シークレットキー生成
    const secret = speakeasy.generateSecret({
      name: `MyApp (${user.email})`,
      issuer: 'MyApp'
    });
    
    // QRコード生成
    const qrCodeUrl = await QRCode.toDataURL(secret.otpauth_url);
    
    // 一時的にシークレットを保存（検証後に永続化）
    user.tempMfaSecret = secret.base32;
    await user.save();
    
    res.json({
      secret: secret.base32,
      qrCode: qrCodeUrl,
      manualEntryKey: secret.base32
    });
  } catch (error) {
    res.status(500).json({ error: 'MFA setup failed' });
  }
});

// MFA検証
app.post('/api/auth/verify-mfa', requireAuth, async (req, res) => {
  try {
    const { token } = req.body;
    const user = req.user;
    
    // 一時シークレットまたは永続シークレットを使用
    const secret = user.tempMfaSecret || user.mfaSecret;
    
    if (!secret) {
      return res.status(400).json({ error: 'MFA not configured' });
    }
    
    // トークン検証
    const verified = speakeasy.totp.verify({
      secret: secret,
      encoding: 'base32',
      token: token,
      window: 2 // 2分の許容範囲
    });
    
    if (verified) {
      // 初回検証の場合は永続化
      if (user.tempMfaSecret) {
        user.mfaSecret = user.tempMfaSecret;
        user.tempMfaSecret = undefined;
        user.mfaEnabled = true;
        await user.save();
      }
      
      res.json({ success: true, verified: true });
    } else {
      res.status(400).json({ error: 'Invalid MFA token' });
    }
  } catch (error) {
    res.status(500).json({ error: 'MFA verification failed' });
  }
});

// MFA有効化チェック
const requireMFA = async (req, res, next) => {
  const user = req.user;
  
  if (user.mfaEnabled) {
    const mfaVerified = req.session.mfaVerified;
    
    if (!mfaVerified) {
      return res.status(403).json({ 
        error: 'MFA verification required',
        requiresMFA: true 
      });
    }
  }
  
  next();
};
```

### メリット
- ✅ **最高レベルのセキュリティ**: 複数の認証要素で強固な保護
- ✅ **規制対応**: 金融機関等の規制要件に対応
- ✅ **アカウント保護**: パスワード漏洩時の被害を最小化
- ✅ **柔軟な実装**: 複数の認証方式を組み合わせ可能
- ✅ **監査対応**: セキュリティ監査で高評価

### デメリット
- ❌ **実装が複雑**: 複数の認証システムの統合が必要
- ❌ **ユーザー体験**: 認証手順が煩雑になる
- ❌ **コスト**: SMS送信等の外部サービス費用
- ❌ **サポート負荷**: ユーザーの認証トラブル対応
- ❌ **デバイス依存**: 認証アプリやスマートフォンが必要

### 使用場面
- 🏦 **金融機関**: 銀行、証券会社のシステム
- 🏢 **企業システム**: 機密情報を扱うシステム

```javascript
// ハイブリッド認証：JWTとセッションを併用
const hybridAuth = (req, res, next) => {
  // まずJWTトークンをチェック
  const authHeader = req.headers['authorization'];
  const token = authHeader && authHeader.split(' ')[1];
  
  if (token) {
    jwt.verify(token, process.env.JWT_SECRET, (err, user) => {
      if (!err) {
        req.user = user;
        return next();
      }
    });
  }
  
  // JWTが無効または存在しない場合、セッションをチェック
  if (req.session && req.session.isAuthenticated) {
    req.user = { id: req.session.userId, email: req.session.email };
    return next();
  }
  
  res.status(401).json({ error: 'Authentication required' });
};
```

```javascript
  const minLength = 8;
  const hasUpperCase = /[A-Z]/.test(password);
  const hasLowerCase = /[a-z]/.test(password);
  const hasNumbers = /\d/.test(password);
  const hasSpecialChar = /[!@#$%^&*(),.?":{}|<>]/.test(password);
  
  return {
    isValid: password.length >= minLength && 
             hasUpperCase && hasLowerCase && 
             hasNumbers && hasSpecialChar,
    errors: [
      password.length < minLength && 'Password must be at least 8 characters',
      !hasUpperCase && 'Password must contain uppercase letter',
      !hasLowerCase && 'Password must contain lowercase letter',
      !hasNumbers && 'Password must contain number',
      !hasSpecialChar && 'Password must contain special character'
    ].filter(Boolean)
  };
};
```

## まとめ

| 認証方式 | 適用場面 | 主なメリット | 主なデメリット |
|---------|---------|-------------|---------------|
| **JWT** | SPA、モバイルアプリ | ステートレス、スケーラブル | 取り消し困難、実装複雑 |
| **セッション管理** | 従来のWebアプリ | セキュリティ高い、直感的 | ステートフル、スケーラビリティ低 |
| **OAuth 2.0** | サードパーティ連携 | UX良い、実装簡素化 | 外部依存、プライバシー懸念 |
| **API Key** | API サービス、B2B | 実装簡単、パフォーマンス良い | セキュリティリスク、権限管理困難 |
| **多要素認証** | 高セキュリティシステム | 最高セキュリティ、規制対応 | 実装複雑、UX悪化 |

### 選択の指針

1. **一般的なWebアプリ** → セッション管理
2. **SPA・モバイルアプリ** → JWT
3. **ソーシャルログイン** → OAuth 2.0
4. **API サービス** → API Key
5. **高セキュリティシステム** → 多要素認証
6. **複数の要件** → 組み合わせパターン
