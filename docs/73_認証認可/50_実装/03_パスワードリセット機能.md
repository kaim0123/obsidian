# パスワードリセット機能の実装方法とベストプラクティス

## 概要

パスワードリセット機能は、ユーザーがパスワードを忘れた場合に安全にパスワードを再設定できる重要なセキュリティ機能です。適切な実装により、アカウントのセキュリティを維持しながら、ユーザビリティを向上させることができます。

## 機能一覧

### 基本機能
- **パスワードリセット要求** - メールアドレスによるリセット要求
- **リセットトークン生成** - 安全なトークン生成と有効期限管理
- **メール送信** - リセットリンクを含むメール送信
- **パスワード更新** - 新しいパスワードの設定
- **トークン無効化** - 使用済みトークンの即座無効化

### 高度な機能
- **リセット試行回数制限** - ブルートフォース攻撃対策
- **IP制限** - 同一IPからの過度な要求制限
- **セキュリティログ** - リセット試行の記録
- **複数デバイス対応** - 全デバイスからのログアウト

```javascript
// バックエンド（Node.js + Express）
const crypto = require('crypto');
const bcrypt = require('bcrypt');
const nodemailer = require('nodemailer');

// パスワードリセット要求
app.post('/api/auth/forgot-password', async (req, res) => {
  try {
    const { email } = req.body;
    
    // 入力値検証
    if (!email || !validator.isEmail(email)) {
      return res.status(400).json({ error: 'Valid email address is required' });
    }
    
    // レート制限チェック
    const rateLimitKey = `forgot_password:${req.ip}:${email}`;
    const attempts = await redis.get(rateLimitKey);
    
    if (attempts && parseInt(attempts) >= 3) {
      return res.status(429).json({ 
        error: 'Too many password reset attempts. Please try again later.' 
      });
    }
    
    // ユーザー検索
    const user = await User.findOne({ email: email.toLowerCase() });
    
    // セキュリティ上の理由で、ユーザーが存在しない場合でも成功レスポンスを返す
    if (!user) {
      // レート制限カウンターを増加
      await redis.incr(rateLimitKey);
      await redis.expire(rateLimitKey, 3600); // 1時間有効
      
      return res.json({ 
        success: true, 
        message: 'If an account with that email exists, a password reset link has been sent.' 
      });
    }
    
    // 既存のリセットトークンを無効化
    await PasswordResetToken.deleteMany({ userId: user.id });
    
    // 新しいリセットトークン生成
    const resetToken = crypto.randomBytes(32).toString('hex');
    const hashedToken = crypto.createHash('sha256').update(resetToken).digest('hex');
    
    // トークンをデータベースに保存
    await PasswordResetToken.create({
      token: hashedToken,
      userId: user.id,
      expiresAt: new Date(Date.now() + 15 * 60 * 1000), // 15分有効
      ipAddress: req.ip,
      userAgent: req.get('User-Agent')
    });
    
    // リセットメール送信
    await sendPasswordResetEmail(email, resetToken, user.name);
    
    // レート制限カウンターを増加
    await redis.incr(rateLimitKey);
    await redis.expire(rateLimitKey, 3600);
    
    // セキュリティログ記録
    await SecurityLog.create({
      userId: user.id,
      action: 'password_reset_requested',
      ipAddress: req.ip,
      userAgent: req.get('User-Agent'),
      details: { email: user.email }
    });
    
    res.json({ 
      success: true, 
      message: 'If an account with that email exists, a password reset link has been sent.' 
    });
    
  } catch (error) {
    console.error('Password reset request error:', error);
    res.status(500).json({ error: 'Password reset request failed' });
  }
});

// パスワードリセット実行
app.post('/api/auth/reset-password', async (req, res) => {
  try {
    const { token, password } = req.body;
    
    // 入力値検証
    if (!token || !password) {
      return res.status(400).json({ 
        error: 'Reset token and new password are required' 
      });
    }
    
    // パスワード強度チェック
    const passwordValidation = validatePassword(password);
    if (!passwordValidation.isValid) {
      return res.status(400).json({ 
        error: 'Password does not meet requirements',
        details: passwordValidation.errors 
      });
    }
    
    // トークンをハッシュ化
    const hashedToken = crypto.createHash('sha256').update(token).digest('hex');
    
    // トークン検索
    const resetToken = await PasswordResetToken.findOne({
      token: hashedToken,
      expiresAt: { $gt: new Date() },
      used: false
    }).populate('userId');
    
    if (!resetToken) {
      return res.status(400).json({ 
        error: 'Invalid or expired reset token' 
      });
    }
    
    const user = resetToken.userId;
    
    // 新しいパスワードをハッシュ化
    const hashedPassword = await bcrypt.hash(password, 12);
    
    // ユーザーのパスワードを更新
    user.password = hashedPassword;
    user.passwordChangedAt = new Date();
    await user.save();
    
    // トークンを無効化
    resetToken.used = true;
    resetToken.usedAt = new Date();
    await resetToken.save();
    
    // 全デバイスからのログアウト（セッション無効化）
    await Session.deleteMany({ userId: user.id });
    await RefreshToken.deleteMany({ userId: user.id });
    
    // セキュリティログ記録
    await SecurityLog.create({
      userId: user.id,
      action: 'password_reset_completed',
      ipAddress: req.ip,
      userAgent: req.get('User-Agent'),
      details: { 
        email: user.email,
        resetTokenId: resetToken.id 
      }
    });
    
    res.json({ 
      success: true, 
      message: 'Password has been reset successfully. Please log in with your new password.' 
    });
    
  } catch (error) {
    console.error('Password reset error:', error);
    res.status(500).json({ error: 'Password reset failed' });
  }
});

// パスワード強度検証
const validatePassword = (password) => {
  const errors = [];
  
  if (password.length < 8) {
    errors.push('Password must be at least 8 characters long');
  }
  
  if (password.length > 128) {
    errors.push('Password must be less than 128 characters long');
  }
  
  if (!/[a-z]/.test(password)) {
    errors.push('Password must contain at least one lowercase letter');
  }
  
  if (!/[A-Z]/.test(password)) {
    errors.push('Password must contain at least one uppercase letter');
  }
  
  if (!/\d/.test(password)) {
    errors.push('Password must contain at least one number');
  }
  
  if (!/[@$!%*?&]/.test(password)) {
    errors.push('Password must contain at least one special character (@$!%*?&)');
  }
  
  // 一般的なパスワードチェック
  const commonPasswords = ['password', '123456', 'qwerty', 'abc123', 'password123'];
  if (commonPasswords.includes(password.toLowerCase())) {
    errors.push('Password is too common. Please choose a more secure password');
  }
  
  return {
    isValid: errors.length === 0,
    errors
  };
};
```

```javascript
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ token, password })
      });

      const data = await response.json();

        setTimeout(() => {

      } else {
        setError(data.error);
      }

      setIsLoading(false);
    }
  };

  if (!isTokenValid && !error) {
```

### レート制限とブルートフォース対策

```javascript
// Redis を使用したレート制限
const redis = require('redis');
const client = redis.createClient(process.env.REDIS_URL);

// IP別レート制限
const ipRateLimit = async (req, res, next) => {
  const ip = req.ip;
  const key = `rate_limit:forgot_password:${ip}`;
  
  try {
    const attempts = await client.get(key);
    const maxAttempts = 5;
    const windowMs = 15 * 60 * 1000; // 15分
    
    if (attempts && parseInt(attempts) >= maxAttempts) {
      return res.status(429).json({
        error: 'Too many password reset attempts from this IP. Please try again later.',
        retryAfter: Math.ceil(windowMs / 1000)
      });
    }
    
    // カウンターを増加
    await client.incr(key);
    await client.expire(key, Math.ceil(windowMs / 1000));
    
    next();
  } catch (error) {
    console.error('Rate limit error:', error);
    next(); // エラーの場合は制限をスキップ
  }
};

// メール別レート制限
const emailRateLimit = async (req, res, next) => {
  const { email } = req.body;
  const key = `rate_limit:forgot_password:${email}`;
  
  try {
    const attempts = await client.get(key);
    const maxAttempts = 3;
    const windowMs = 60 * 60 * 1000; // 1時間
    
    if (attempts && parseInt(attempts) >= maxAttempts) {
      return res.status(429).json({
        error: 'Too many password reset attempts for this email. Please try again later.',
        retryAfter: Math.ceil(windowMs / 1000)
      });
    }
    
    await client.incr(key);
    await client.expire(key, Math.ceil(windowMs / 1000));
    
    next();
  } catch (error) {
    console.error('Email rate limit error:', error);
    next();
  }
};
```

```javascript
  details: mongoose.Schema.Types.Mixed,

// 異常な活動の検出
const detectSuspiciousActivity = async (userId, ipAddress) => {
  const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000);
  
  const recentAttempts = await SecurityLog.countDocuments({
    userId,
    action: 'password_reset_requested',
    timestamp: { $gt: oneHourAgo }
  });
  
  if (recentAttempts >= 3) {
    // 管理者に通知
    await notifyAdministrators({
      type: 'suspicious_password_reset',
      userId,
      ipAddress,
      attempts: recentAttempts
    });
    
    // アカウントを一時的にロック
    await User.findByIdAndUpdate(userId, {
      status: 'temporarily_locked',
      lockedUntil: new Date(Date.now() + 24 * 60 * 60 * 1000) // 24時間
    });
  }
```
