## 1. 基本コメント機能

### 概要
シンプルなコメント投稿・表示機能

### データベース設計

```sql
-- コメントテーブル
CREATE TABLE comments (
  id INT PRIMARY KEY AUTO_INCREMENT,
  content TEXT NOT NULL,
  user_id INT NOT NULL,
  post_id INT NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  is_deleted BOOLEAN DEFAULT FALSE,
  
  FOREIGN KEY (user_id) REFERENCES users(id),
  FOREIGN KEY (post_id) REFERENCES posts(id),
  INDEX idx_post_id (post_id),
  INDEX idx_user_id (user_id),
  INDEX idx_created_at (created_at)
);
```

### 実装例

```javascript
// コメントコンポーネント
function CommentSection({ postId }) {
  const [comments, setComments] = useState([]);
  const [newComment, setNewComment] = useState('');
  const [loading, setLoading] = useState(false);
  const [user] = useAuth();

  // コメントの取得
  const fetchComments = useCallback(async () => {
    try {
      const response = await fetch(`/api/posts/${postId}/comments`);
      const data = await response.json();
      setComments(data);
    } catch (error) {
      console.error('コメント取得エラー:', error);
    }
  }, [postId]);

  // コメントの投稿
  const submitComment = async (e) => {
    e.preventDefault();
    if (!newComment.trim() || !user) return;

    setLoading(true);
    try {
      const response = await fetch(`/api/posts/${postId}/comments`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${user.token}`
        },
        body: JSON.stringify({
          content: newComment.trim()
        })
      });

      if (response.ok) {
        const newCommentData = await response.json();
        setComments(prev => [newCommentData, ...prev]);
        setNewComment('');
      }
    } catch (error) {
      console.error('コメント投稿エラー:', error);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchComments();
  }, [fetchComments]);

  return (
    <div className="comment-section">
      <h3>コメント ({comments.length})</h3>
      
      {/* コメント投稿フォーム */}
      {user && (
        <form onSubmit={submitComment} className="comment-form">
          <textarea
            value={newComment}
            onChange={(e) => setNewComment(e.target.value)}
            placeholder="コメントを入力..."
            rows={3}
            maxLength={1000}
            required
          />
          <div className="comment-form-actions">
            <span className="char-count">
              {newComment.length}/1000
            </span>
            <button 
              type="submit" 
              disabled={loading || !newComment.trim()}
              className="submit-button"
            >
              {loading ? '投稿中...' : 'コメント投稿'}
            </button>
          </div>
        </form>
      )}

      {/* コメント一覧 */}
      <div className="comments-list">
        {comments.map(comment => (
          <CommentItem 
            key={comment.id} 
            comment={comment}
            onUpdate={fetchComments}
          />
        ))}
      </div>
    </div>
  );
}

// 個別コメントアイテム
function CommentItem({ comment, onUpdate }) {
  const [isEditing, setIsEditing] = useState(false);
  const [editContent, setEditContent] = useState(comment.content);
  const [user] = useAuth();
  const isOwner = user && user.id === comment.user_id;

  const handleEdit = async () => {
    if (!editContent.trim()) return;

    try {
      const response = await fetch(`/api/comments/${comment.id}`, {
        method: 'PUT',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${user.token}`
        },
        body: JSON.stringify({
          content: editContent.trim()
        })
      });

      if (response.ok) {
        setIsEditing(false);
        onUpdate();
      }
    } catch (error) {
      console.error('コメント編集エラー:', error);
    }
  };

  const handleDelete = async () => {
    if (!confirm('このコメントを削除しますか？')) return;

    try {
      const response = await fetch(`/api/comments/${comment.id}`, {
        method: 'DELETE',
        headers: {
          'Authorization': `Bearer ${user.token}`
        }
      });

      if (response.ok) {
        onUpdate();
      }
    } catch (error) {
      console.error('コメント削除エラー:', error);
    }
  };

  return (
    <div className="comment-item">
      <div className="comment-header">
        <div className="comment-author">
          <img 
            src={comment.user.avatar || '/default-avatar.png'} 
            alt={comment.user.name}
            className="avatar"
          />
          <span className="author-name">{comment.user.name}</span>
        </div>
        <div className="comment-meta">
          <span className="comment-date">
            {new Date(comment.created_at).toLocaleString('ja-JP')}
          </span>
          {comment.updated_at !== comment.created_at && (
            <span className="edited-indicator">（編集済み）</span>
          )}
        </div>
      </div>

      <div className="comment-content">
        {isEditing ? (
          <div className="edit-form">
            <textarea
              value={editContent}
              onChange={(e) => setEditContent(e.target.value)}
              rows={3}
              maxLength={1000}
            />
            <div className="edit-actions">
              <button onClick={handleEdit} className="save-button">
                保存
              </button>
              <button 
                onClick={() => {
                  setIsEditing(false);
                  setEditContent(comment.content);
                }}
                className="cancel-button"
              >
                キャンセル
              </button>
            </div>
          </div>
        ) : (
          <p className="comment-text">{comment.content}</p>
        )}
      </div>

      {isOwner && !isEditing && (
        <div className="comment-actions">
          <button 
            onClick={() => setIsEditing(true)}
            className="edit-button"
          >
            編集
          </button>
          <button 
            onClick={handleDelete}
            className="delete-button"
          >
            削除
          </button>
        </div>
      )}
    </div>
  );
}
```

### メリット
- ✅ **実装が簡単**: 基本的なCRUD操作で実装可能
- ✅ **軽量**: 最小限のリソースで動作
- ✅ **理解しやすい**: ユーザーにとって直感的
- ✅ **高速**: シンプルな構造で高速表示

### デメリット
- ❌ **機能が限定的**: 返信やリアクション機能がない
- ❌ **ユーザビリティ**: 長い議論が読みにくい
- ❌ **エンゲージメント**: ユーザーの参加度が低い

### 使用場面
- 📝 **ブログ**: 記事への感想・意見
- 📰 **ニュースサイト**: 記事へのコメント
- 🎯 **シンプルな交流**: 基本的な意見交換

## 2. 返信機能付きコメント

### 概要
コメントに対する返信機能を追加した階層構造のコメントシステム

### データベース設計

```sql
-- 返信機能付きコメントテーブル
CREATE TABLE comments (
  id INT PRIMARY KEY AUTO_INCREMENT,
  content TEXT NOT NULL,
  user_id INT NOT NULL,
  post_id INT NOT NULL,
  parent_id INT NULL, -- 返信の場合は親コメントのID
  depth INT DEFAULT 0, -- 階層の深さ
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  is_deleted BOOLEAN DEFAULT FALSE,
  
  FOREIGN KEY (user_id) REFERENCES users(id),
  FOREIGN KEY (post_id) REFERENCES posts(id),
  FOREIGN KEY (parent_id) REFERENCES comments(id),
  INDEX idx_post_id (post_id),
  INDEX idx_parent_id (parent_id),
  INDEX idx_depth (depth)
);
```

### 実装例

```javascript
// 階層構造コメントコンポーネント
function ThreadedCommentSection({ postId }) {
  const [comments, setComments] = useState([]);
  const [newComment, setNewComment] = useState('');
  const [replyingTo, setReplyingTo] = useState(null);
  const [replyContent, setReplyContent] = useState('');
  const [user] = useAuth();

  // コメントの取得（階層構造で整理）
  const fetchComments = useCallback(async () => {
    try {
      const response = await fetch(`/api/posts/${postId}/comments?threaded=true`);
      const data = await response.json();
      setComments(data);
    } catch (error) {
      console.error('コメント取得エラー:', error);
    }
  }, [postId]);

  // 返信の投稿
  const submitReply = async (parentId) => {
    if (!replyContent.trim() || !user) return;

    try {
      const response = await fetch(`/api/posts/${postId}/comments`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${user.token}`
        },
        body: JSON.stringify({
          content: replyContent.trim(),
          parent_id: parentId
        })
      });

      if (response.ok) {
        setReplyContent('');
        setReplyingTo(null);
        fetchComments();
      }
    } catch (error) {
      console.error('返信投稿エラー:', error);
    }
  };

  // コメントツリーの構築
  const buildCommentTree = (comments) => {
    const commentMap = new Map();
    const rootComments = [];

    // コメントをマップに格納
    comments.forEach(comment => {
      commentMap.set(comment.id, { ...comment, replies: [] });
    });

    // 階層構造を構築
    comments.forEach(comment => {
      if (comment.parent_id) {
        const parent = commentMap.get(comment.parent_id);
        if (parent) {
          parent.replies.push(commentMap.get(comment.id));
        }
      } else {
        rootComments.push(commentMap.get(comment.id));
      }
    });

    return rootComments;
  };

  const commentTree = useMemo(() => buildCommentTree(comments), [comments]);

  useEffect(() => {
    fetchComments();
  }, [fetchComments]);

  return (
    <div className="threaded-comment-section">
      <h3>コメント ({comments.length})</h3>
      
      {/* 新規コメント投稿フォーム */}
      {user && (
        <form onSubmit={(e) => {
          e.preventDefault();
          submitReply(null);
        }} className="comment-form">
          <textarea
            value={newComment}
            onChange={(e) => setNewComment(e.target.value)}
            placeholder="コメントを入力..."
            rows={3}
            maxLength={1000}
            required
          />
          <button type="submit" disabled={!newComment.trim()}>
            コメント投稿
          </button>
        </form>
      )}

      {/* 階層構造コメント一覧 */}
      <div className="comments-tree">
        {commentTree.map(comment => (
          <ThreadedCommentItem
            key={comment.id}
            comment={comment}
            onReply={setReplyingTo}
            onUpdate={fetchComments}
            user={user}
          />
        ))}
      </div>
    </div>
  );
}
```

## 3. リアクション機能付きコメント

### 概要
コメントに対する感情表現（いいね、面白い、悲しいなど）を追加

### データベース設計

```sql
-- リアクションテーブル
CREATE TABLE comment_reactions (
  id INT PRIMARY KEY AUTO_INCREMENT,
  comment_id INT NOT NULL,
  user_id INT NOT NULL,
  reaction_type ENUM('like', 'love', 'laugh', 'angry', 'sad', 'wow') NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (comment_id) REFERENCES comments(id) ON DELETE CASCADE,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  UNIQUE KEY unique_user_comment_reaction (user_id, comment_id, reaction_type),
  INDEX idx_comment_id (comment_id),
  INDEX idx_user_id (user_id)
);
```

### 実装例

```javascript
// リアクション機能付きコメント
function ReactionComment({ comment, user, onUpdate }) {
  const [reactions, setReactions] = useState(comment.reactions || {});
  const [userReaction, setUserReaction] = useState(comment.user_reaction || null);

  const reactionTypes = {
    like: { emoji: '👍', label: 'いいね' },
    love: { emoji: '❤️', label: 'いいね' },
    laugh: { emoji: '😂', label: '面白い' },
    angry: { emoji: '😠', label: '怒り' },
    sad: { emoji: '😢', label: '悲しい' },
    wow: { emoji: '😮', label: '驚き' }
  };

  const handleReaction = async (reactionType) => {
    if (!user) return;

    try {
      const response = await fetch(`/api/comments/${comment.id}/reactions`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${user.token}`
        },
        body: JSON.stringify({
          reaction_type: reactionType
        })
      });

      if (response.ok) {
        const data = await response.json();
        setReactions(data.reactions);
        setUserReaction(data.user_reaction);
        onUpdate();
      }
    } catch (error) {
      console.error('リアクションエラー:', error);
    }
  };

  const removeReaction = async () => {
    if (!user || !userReaction) return;

    try {
      const response = await fetch(`/api/comments/${comment.id}/reactions`, {
        method: 'DELETE',
        headers: {
          'Authorization': `Bearer ${user.token}`
        }
      });

      if (response.ok) {
        const data = await response.json();
        setReactions(data.reactions);
        setUserReaction(null);
        onUpdate();
      }
    } catch (error) {
      console.error('リアクション削除エラー:', error);
    }
  };

  return (
    <div className="reaction-comment">
      <div className="comment-content">
        <p>{comment.content}</p>
      </div>

      <div className="comment-reactions">
        <div className="reaction-buttons">
          {Object.entries(reactionTypes).map(([type, config]) => (
            <button
              key={type}
              onClick={() => {
                if (userReaction === type) {
                  removeReaction();
                } else {
                  handleReaction(type);
                }
              }}
              className={`reaction-button ${userReaction === type ? 'active' : ''}`}
              title={config.label}
            >
              <span className="emoji">{config.emoji}</span>
              <span className="count">{reactions[type] || 0}</span>
            </button>
          ))}
        </div>

        {userReaction && (
          <button 
            onClick={removeReaction}
            className="remove-reaction"
            title="リアクションを削除"
          >
            ✕
          </button>
        )}
      </div>
    </div>
  );
}

// リアクション統計表示
function ReactionStats({ reactions }) {
  const totalReactions = Object.values(reactions).reduce((sum, count) => sum + count, 0);
  
  if (totalReactions === 0) return null;

  return (
    <div className="reaction-stats">
      <span className="total-count">{totalReactions}件のリアクション</span>
      <div className="reaction-breakdown">
        {Object.entries(reactions)
          .filter(([_, count]) => count > 0)
          .map(([type, count]) => (
            <span key={type} className="reaction-item">
              {getReactionEmoji(type)} {count}
            </span>
          ))
        }
      </div>
    </div>
  );
}

function getReactionEmoji(type) {
  const emojis = {
    like: '👍',
    love: '❤️',
    laugh: '😂',
    angry: '😠',
    sad: '😢',
    wow: '😮'
  };
  return emojis[type] || '👍';
}
```

### 概要
不適切なコメントの管理・削除機能

### データベース設計

```sql
-- モデレーションテーブル
CREATE TABLE comment_moderations (
  id INT PRIMARY KEY AUTO_INCREMENT,
  comment_id INT NOT NULL,
  moderator_id INT NOT NULL,
  action ENUM('approve', 'reject', 'delete', 'flag') NOT NULL,
  reason TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (comment_id) REFERENCES comments(id) ON DELETE CASCADE,
  FOREIGN KEY (moderator_id) REFERENCES users(id),
  INDEX idx_comment_id (comment_id),
  INDEX idx_moderator_id (moderator_id)
);

-- フラグテーブル
CREATE TABLE comment_flags (
  id INT PRIMARY KEY AUTO_INCREMENT,
  comment_id INT NOT NULL,
  user_id INT NOT NULL,
  reason ENUM('spam', 'inappropriate', 'harassment', 'other') NOT NULL,
  description TEXT,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (comment_id) REFERENCES comments(id) ON DELETE CASCADE,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  UNIQUE KEY unique_user_comment_flag (user_id, comment_id),
  INDEX idx_comment_id (comment_id)
);
```

```javascript
        method: 'POST',

          'Content-Type': 'application/json',

          reason: flagReason,
          description: flagReason === 'other' ? prompt('詳細を入力してください:') : ''
        })

        setIsFlagged(true);
        setShowFlagForm(false);

      }
    } catch (error) {
      console.error('フラグエラー:', error);
    }
  };

  const isOwner = user && user.id === comment.user_id;

  return (
    <div className="moderated-comment">
      <div className="comment-content">

      </div>

            onClick={() => setShowFlagForm(true)}
            className="flag-button"
          >
            報告
          </button>
        )}

              承認
            </button>
```

```javascript
function PaginatedComments({ postId }) {
```

```javascript
// WebSocketを使ったリアルタイムコメント
function RealtimeComments({ postId }) {
  const [comments, setComments] = useState([]);
  const [ws, setWs] = useState(null);

  useEffect(() => {
    const websocket = new WebSocket(`ws://localhost:8080/comments/${postId}`);
    
    websocket.onmessage = (event) => {
      const data = JSON.parse(event.data);
      
      switch (data.type) {
        case 'new_comment':
          setComments(prev => [data.comment, ...prev]);
          break;
        case 'updated_comment':
          setComments(prev => prev.map(comment => 
            comment.id === data.comment.id ? data.comment : comment
          ));
          break;
        case 'deleted_comment':
          setComments(prev => prev.filter(comment => comment.id !== data.commentId));
          break;
      }
    };

    setWs(websocket);

    return () => {
      websocket.close();
    };
  }, [postId]);

  return (
    <div className="realtime-comments">
      {comments.map(comment => (
        <CommentItem key={comment.id} comment={comment} />
      ))}
    </div>
  );
}
```

### 3. キャッシュの実装

```javascript
// コメントのキャッシュ
const commentCache = new Map();

function CachedComments({ postId }) {
  const [comments, setComments] = useState([]);

  const fetchComments = useCallback(async () => {
    // キャッシュから取得
    if (commentCache.has(postId)) {
      setComments(commentCache.get(postId));
      return;
    }

    try {
      const response = await fetch(`/api/posts/${postId}/comments`);
      const data = await response.json();
      
      // キャッシュに保存
      commentCache.set(postId, data);
      setComments(data);
    } catch (error) {
      console.error('コメント取得エラー:', error);
    }
  }, [postId]);

  useEffect(() => {
    fetchComments();
  }, [fetchComments]);

  return (
    <div>
      {comments.map(comment => (
        <CommentItem key={comment.id} comment={comment} />
      ))}
    </div>
  );
}
```
