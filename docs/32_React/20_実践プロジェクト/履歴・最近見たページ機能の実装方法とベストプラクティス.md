## 履歴・最近見たページとは

履歴・最近見たページ機能は、ユーザーが過去に訪問したページの記録を保持し、簡単に再アクセスできるようにする機能です。ブラウザの履歴機能と同様の機能をWebアプリケーション内で提供します。

### 主な特徴
- **自動記録**: ページ訪問時に自動的に履歴に追加
- **時系列表示**: 訪問日時の順序で表示
- **検索機能**: 履歴内での検索・フィルタリング
- **プライバシー保護**: ユーザーが履歴を削除可能

## フロントエンド実装例

### React での実装
```jsx
import React, { useState, useEffect, useCallback } from 'react';

const HistoryManager = () => {
  const [history, setHistory] = useState([]);
  const [filteredHistory, setFilteredHistory] = useState([]);
  const [searchTerm, setSearchTerm] = useState('');
  const [loading, setLoading] = useState(true);
  const [page, setPage] = useState(1);
  const [hasMore, setHasMore] = useState(true);

  // 履歴の取得
  const fetchHistory = useCallback(async (pageNum = 1, search = '') => {
    try {
      setLoading(true);
      const response = await fetch(
        `/api/history?page=${pageNum}&search=${encodeURIComponent(search)}`
      );
      const data = await response.json();
      
      if (pageNum === 1) {
        setHistory(data.items);
      } else {
        setHistory(prev => [...prev, ...data.items]);
      }
      
      setHasMore(data.hasMore);
    } catch (error) {
      console.error('履歴の取得に失敗:', error);
    } finally {
      setLoading(false);
    }
  }, []);

  // 初期読み込み
  useEffect(() => {
    fetchHistory();
  }, [fetchHistory]);

  // 検索フィルタリング
  useEffect(() => {
    if (searchTerm) {
      const filtered = history.filter(item =>
        item.title.toLowerCase().includes(searchTerm.toLowerCase()) ||
        item.url.toLowerCase().includes(searchTerm.toLowerCase())
      );
      setFilteredHistory(filtered);
    } else {
      setFilteredHistory(history);
    }
  }, [history, searchTerm]);

  // 履歴の追加（ページ訪問時に呼び出し）
  const addToHistory = async (pageData) => {
    try {
      await fetch('/api/history', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(pageData)
      });
      
      // ローカル状態を更新
      setHistory(prev => [pageData, ...prev]);
    } catch (error) {
      console.error('履歴の追加に失敗:', error);
    }
  };

  // 履歴アイテムの削除
  const removeFromHistory = async (historyId) => {
    try {
      await fetch(`/api/history/${historyId}`, {
        method: 'DELETE'
      });
      
      setHistory(prev => prev.filter(item => item.id !== historyId));
    } catch (error) {
      console.error('履歴の削除に失敗:', error);
    }
  };

  // 履歴のクリア
  const clearHistory = async () => {
    if (!window.confirm('すべての履歴を削除しますか？')) return;
    
    try {
      await fetch('/api/history', {
        method: 'DELETE'
      });
      
      setHistory([]);
      setFilteredHistory([]);
    } catch (error) {
      console.error('履歴のクリアに失敗:', error);
    }
  };

  // さらに読み込み
  const loadMore = () => {
    const nextPage = page + 1;
    setPage(nextPage);
    fetchHistory(nextPage, searchTerm);
  };

  // 検索ハンドラー
  const handleSearch = (e) => {
    setSearchTerm(e.target.value);
  };

  return (
    <div className="history-section">
      <div className="history-header">
        <h2>最近見たページ ({filteredHistory.length}件)</h2>
        <div className="history-controls">
          <input
            type="text"
            className="history-search"
            placeholder="履歴を検索..."
            value={searchTerm}
            onChange={handleSearch}
          />
          <button className="clear-history-btn" onClick={clearHistory}>
            履歴をクリア
          </button>
        </div>
      </div>

      {loading && history.length === 0 ? (
        <div className="loading">読み込み中...</div>
      ) : filteredHistory.length === 0 ? (
        <div className="empty-message">
          {searchTerm ? '検索結果が見つかりません' : '履歴がありません'}
        </div>
      ) : (
        <>
          <div className="history-list">
            {filteredHistory.map(item => (
              <HistoryItem
                key={item.id}
                item={item}
                onRemove={removeFromHistory}
              />
            ))}
          </div>
          
          {hasMore && !searchTerm && (
            <div className="history-pagination">
              <button
                className="load-more-btn"
                onClick={loadMore}
                disabled={loading}
              >
                {loading ? '読み込み中...' : 'さらに表示'}
              </button>
            </div>
          )}
        </>
      )}
    </div>
  );
};

// 履歴アイテムコンポーネント
const HistoryItem = ({ item, onRemove }) => {
  const formatTime = (timestamp) => {
    const date = new Date(timestamp);
    const now = new Date();
    const diff = now - date;
    
    if (diff < 60000) return 'たった今';
    if (diff < 3600000) return `${Math.floor(diff / 60000)}分前`;
    if (diff < 86400000) return `${Math.floor(diff / 3600000)}時間前`;
    if (diff < 604800000) return `${Math.floor(diff / 86400000)}日前`;
    
    return date.toLocaleDateString();
  };

  const handleVisit = () => {
    window.location.href = item.url;
  };

  const handleRemove = () => {
    onRemove(item.id);
  };

  return (
    <div className="history-item">
      <div className="history-item-content">
        <div className="history-item-title">{item.title}</div>
        <div className="history-item-url">{item.url}</div>
        <div className="history-item-time">
          {formatTime(item.visitedAt)}
        </div>
      </div>
      <div className="history-item-actions">
        <button className="visit-btn" onClick={handleVisit}>
          訪問
        </button>
        <button className="remove-btn" onClick={handleRemove}>
          削除
        </button>
      </div>
    </div>
  );
};

// 履歴追跡のためのカスタムフック
export const useHistoryTracking = () => {
  const addToHistory = useCallback(async (pageData) => {
    try {
      await fetch('/api/history', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          title: document.title,
          url: window.location.href,
          ...pageData
        })
      });
    } catch (error) {
      console.error('履歴の追加に失敗:', error);
    }
  }, []);

  return { addToHistory };
};

// 使用例
function ProductPage({ product }) {
  const { addToHistory } = useHistoryTracking();

  useEffect(() => {
    // ページ訪問時に履歴に追加
    addToHistory({
      title: product.name,
      url: window.location.href,
      category: 'product',
      itemId: product.id
    });
  }, [product, addToHistory]);

  return (
    <div>
      <h1>{product.name}</h1>
      <p>{product.description}</p>
    </div>
  );
}

export default HistoryManager;
```

### Node.js + Express での実装

```javascript
const express = require('express');
const router = express.Router();

// 履歴一覧の取得
router.get('/history', authenticateUser, async (req, res) => {
  try {
    const { page = 1, limit = 20, search = '' } = req.query;
    const userId = req.user.id;
    const offset = (page - 1) * limit;
    
    let whereClause = { userId };
    
    // 検索条件
    if (search) {
      whereClause = {
        ...whereClause,
        [Op.or]: [
          { title: { [Op.like]: `%${search}%` } },
          { url: { [Op.like]: `%${search}%` } }
        ]
      };
    }
    
    const { count, rows: history } = await History.findAndCountAll({
      where: whereClause,
      order: [['visitedAt', 'DESC']],
      limit: parseInt(limit),
      offset: parseInt(offset)
    });
    
    const hasMore = offset + history.length < count;
    
    res.json({
      items: history,
      total: count,
      page: parseInt(page),
      hasMore
    });
  } catch (error) {
    console.error('履歴取得エラー:', error);
    res.status(500).json({ error: 'サーバーエラーが発生しました' });
  }
});

// 履歴の追加
router.post('/history', authenticateUser, async (req, res) => {
  try {
    const { title, url, category, itemId } = req.body;
    const userId = req.user.id;
    
    // 重複チェック（同じURLの最近の履歴）
    const existingHistory = await History.findOne({
      where: {
        userId,
        url,
        visitedAt: {
          [Op.gte]: new Date(Date.now() - 5 * 60 * 1000) // 5分以内
        }
      }
    });
    
    if (existingHistory) {
      // 既存の履歴の訪問時刻を更新
      await existingHistory.update({ visitedAt: new Date() });
      return res.json(existingHistory);
    }
    
    // 新しい履歴を作成
    const history = await History.create({
      userId,
      title,
      url,
      category,
      itemId,
      visitedAt: new Date()
    });
    
    res.status(201).json(history);
  } catch (error) {
    console.error('履歴追加エラー:', error);
    res.status(500).json({ error: 'サーバーエラーが発生しました' });
  }
});

// 履歴の削除
router.delete('/history/:id', authenticateUser, async (req, res) => {
  try {
    const { id } = req.params;
    const userId = req.user.id;
    
    const deleted = await History.destroy({
      where: { id, userId }
    });
    
    if (deleted === 0) {
      return res.status(404).json({ error: '履歴が見つかりません' });
    }
    
    res.json({ message: '履歴を削除しました' });
  } catch (error) {
    console.error('履歴削除エラー:', error);
    res.status(500).json({ error: 'サーバーエラーが発生しました' });
  }
});

// 履歴のクリア
router.delete('/history', authenticateUser, async (req, res) => {
  try {
    const userId = req.user.id;
    
    await History.destroy({
      where: { userId }
    });
    
    res.json({ message: '履歴をクリアしました' });
  } catch (error) {
    console.error('履歴クリアエラー:', error);
    res.status(500).json({ error: 'サーバーエラーが発生しました' });
  }
});

// 履歴の統計情報
```

### MySQL テーブル設計
```sql
-- 履歴テーブル
CREATE TABLE history (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    title VARCHAR(255) NOT NULL,
    url TEXT NOT NULL,
    category VARCHAR(100),
    item_id VARCHAR(255),
    visited_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- インデックス
    INDEX idx_user_id (user_id),
    INDEX idx_visited_at (visited_at),
    INDEX idx_user_visited (user_id, visited_at),
    INDEX idx_category (category),
    
    -- 外部キー制約
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

-- 履歴統計テーブル（パフォーマンス向上のため）
CREATE TABLE history_stats (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    category VARCHAR(100),
    visit_count INT DEFAULT 1,
    last_visited TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    INDEX idx_user_category (user_id, category),
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    UNIQUE KEY unique_user_category (user_id, category)
);
```

## プライバシーとセキュリティ

### 1. データの暗号化
```javascript
// 機密情報の暗号化
const crypto = require('crypto');

class HistoryEncryption {
  constructor(secretKey) {
    this.algorithm = 'aes-256-gcm';
    this.secretKey = crypto.scryptSync(secretKey, 'salt', 32);
  }
  
  encrypt(text) {
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipher(this.algorithm, this.secretKey);
    cipher.setAAD(Buffer.from('history-data'));
    
    let encrypted = cipher.update(text, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    
    const authTag = cipher.getAuthTag();
    
    return {
      encrypted,
      iv: iv.toString('hex'),
      authTag: authTag.toString('hex')
    };
  }
  
  decrypt(encryptedData) {
    const decipher = crypto.createDecipher(
      this.algorithm, 
      this.secretKey
    );
    decipher.setAAD(Buffer.from('history-data'));
    decipher.setAuthTag(Buffer.from(encryptedData.authTag, 'hex'));
    
    let decrypted = decipher.update(encryptedData.encrypted, 'hex', 'utf8');
    decrypted += decipher.final('utf8');
    
    return decrypted;
  }
}
```

### 1. パフォーマンス最適化
```javascript
// バッチ処理での履歴追加
class HistoryBatchProcessor {
  constructor() {
    this.batch = [];
    this.batchSize = 10;
    this.flushInterval = 5000; // 5秒
    this.startFlushTimer();
  }
  
  addToBatch(historyData) {
    this.batch.push(historyData);
    
    if (this.batch.length >= this.batchSize) {
      this.flush();
    }
  }
  
  async flush() {
    if (this.batch.length === 0) return;
    
    try {
      await History.bulkCreate(this.batch);
      this.batch = [];
    } catch (error) {
      console.error('バッチ履歴追加エラー:', error);
    }
  }
  
  startFlushTimer() {
    setInterval(() => {
      this.flush();
    }, this.flushInterval);
  }
}
```

### 2. キャッシュ戦略
```javascript
// Redis を使った履歴キャッシュ
const redis = require('redis');
const client = redis.createClient();

class HistoryCache {
  constructor() {
    this.cacheExpiry = 300; // 5分
  }
  
  async getHistory(userId, page = 1) {
    const cacheKey = `history:${userId}:${page}`;
    
    try {
      const cached = await client.get(cacheKey);
      if (cached) {
        return JSON.parse(cached);
      }
    } catch (error) {
      console.error('キャッシュ取得エラー:', error);
    }
    
    return null;
  }
  
  async setHistory(userId, page, data) {
    const cacheKey = `history:${userId}:${page}`;
    
    try {
      await client.setex(cacheKey, this.cacheExpiry, JSON.stringify(data));
    } catch (error) {
      console.error('キャッシュ設定エラー:', error);
    }
  }
  
  async invalidateUserHistory(userId) {
    const pattern = `history:${userId}:*`;
    
    try {
      const keys = await client.keys(pattern);
      if (keys.length > 0) {
        await client.del(keys);
      }
    } catch (error) {
      console.error('キャッシュ無効化エラー:', error);
    }
  }
}
```

## よくある問題と解決方法

### 1. 重複履歴の問題
**問題**: 同じページが短時間で複数回記録される
**解決方法**:
```javascript
// 重複チェック機能
const checkDuplicateHistory = async (userId, url) => {
  const recentThreshold = 5 * 60 * 1000; // 5分
  
  const recentHistory = await History.findOne({
    where: {
      userId,
      url,
      visitedAt: {
        [Op.gte]: new Date(Date.now() - recentThreshold)
      }
    }
  });
  
  return !!recentHistory;
};
```

```javascript
// プライバシー設定の実装
const PrivacySettings = {
  // 履歴記録の有効/無効
  enableHistory: true,
  
  // 記録する情報の制限
  recordUserAgent: false,
  recordReferrer: false,

    '/admin/*',
    '/settings/*',
    '/payment/*'
  ]

// 履歴記録前のチェック
const shouldRecordHistory = (url) => {
  if (!PrivacySettings.enableHistory) return false;
  
  return !PrivacySettings.excludePatterns.some(pattern => {
    const regex = new RegExp(pattern.replace('*', '.*'));
    return regex.test(url);
  });
};
```

## まとめ

履歴・最近見たページ機能は、ユーザビリティを大幅に向上させる重要な機能です。適切な実装により：

- **ナビゲーションの改善**: ユーザーが過去のページに簡単に戻れる
- **ユーザーエクスペリエンスの向上**: 継続的な利用を促進
- **データ分析**: ユーザーの行動パターンを理解できる

実装時は、プライバシー保護、パフォーマンス、セキュリティを考慮し、ユーザーにとって価値のある履歴機能を提供しましょう。
