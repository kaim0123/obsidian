# 共有機能の実装方法とベストプラクティス

## 概要

共有機能は、ユーザーがコンテンツを他のユーザーや外部サービスと共有するための重要な機能です。基本的なURL共有から、ソーシャルメディア連携、埋め込み機能まで、様々な共有手法とその実装方法を詳しく解説します。

## 共有機能の種類と比較表

| 項目 | URL共有 | ソーシャルメディア連携 | 埋め込み機能 | ファイル共有 |
|------|---------|---------------------|-------------|-------------|
| **実装の複雑さ** | ⭐ 簡単 | ⭐⭐ 中程度 | ⭐⭐⭐ 複雑 | ⭐⭐⭐ 複雑 |
| **ユーザビリティ** | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い | ⭐⭐ 中程度 | ⭐⭐⭐ 高い |
| **拡散効果** | ⭐⭐ 中程度 | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い | ⭐⭐ 中程度 |
| **セキュリティ** | ⭐⭐⭐ 高い | ⭐⭐ 中程度 | ⭐⭐ 中程度 | ⭐⭐ 中程度 |
| **分析の価値** | ⭐⭐ 中程度 | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い | ⭐⭐ 中程度 |
| **メンテナンス** | ⭐⭐⭐ 簡単 | ⭐⭐ 中程度 | ⭐⭐ 中程度 | ⭐⭐ 中程度 |

### 推奨使用場面

| 共有タイプ | 最適な使用場面 | 例 |
|-----------|----------------|-----|
| **URL共有** | 基本的な共有 | 記事、商品、ページ |

## 1. URL共有機能

### 概要
基本的なURLコピー・共有機能

### 実装例

```javascript
// URL共有コンポーネント
function ShareButton({ url, title, description }) {
  const [isCopied, setIsCopied] = useState(false);
  const [showShareMenu, setShowShareMenu] = useState(false);

  const handleCopyUrl = async () => {
    try {
      await navigator.clipboard.writeText(url);
      setIsCopied(true);
      setTimeout(() => setIsCopied(false), 2000);
    } catch (error) {
      console.error('URLコピーエラー:', error);
      // フォールバック
      const textArea = document.createElement('textarea');
      textArea.value = url;
      document.body.appendChild(textArea);
      textArea.select();
      document.execCommand('copy');
      document.body.removeChild(textArea);
      setIsCopied(true);
      setTimeout(() => setIsCopied(false), 2000);
    }
  };

  const handleShare = async () => {
    if (navigator.share) {
      try {
        await navigator.share({
          title: title,
          text: description,
          url: url
        });
      } catch (error) {
        console.error('共有エラー:', error);
      }
    } else {
      setShowShareMenu(true);
    }
  };

  return (
    <div className="share-button">
      <button onClick={handleShare} className="share-main-button">
        <span className="share-icon">📤</span>
        共有
      </button>
      
      <button onClick={handleCopyUrl} className="copy-url-button">
        <span className="copy-icon">
          {isCopied ? '✅' : '📋'}
        </span>
        {isCopied ? 'コピー済み' : 'URLコピー'}
      </button>

      {showShareMenu && (
        <div className="share-menu">
          <div className="share-menu-header">
            <h3>共有</h3>
            <button onClick={() => setShowShareMenu(false)}>×</button>
          </div>
          
          <div className="share-options">
            <button onClick={handleCopyUrl} className="share-option">
              <span>📋</span>
              URLをコピー
            </button>
            
            <button 
              onClick={() => {
                const mailtoUrl = `mailto:?subject=${encodeURIComponent(title)}&body=${encodeURIComponent(description + '\n\n' + url)}`;
                window.open(mailtoUrl);
              }}
              className="share-option"
            >
              <span>📧</span>
              メールで共有
            </button>
            
            <button 
              onClick={() => {
                const smsUrl = `sms:?body=${encodeURIComponent(title + '\n' + url)}`;
                window.open(smsUrl);
              }}
              className="share-option"
            >
              <span>💬</span>
              SMSで共有
            </button>
          </div>
        </div>
      )}
    </div>
  );
}

// 共有統計コンポーネント
function ShareStats({ targetId, targetType }) {
  const [stats, setStats] = useState({
    totalShares: 0,
    shareCount: 0,
    clickCount: 0
  });
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const fetchStats = async () => {
      try {
        const response = await fetch(`/api/shares/stats?target_id=${targetId}&target_type=${targetType}`);

        setStats(data);
      } catch (error) {
```

```javascript
    try {
      await fetch('/api/shares/track', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({
          platform: platform,
          url: sharedUrl,
          timestamp: new Date().toISOString()
        })
      });
    } catch (error) {
      console.error('共有追跡エラー:', error);
    }
  };

  return (
    <div className="social-share">
      <button 
        onClick={() => setShowSocialMenu(!showSocialMenu)}
        className="social-share-button"
      >
        <span className="share-icon">📤</span>
        ソーシャルで共有
      </button>

        <div className="social-share-menu">
          <div className="social-share-header">

            <button onClick={() => setShowSocialMenu(false)}>×</button>

                key={platform.name}
                onClick={() => handleSocialShare(platform)}
                className="social-platform-button"

              >
                <span className="platform-icon">{platform.icon}</span>
```

## 3. 埋め込み機能

```javascript
// 埋め込みコード生成コンポーネント
function EmbedCode({ targetId, targetType, title, description }) {
  const [embedCode, setEmbedCode] = useState('');
  const [embedOptions, setEmbedOptions] = useState({
    width: 560,
    height: 315,
    autoplay: false,
    showTitle: true,
    showDescription: true
  });
  const [showEmbedMenu, setShowEmbedMenu] = useState(false);

  const generateEmbedCode = () => {
    const baseUrl = window.location.origin;
    const embedUrl = `${baseUrl}/embed/${targetType}/${targetId}`;
    
    let code = `<iframe src="${embedUrl}"`;
    code += ` width="${embedOptions.width}"`;
    code += ` height="${embedOptions.height}"`;
    code += ` frameborder="0"`;
    code += ` allowfullscreen`;
    code += `></iframe>`;
    
    if (embedOptions.showTitle) {
      code += `<p><strong>${title}</strong></p>`;
    }
    
    if (embedOptions.showDescription) {
      code += `<p>${description}</p>`;
    }
    
    setEmbedCode(code);
  };

  const copyEmbedCode = async () => {
    try {
      await navigator.clipboard.writeText(embedCode);
      alert('埋め込みコードをコピーしました');
    } catch (error) {
      console.error('埋め込みコードコピーエラー:', error);
    }
  };

  useEffect(() => {
    generateEmbedCode();
  }, [embedOptions, targetId, targetType, title, description]);

  return (
    <div className="embed-code">
      <button 
        onClick={() => setShowEmbedMenu(!showEmbedMenu)}
        className="embed-button"
      >
        <span className="embed-icon">🔗</span>
        埋め込みコード
      </button>

      {showEmbedMenu && (
        <div className="embed-menu">
          <div className="embed-header">
            <h3>埋め込みコード</h3>
            <button onClick={() => setShowEmbedMenu(false)}>×</button>
          </div>
          
          <div className="embed-options">
            <div className="option-group">
              <label>幅</label>
              <input
                type="number"
                value={embedOptions.width}
                onChange={(e) => setEmbedOptions(prev => ({ ...prev, width: parseInt(e.target.value) }))}
                min="200"
                max="1200"
              />
            </div>
            
            <div className="option-group">
              <label>高さ</label>
              <input
                type="number"
                value={embedOptions.height}
                onChange={(e) => setEmbedOptions(prev => ({ ...prev, height: parseInt(e.target.value) }))}
                min="150"
                max="800"
              />
            </div>
            
            <div className="option-group">
              <label>
                <input
                  type="checkbox"
                  checked={embedOptions.showTitle}
                  onChange={(e) => setEmbedOptions(prev => ({ ...prev, showTitle: e.target.checked }))}
                />
                タイトルを表示
              </label>
            </div>
            
            <div className="option-group">
              <label>
                <input
                  type="checkbox"
                  checked={embedOptions.showDescription}
                  onChange={(e) => setEmbedOptions(prev => ({ ...prev, showDescription: e.target.checked }))}
                />
                説明を表示
              </label>
            </div>
          </div>
          
          <div className="embed-preview">
            <h4>プレビュー</h4>
            <div 
              className="preview-container"
              style={{ width: `${Math.min(embedOptions.width, 400)}px`, height: `${Math.min(embedOptions.height, 300)}px` }}
            >
              <iframe
                src={`/embed/${targetType}/${targetId}`}
                width={Math.min(embedOptions.width, 400)}
                height={Math.min(embedOptions.height, 300)}
                frameBorder="0"
                allowFullScreen
              />
            </div>
          </div>
          
          <div className="embed-code-section">
            <h4>埋め込みコード</h4>
            <textarea
              value={embedCode}

// 埋め込みコンテンツ表示コンポーネント
function EmbedContent({ targetId, targetType }) {
  const [content, setContent] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const fetchEmbedContent = async () => {
      try {
        const response = await fetch(`/api/embed/${targetType}/${targetId}`);
        const data = await response.json();
        setContent(data);
      } catch (error) {
        console.error('埋め込みコンテンツ取得エラー:', error);
      } finally {
        setLoading(false);
      }
    };

    fetchEmbedContent();
  }, [targetId, targetType]);

  if (loading) return <div>読み込み中...</div>;
  if (!content) return <div>コンテンツが見つかりません</div>;

  return (
    <div className="embed-content">
      <div className="embed-header">
        <h2>{content.title}</h2>
        <p>{content.description}</p>
      </div>
      
      <div className="embed-body">
        {targetType === 'video' && (
          <video controls width="100%" height="auto">
            <source src={content.videoUrl} type="video/mp4" />
            お使いのブラウザは動画をサポートしていません。
          </video>
        )}
        
        {targetType === 'article' && (
          <div className="article-content">
            <div dangerouslySetInnerHTML={{ __html: content.content }} />
          </div>
        )}
        
        {targetType === 'chart' && (
          <div className="chart-container">
            <canvas id="embed-chart" width="400" height="200"></canvas>
          </div>
        )}
      </div>
      
      <div className="embed-footer">
        <a href={content.originalUrl} target="_blank" rel="noopener noreferrer">
          元のページを見る
        </a>
      </div>
    </div>
  );
}
```

## 4. ファイル共有機能

### 概要

### 実装例

```javascript
// ファイル共有コンポーネント
function FileShare({ fileId, fileName, fileSize, fileType }) {
  const [shareOptions, setShareOptions] = useState({
    expiresIn: '7d',
    password: '',
    allowDownload: true,
    allowPreview: true
  });
  const [shareLink, setShareLink] = useState('');
  const [showShareMenu, setShowShareMenu] = useState(false);

  const generateShareLink = async () => {
    try {
      const response = await fetch('/api/files/share', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${user.token}`
        },
        body: JSON.stringify({
          file_id: fileId,
          expires_in: shareOptions.expiresIn,
          password: shareOptions.password,
          allow_download: shareOptions.allowDownload,
          allow_preview: shareOptions.allowPreview
        })
      });

      if (response.ok) {
        const data = await response.json();
        setShareLink(data.share_link);
      }
    } catch (error) {
      console.error('共有リンク生成エラー:', error);
    }
  };

  const copyShareLink = async () => {
    try {
      await navigator.clipboard.writeText(shareLink);
      alert('共有リンクをコピーしました');
    } catch (error) {
      console.error('共有リンクコピーエラー:', error);
    }
  };

  const sendEmailShare = () => {
    const subject = `ファイル共有: ${fileName}`;
    const body = `以下のファイルを共有します:\n\nファイル名: ${fileName}\nサイズ: ${fileSize}\n\nダウンロードリンク: ${shareLink}`;
    const mailtoUrl = `mailto:?subject=${encodeURIComponent(subject)}&body=${encodeURIComponent(body)}`;
    window.open(mailtoUrl);
  };

  return (
    <div className="file-share">
      <button

        className="file-share-button"
      >
        <span className="share-icon">📤</span>

            <h4>{fileName}</h4>
            <p>サイズ: {fileSize}</p>
            <p>タイプ: {fileType}</p>
```

```javascript
// 共有追跡のキャッシュ
class ShareTrackingCache {
  constructor() {
    this.cache = new Map();
    this.ttl = 10 * 60 * 1000; // 10分
  }

  get(key) {
    const item = this.cache.get(key);
    if (item && Date.now() - item.timestamp < this.ttl) {
      return item.data;
    }
    return null;
  }

  set(key, data) {
    this.cache.set(key, {
      data,
      timestamp: Date.now()
    });
  }

  clear(key) {
    if (key) {
      this.cache.delete(key);
    } else {
      this.cache.clear();
    }
  }
}

const shareTrackingCache = new ShareTrackingCache();
```

### 2. 共有のバッチ処理

```javascript
// 共有のバッチ処理
class ShareBatch {
  constructor() {
    this.batch = [];
    this.batchSize = 20;
    this.batchTimeout = 2000; // 2秒
  }

  add(shareData) {
    this.batch.push(shareData);
    
    if (this.batch.length >= this.batchSize) {
      this.flush();
    } else {
      setTimeout(() => this.flush(), this.batchTimeout);
    }
  }

  async flush() {
    if (this.batch.length === 0) return;

    const batch = [...this.batch];
    this.batch = [];

    try {
      await fetch('/api/shares/batch', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'
        },
        body: JSON.stringify({ shares: batch })
      });
    } catch (error) {
      console.error('共有バッチ処理エラー:', error);
    }
  }
}

const shareBatch = new ShareBatch();
```

```javascript
// 共有のリアルタイム更新
function useRealtimeShares(targetId, targetType) {
  const [shareStats, setShareStats] = useState({
    totalShares: 0,
    recentShares: []
  });
  const [ws, setWs] = useState(null);

  useEffect(() => {
    const websocket = new WebSocket(`ws://localhost:8080/shares/${targetId}`);
    
    websocket.onmessage = (event) => {
      const data = JSON.parse(event.data);
      
      if (data.type === 'share_update') {
        setShareStats(prev => ({
          ...prev,
          totalShares: data.total_shares,
          recentShares: [data.share, ...prev.recentShares.slice(0, 9)]
        }));
      }
    };

    setWs(websocket);

    return () => {
      websocket.close();
    };
  }, [targetId, targetType]);

  return shareStats;
}
```

## まとめ

| 共有タイプ | 適用場面 | 主なメリット | 主なデメリット |
|-----------|---------|-------------|---------------|
| **URL共有** | 基本的な共有 | 実装が簡単、軽量 | 拡散効果が低い |
| **ソーシャルメディア連携** | 拡散・宣伝 | 拡散効果が高い、分析可能 | 外部依存、制限 |
| **埋め込み機能** | 他サイトでの表示 | 拡散効果が高い、ブランド認知 | 実装が複雑、セキュリティ |
| **ファイル共有** | ファイルの配布 | 直接共有、制御可能 | 実装が複雑、ストレージ |

### 選択の指針

1. **基本的な共有** → URL共有
2. **拡散・宣伝** → ソーシャルメディア連携
3. **他サイトでの表示** → 埋め込み機能
4. **ファイルの配布** → ファイル共有
5. **複数の要件** → 組み合わせパターン
