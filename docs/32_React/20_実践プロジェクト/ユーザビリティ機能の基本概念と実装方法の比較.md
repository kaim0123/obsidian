# ユーザビリティ機能の基本概念と実装方法の比較

## 概要

ユーザビリティ機能は、Webアプリケーションの使いやすさを向上させるための重要な要素です。ユーザーが直感的に操作でき、効率的に目的を達成できるよう支援する機能を指します。この記事では、初心者向けにユーザビリティ機能の基本概念と、主要な実装方法の特徴、メリット・デメリットを詳しく解説します。

## ユーザビリティとは

ユーザビリティとは、ユーザーが製品やサービスを「使いやすい」と感じる度合いを表す概念です。Webアプリケーションにおいては、以下の要素が重要になります：

### ユーザビリティの5つの原則

1. **学習しやすさ**: 初回利用時でも直感的に操作できる
2. **効率性**: 慣れたユーザーが迅速に作業を完了できる
3. **記憶しやすさ**: しばらく使わなくても操作方法を覚えている
4. **エラー防止**: ユーザーのミスを防ぎ、エラーから回復しやすくする
5. **満足度**: ユーザーが使用を楽しいと感じる

## 1. ブレッドクラム（パンくずリスト）

### 概要
ユーザーが現在いる場所を階層構造で表示し、ナビゲーションを支援する機能

### 実装例

```javascript
import { useRouter } from 'next/router';
import Link from 'next/link';

// ブレッドクラムコンポーネント
function Breadcrumb() {
  const router = useRouter();
  const { pathname } = router;

  // パスを階層に分割
  const generateBreadcrumbs = (pathname) => {
    const pathSegments = pathname.split('/').filter(segment => segment !== '');
    const breadcrumbs = [{ name: 'ホーム', href: '/' }];

    let currentPath = '';
    pathSegments.forEach((segment, index) => {
      currentPath += `/${segment}`;
      
      // セグメント名を日本語に変換
      const segmentNames = {
        'products': '商品一覧',
        'category': 'カテゴリ',
        'electronics': '電子機器',
        'smartphones': 'スマートフォン',
        'detail': '商品詳細'
      };

      breadcrumbs.push({
        name: segmentNames[segment] || segment,
        href: currentPath,
        isLast: index === pathSegments.length - 1
      });
    });

    return breadcrumbs;
  };

  const breadcrumbs = generateBreadcrumbs(pathname);

  return (
    <nav aria-label="パンくずリスト" className="breadcrumb">
      <ol className="breadcrumb-list">
        {breadcrumbs.map((breadcrumb, index) => (
          <li key={index} className="breadcrumb-item">
            {breadcrumb.isLast ? (
              <span className="breadcrumb-current" aria-current="page">
                {breadcrumb.name}
              </span>
            ) : (
              <Link href={breadcrumb.href} className="breadcrumb-link">
                {breadcrumb.name}
              </Link>
            )}
            {!breadcrumb.isLast && (
              <span className="breadcrumb-separator" aria-hidden="true">
                &gt;
              </span>
            )}
          </li>
        ))}
      </ol>
    </nav>
  );
}
```

## 2. お気に入り・ブックマーク機能

### 概要
ユーザーが気に入ったコンテンツを保存し、後で簡単にアクセスできる機能

### 実装例

```javascript
import { useState, useEffect } from 'react';
import { useAuth } from '../hooks/useAuth';

// お気に入り管理フック
function useFavorites() {
  const { user } = useAuth();
  const [favorites, setFavorites] = useState([]);
  const [loading, setLoading] = useState(false);

  // お気に入り一覧の取得
  const fetchFavorites = async () => {
    if (!user) return;
    
    setLoading(true);
    try {
      const response = await fetch('/api/favorites', {
        headers: {
          'Authorization': `Bearer ${user.token}`
        }
      });
      const data = await response.json();
      setFavorites(data);
    } catch (error) {
      console.error('お気に入り取得エラー:', error);
    } finally {
      setLoading(false);
    }
  };

  // お気に入りの追加
  const addFavorite = async (item) => {
    if (!user) {
      alert('ログインが必要です');
      return;
    }

    try {
      const response = await fetch('/api/favorites', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${user.token}`
        },
        body: JSON.stringify({
          itemId: item.id,
          itemType: item.type,
          title: item.title,
          url: item.url,
          imageUrl: item.imageUrl
        })
      });

      if (response.ok) {
        const newFavorite = await response.json();
        setFavorites(prev => [...prev, newFavorite]);
        return true;
      }
    } catch (error) {
      console.error('お気に入り追加エラー:', error);
    }
    return false;
  };

  // お気に入りの削除
  const removeFavorite = async (itemId) => {
    try {
      const response = await fetch(`/api/favorites/${itemId}`, {
        method: 'DELETE',
        headers: {
          'Authorization': `Bearer ${user.token}`
        }
      });

      if (response.ok) {
        setFavorites(prev => prev.filter(fav => fav.id !== itemId));
        return true;
      }
    } catch (error) {
      console.error('お気に入り削除エラー:', error);
    }
    return false;
  };

  // お気に入り状態の確認
  const isFavorite = (itemId) => {
    return favorites.some(fav => fav.itemId === itemId);
  };

  useEffect(() => {
    fetchFavorites();
  }, [user]);

  return {
    favorites,
    loading,
    addFavorite,
    removeFavorite,
    isFavorite,
    fetchFavorites
  };
}

// お気に入りボタンコンポーネント
function FavoriteButton({ item }) {
  const { addFavorite, removeFavorite, isFavorite } = useFavorites();
  const [isToggling, setIsToggling] = useState(false);

  const handleToggle = async () => {
    setIsToggling(true);
    
    try {
      if (isFavorite(item.id)) {
        await removeFavorite(item.id);
      } else {
        await addFavorite(item);
      }
    } finally {
      setIsToggling(false);
    }
  };

  return (
    <button
      onClick={handleToggle}
      disabled={isToggling}
      className={`favorite-button ${isFavorite(item.id) ? 'favorited' : ''}`}
      aria-label={isFavorite(item.id) ? 'お気に入りから削除' : 'お気に入りに追加'}
    >
      {isToggling ? (
        <span className="loading">...</span>
      ) : (
        <span className="heart">♥</span>
      )}
    </button>
  );
}

// お気に入り一覧ページ
function FavoritesPage() {
  const { favorites, loading, removeFavorite } = useFavorites();

  if (loading) {
    return <div className="loading">お気に入りを読み込み中...</div>;
  }

  return (
    <div className="favorites-page">
      <h1>お気に入り</h1>
      
      {favorites.length === 0 ? (
        <div className="empty-state">
          <p>お気に入りに追加したアイテムがありません</p>
        </div>
      ) : (
        <div className="favorites-grid">
          {favorites.map(favorite => (
            <div key={favorite.id} className="favorite-item">
              <img src={favorite.imageUrl} alt={favorite.title} />
              <h3>{favorite.title}</h3>
              <a href={favorite.url} target="_blank" rel="noopener noreferrer">
                詳細を見る
              </a>
              <button
                onClick={() => removeFavorite(favorite.id)}
                className="remove-button"
              >
                削除
              </button>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}
```

## 3. 履歴・最近見たページ

### 概要
ユーザーの閲覧履歴を記録し、過去に訪問したページを簡単に再訪問できる機能

### 実装例

```javascript
import { useState, useEffect } from 'react';
import { useRouter } from 'next/router';

// 閲覧履歴管理フック
function useViewHistory() {
  const router = useRouter();
  const [history, setHistory] = useState([]);
  const [loading, setLoading] = useState(false);

  // 履歴の保存
  const addToHistory = async (page) => {
    const historyItem = {
      id: Date.now(),
      title: page.title,
      url: page.url,
      imageUrl: page.imageUrl,
      visitedAt: new Date().toISOString(),
      category: page.category
    };

    try {
      // ローカルストレージに保存（オフライン対応）
      const localHistory = JSON.parse(localStorage.getItem('viewHistory') || '[]');
      const updatedHistory = [historyItem, ...localHistory.filter(item => item.url !== page.url)].slice(0, 50);
      localStorage.setItem('viewHistory', JSON.stringify(updatedHistory));

      // サーバーにも保存（ログイン済みの場合）
      if (user) {
        await fetch('/api/view-history', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${user.token}`
          },
          body: JSON.stringify(historyItem)
        });
      }

      setHistory(updatedHistory);
    } catch (error) {
      console.error('履歴保存エラー:', error);
    }
  };

  // 履歴の取得
  const fetchHistory = async () => {
    setLoading(true);
    try {
      // ローカルストレージから取得
      const localHistory = JSON.parse(localStorage.getItem('viewHistory') || '[]');
      setHistory(localHistory);

      // サーバーからも取得（ログイン済みの場合）
      if (user) {
        const response = await fetch('/api/view-history', {
          headers: {
            'Authorization': `Bearer ${user.token}`
          }
        });
        if (response.ok) {
          const serverHistory = await response.json();
          setHistory(serverHistory);
        }
      }
    } catch (error) {
      console.error('履歴取得エラー:', error);
    } finally {
      setLoading(false);
    }
  };

  // 履歴のクリア
  const clearHistory = async () => {
    try {
      localStorage.removeItem('viewHistory');
      setHistory([]);

      if (user) {
        await fetch('/api/view-history', {
          method: 'DELETE',
          headers: {
            'Authorization': `Bearer ${user.token}`
          }
        });
      }
    } catch (error) {
      console.error('履歴クリアエラー:', error);
    }
  };

  // 特定の履歴を削除
  const removeFromHistory = (itemId) => {
    const updatedHistory = history.filter(item => item.id !== itemId);
    setHistory(updatedHistory);
    localStorage.setItem('viewHistory', JSON.stringify(updatedHistory));
  };

  useEffect(() => {
    fetchHistory();
  }, []);

  return {
    history,
    loading,
    addToHistory,
    clearHistory,
    removeFromHistory,
    fetchHistory
  };
}

// 履歴追跡コンポーネント
function HistoryTracker({ children, pageData }) {
  const { addToHistory } = useViewHistory();
  const router = useRouter();

  useEffect(() => {
    if (pageData && router.asPath) {
      addToHistory({
        title: pageData.title || document.title,
        url: router.asPath,
        imageUrl: pageData.imageUrl,
        category: pageData.category
      });
    }
  }, [router.asPath, pageData]);

  return children;
}

// 最近見たページコンポーネント
function RecentPages() {
  const { history, loading, clearHistory, removeFromHistory } = useViewHistory();

  if (loading) {
    return <div className="loading">履歴を読み込み中...</div>;
  }

  return (
    <div className="recent-pages">
      <div className="recent-pages-header">
        <h2>最近見たページ</h2>
        {history.length > 0 && (
          <button onClick={clearHistory} className="clear-button">
            履歴をクリア
          </button>
        )}
      </div>

      {history.length === 0 ? (
        <div className="empty-state">
          <p>最近見たページがありません</p>
        </div>
      ) : (
        <div className="history-list">
          {history.slice(0, 10).map(item => (
            <div key={item.id} className="history-item">
              <img src={item.imageUrl} alt={item.title} />
              <div className="history-content">
                <h3>{item.title}</h3>
                <p className="history-url">{item.url}</p>
                <p className="history-date">
                  {new Date(item.visitedAt).toLocaleString()}
                </p>
              </div>
              <div className="history-actions">
                <a href={item.url} className="visit-button">
                  訪問
                </a>
                <button
                  onClick={() => removeFromHistory(item.id)}
                  className="remove-button"
                >
                  削除
                </button>
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}
```

## 4. おすすめ・関連アイテム

### 概要
ユーザーの行動や好みに基づいて、関連性の高いコンテンツを推薦する機能

### 実装例

```javascript
import { useState, useEffect } from 'react';

// 推薦システムフック
function useRecommendations() {
  const [recommendations, setRecommendations] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  // 協調フィルタリングによる推薦
  const getCollaborativeRecommendations = async (userId) => {
    try {
      const response = await fetch(`/api/recommendations/collaborative/${userId}`);
      const data = await response.json();
      return data;
    } catch (error) {
      console.error('協調フィルタリング推薦エラー:', error);
      return [];
    }
  };

  // コンテンツベースフィルタリングによる推薦
  const getContentBasedRecommendations = async (itemId) => {
    try {
      const response = await fetch(`/api/recommendations/content-based/${itemId}`);
      const data = await response.json();
      return data;
    } catch (error) {
      console.error('コンテンツベース推薦エラー:', error);
      return [];
    }
  };

  // ハイブリッド推薦（複数の手法を組み合わせ）
  const getHybridRecommendations = async (userId, itemId) => {
    setLoading(true);
    setError(null);

    try {
      const [collaborative, contentBased] = await Promise.all([
        getCollaborativeRecommendations(userId),
        getContentBasedRecommendations(itemId)
      ]);

      // 推薦結果の統合と重み付け
      const combinedRecommendations = combineRecommendations(collaborative, contentBased);
      setRecommendations(combinedRecommendations);
    } catch (error) {
      setError('推薦の取得に失敗しました');
    } finally {
      setLoading(false);
    }
  };

  // 推薦結果の統合
  const combineRecommendations = (collaborative, contentBased) => {
    const combined = new Map();

    // 協調フィルタリング結果の追加
    collaborative.forEach(item => {
      combined.set(item.id, {
        ...item,
        score: item.score * 0.6 // 重み付け
      });
    });

    // コンテンツベース結果の追加
    contentBased.forEach(item => {
      const existing = combined.get(item.id);
      if (existing) {
        existing.score += item.score * 0.4;
      } else {
        combined.set(item.id, {
          ...item,
          score: item.score * 0.4
        });
      }
    });

    // スコア順でソート
    return Array.from(combined.values())
      .sort((a, b) => b.score - a.score)
      .slice(0, 10);
  };

  return {
    recommendations,
    loading,
    error,
    getHybridRecommendations
  };
}

        userId,
        recommendedItemId: item.id,
        sourceItemId: currentItemId,
        timestamp: new Date().toISOString()
      })
    });

    // ページ遷移
    window.location.href = `/products/${item.id}`;
  };

  if (loading) {
    return (
      <div className="recommendation-section">
        <h2>{title}</h2>
        <div className="loading">推薦を生成中...</div>
      </div>
    );

  if (error) {
    return (
      <div className="recommendation-section">
        <h2>{title}</h2>
        <div className="error">{error}</div>
      </div>
    );

  return (
    <div className="recommendation-section">
      <h2>{title}</h2>
      <div className="recommendation-grid">
        {recommendations.map(item => (
          <RecommendationItem
            key={item.id}
            item={item}
            onItemClick={handleItemClick}
          />
        ))}

  );
}
```

```javascript
  const [formData, setFormData] = useState({
    rating: 0,
    category: '',
    message: '',
    email: '',

      const response = await fetch('/api/feedback', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json'

          ...formData,
          timestamp: new Date().toISOString(),
          userAgent: navigator.userAgent,
          url: window.location.href
        })

      if (response.ok) {
        setSubmitted(true);

        throw new Error('フィードバックの送信に失敗しました');

    } catch (error) {
      console.error('フィードバック送信エラー:', error);

      setSubmitting(false);
```

```javascript
// 推薦結果のキャッシュ
const recommendationCache = new Map();

function useRecommendations() {
  const getRecommendations = async (userId, itemId) => {
    const cacheKey = `${userId}-${itemId}`;
    
    if (recommendationCache.has(cacheKey)) {
      return recommendationCache.get(cacheKey);
    }

    const recommendations = await fetchRecommendations(userId, itemId);
    recommendationCache.set(cacheKey, recommendations);
    
    // 5分後にキャッシュをクリア
    setTimeout(() => {
      recommendationCache.delete(cacheKey);
    }, 5 * 60 * 1000);

    return recommendations;
  };

  return { getRecommendations };
}
```
