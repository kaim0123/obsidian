# チャット・メッセージング機能の実装方法とベストプラクティス

## 概要

チャット・メッセージング機能は、ユーザー間のリアルタイム通信を実現する重要な機能です。基本的な1対1チャットから、グループチャット、リアルタイム通信まで、様々なチャット機能とその実装方法を詳しく解説します。

## チャット機能の種類と比較表

| 項目 | 1対1チャット | グループチャット | リアルタイムチャット | ファイル共有チャット |
|------|-------------|-----------------|-------------------|-------------------|
| **実装の複雑さ** | ⭐⭐ 中程度 | ⭐⭐⭐ 複雑 | ⭐⭐⭐ 複雑 | ⭐⭐⭐ 複雑 |
| **リアルタイム性** | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い | ⭐⭐ 中程度 |
| **スケーラビリティ** | ⭐⭐⭐ 高い | ⭐⭐ 中程度 | ⭐⭐ 中程度 | ⭐⭐ 中程度 |
| **ユーザビリティ** | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い |
| **セキュリティ** | ⭐⭐⭐ 高い | ⭐⭐ 中程度 | ⭐⭐ 中程度 | ⭐⭐ 中程度 |

| チャットタイプ | 最適な使用場面 | 例 |

## 1. 1対1チャット機能

### 概要
二人のユーザー間でのプライベートなメッセージ交換

### データベース設計

```sql
-- チャットルームテーブル
CREATE TABLE chat_rooms (
  id INT PRIMARY KEY AUTO_INCREMENT,
  user1_id INT NOT NULL,
  user2_id INT NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  
  FOREIGN KEY (user1_id) REFERENCES users(id),
  FOREIGN KEY (user2_id) REFERENCES users(id),
  UNIQUE KEY unique_user_pair (user1_id, user2_id),
  INDEX idx_user1_id (user1_id),
  INDEX idx_user2_id (user2_id)
);

-- メッセージテーブル
CREATE TABLE messages (
  id INT PRIMARY KEY AUTO_INCREMENT,
  room_id INT NOT NULL,
  sender_id INT NOT NULL,
  content TEXT NOT NULL,
  message_type ENUM('text', 'image', 'file') DEFAULT 'text',
  file_url VARCHAR(500),
  is_read BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (room_id) REFERENCES chat_rooms(id) ON DELETE CASCADE,
  FOREIGN KEY (sender_id) REFERENCES users(id),
  INDEX idx_room_id (room_id),
  INDEX idx_sender_id (sender_id),
  INDEX idx_created_at (created_at)
);
```

### 実装例

```javascript
// 1対1チャットコンポーネント
function PrivateChat({ currentUser, otherUser }) {
  const [messages, setMessages] = useState([]);
  const [newMessage, setNewMessage] = useState('');
  const [roomId, setRoomId] = useState(null);
  const [ws, setWs] = useState(null);
  const messagesEndRef = useRef(null);

  // チャットルームの取得または作成
  const getOrCreateRoom = useCallback(async () => {
    try {
      const response = await fetch('/api/chat/rooms', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${currentUser.token}`
        },
        body: JSON.stringify({
          user1_id: currentUser.id,
          user2_id: otherUser.id
        })
      });

      const data = await response.json();
      setRoomId(data.room_id);
      return data.room_id;
    } catch (error) {
      console.error('ルーム作成エラー:', error);
    }
  }, [currentUser, otherUser]);

  // メッセージの取得
  const fetchMessages = useCallback(async (roomId) => {
    try {
      const response = await fetch(`/api/chat/rooms/${roomId}/messages`);
      const data = await response.json();
      setMessages(data);
    } catch (error) {
      console.error('メッセージ取得エラー:', error);
    }
  }, []);

  // WebSocket接続
  const connectWebSocket = useCallback((roomId) => {
    const websocket = new WebSocket(`ws://localhost:8080/chat/${roomId}`);
    
    websocket.onmessage = (event) => {
      const data = JSON.parse(event.data);
      
      if (data.type === 'new_message') {
        setMessages(prev => [...prev, data.message]);
        scrollToBottom();
      } else if (data.type === 'message_read') {
        setMessages(prev => prev.map(msg => 
          msg.id === data.message_id ? { ...msg, is_read: true } : msg
        ));
      }
    };

    websocket.onopen = () => {
      console.log('WebSocket接続完了');
    };

    websocket.onclose = () => {
      console.log('WebSocket接続終了');
    };

    setWs(websocket);
  }, []);

  // メッセージ送信
  const sendMessage = async (e) => {
    e.preventDefault();
    if (!newMessage.trim() || !roomId) return;

    try {
      const response = await fetch(`/api/chat/rooms/${roomId}/messages`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${currentUser.token}`
        },
        body: JSON.stringify({
          content: newMessage.trim(),
          message_type: 'text'
        })
      });

      if (response.ok) {
        setNewMessage('');
      }
    } catch (error) {
      console.error('メッセージ送信エラー:', error);
    }
  };

  // メッセージ既読マーク
  const markAsRead = useCallback(async (messageId) => {
    try {
      await fetch(`/api/chat/messages/${messageId}/read`, {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${currentUser.token}`
        }
      });
    } catch (error) {
      console.error('既読マークエラー:', error);
    }
  }, [currentUser]);

  const scrollToBottom = () => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  };

  useEffect(() => {
    const initializeChat = async () => {
      const roomId = await getOrCreateRoom();
      if (roomId) {
        await fetchMessages(roomId);
        connectWebSocket(roomId);
      }
    };

    initializeChat();

    return () => {
      if (ws) {
        ws.close();
      }
    };
  }, [getOrCreateRoom, fetchMessages, connectWebSocket]);

  useEffect(() => {
    scrollToBottom();
  }, [messages]);

  return (
    <div className="private-chat">
      <div className="chat-header">
        <img 
          src={otherUser.avatar || '/default-avatar.png'} 
          alt={otherUser.name}
          className="user-avatar"
        />
        <div className="user-info">
          <h3>{otherUser.name}</h3>
          <span className="status">オンライン</span>
        </div>
      </div>

      <div className="messages-container">
        {messages.map(message => (
          <MessageItem
            key={message.id}
            message={message}
            isOwn={message.sender_id === currentUser.id}
            onRead={() => markAsRead(message.id)}
          />
        ))}
        <div ref={messagesEndRef} />
      </div>

      <form onSubmit={sendMessage} className="message-form">
        <input
          type="text"
          value={newMessage}
          onChange={(e) => setNewMessage(e.target.value)}
          placeholder="メッセージを入力..."
          maxLength={1000}
        />
        <button type="submit" disabled={!newMessage.trim()}>
          送信
        </button>
      </form>
    </div>
  );
}
```

## 2. グループチャット機能

### 概要
複数のユーザーが参加するグループでのメッセージ交換

### データベース設計

```sql
-- グループテーブル
CREATE TABLE chat_groups (

  INDEX idx_group_id (group_id),
  INDEX idx_user_id (user_id)

-- グループメッセージテーブル
CREATE TABLE group_messages (
  id INT PRIMARY KEY AUTO_INCREMENT,

-- メッセージ既読テーブル

  id INT PRIMARY KEY AUTO_INCREMENT,
  message_id INT NOT NULL,

  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
```

## 3. リアルタイムチャット機能

### 概要
WebSocketを使用したリアルタイム通信機能

### 実装例

```javascript
// リアルタイムチャット管理
class RealtimeChatManager {
  constructor() {
    this.ws = null;
    this.reconnectAttempts = 0;
    this.maxReconnectAttempts = 5;
    this.reconnectDelay = 1000;
    this.listeners = new Map();
  }

  connect(roomId, token) {
    try {
      this.ws = new WebSocket(`ws://localhost:8080/chat/${roomId}?token=${token}`);
      
      this.ws.onopen = () => {
        console.log('WebSocket接続完了');
        this.reconnectAttempts = 0;
        this.emit('connected');
      };

      this.ws.onmessage = (event) => {
        const data = JSON.parse(event.data);
        this.emit(data.type, data);
      };

      this.ws.onclose = () => {
        console.log('WebSocket接続終了');
        this.emit('disconnected');
        this.handleReconnect(roomId, token);
      };

      this.ws.onerror = (error) => {
        console.error('WebSocketエラー:', error);
        this.emit('error', error);
      };
    } catch (error) {
      console.error('WebSocket接続エラー:', error);
    }
  }

  handleReconnect(roomId, token) {
    if (this.reconnectAttempts < this.maxReconnectAttempts) {
      this.reconnectAttempts++;
      setTimeout(() => {
        console.log(`再接続試行 ${this.reconnectAttempts}/${this.maxReconnectAttempts}`);
        this.connect(roomId, token);
      }, this.reconnectDelay * this.reconnectAttempts);
    }
  }

  sendMessage(type, data) {
    if (this.ws && this.ws.readyState === WebSocket.OPEN) {
      this.ws.send(JSON.stringify({ type, ...data }));
    }
  }

  on(event, callback) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, []);
    }
    this.listeners.get(event).push(callback);
  }

  off(event, callback) {
    if (this.listeners.has(event)) {
      const callbacks = this.listeners.get(event);
      const index = callbacks.indexOf(callback);
      if (index > -1) {
        callbacks.splice(index, 1);
      }
    }
  }

  emit(event, data) {
    if (this.listeners.has(event)) {
      this.listeners.get(event).forEach(callback => callback(data));
    }
  }

  disconnect() {
    if (this.ws) {
      this.ws.close();
      this.ws = null;
    }
  }
}

// リアルタイムチャットコンポーネント
function RealtimeChat({ roomId, currentUser }) {
  const [messages, setMessages] = useState([]);
  const [users, setUsers] = useState([]);
  const [connectionStatus, setConnectionStatus] = useState('connecting');
  const [typingUsers, setTypingUsers] = useState([]);
  const chatManager = useRef(new RealtimeChatManager());

  useEffect(() => {
    const manager = chatManager.current;

    // イベントリスナーの設定
    manager.on('connected', () => {
      setConnectionStatus('connected');
    });

    manager.on('disconnected', () => {
      setConnectionStatus('disconnected');
    });

    manager.on('new_message', (data) => {
      setMessages(prev => [...prev, data.message]);
    });

    manager.on('user_joined', (data) => {
      setUsers(prev => [...prev, data.user]);
    });

    manager.on('user_left', (data) => {
      setUsers(prev => prev.filter(user => user.id !== data.user_id));
    });

    manager.on('typing_start', (data) => {
      setTypingUsers(prev => [...prev.filter(id => id !== data.user_id), data.user_id]);
    });

    manager.on('typing_stop', (data) => {
      setTypingUsers(prev => prev.filter(id => id !== data.user_id));
    });

    // WebSocket接続
    manager.connect(roomId, currentUser.token);

    return () => {
      manager.disconnect();
    };
  }, [roomId, currentUser]);

  const sendMessage = (content) => {
    chatManager.current.sendMessage('send_message', {
      content,
      message_type: 'text'
    });
  };

  const sendTyping = () => {
    chatManager.current.sendMessage('typing_start');
  };

  const stopTyping = () => {
    chatManager.current.sendMessage('typing_stop');
  };

  return (
    <div className="realtime-chat">
      <div className="connection-status">
        接続状態: {connectionStatus}
      </div>

      <div className="online-users">
        オンラインユーザー: {users.length}名
      </div>

      <div className="typing-indicator">
        {typingUsers.length > 0 && (
          <p>{typingUsers.length}名が入力中...</p>
        )}
      </div>

      <div className="messages-container">
        {messages.map(message => (
          <MessageItem key={message.id} message={message} />
        ))}
      </div>

      <MessageInput
        onSend={sendMessage}
        onTyping={sendTyping}
        onStopTyping={stopTyping}
      />
    </div>
  );
}

// メッセージ入力コンポーネント
function MessageInput({ onSend, onTyping, onStopTyping }) {
  const [message, setMessage] = useState('');
  const [isTyping, setIsTyping] = useState(false);
  const typingTimeout = useRef(null);

  const handleInput = (e) => {
    setMessage(e.target.value);

    if (!isTyping) {
      setIsTyping(true);
      onTyping();
    }

    // タイピング停止のタイマーをリセット
    clearTimeout(typingTimeout.current);
    typingTimeout.current = setTimeout(() => {
      setIsTyping(false);
      onStopTyping();
    }, 1000);
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    if (message.trim()) {
      onSend(message.trim());
      setMessage('');
      setIsTyping(false);
      onStopTyping();
    }
  };

  return (
    <form onSubmit={handleSubmit} className="message-input">
      <input
        type="text"
        value={message}
        onChange={handleInput}
        placeholder="メッセージを入力..."
        maxLength={1000}
      />
      <button type="submit" disabled={!message.trim()}>
        送信
      </button>
    </form>
  );
}
```

```javascript
// ファイル共有チャット
function FileShareChat({ roomId, currentUser }) {
  const [messages, setMessages] = useState([]);
  const [uploading, setUploading] = useState(false);
  const fileInputRef = useRef(null);

  // ファイルアップロード
  const handleFileUpload = async (file) => {
    if (!file) return;

    setUploading(true);
    try {
      // ファイルをFormDataに追加
      const formData = new FormData();
      formData.append('file', file);
      formData.append('room_id', roomId);

      const response = await fetch('/api/chat/upload', {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${currentUser.token}`
        },
        body: formData
      });

      if (response.ok) {
        const data = await response.json();
        
        // ファイルメッセージを送信
        await fetch(`/api/chat/rooms/${roomId}/messages`, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${currentUser.token}`
          },
          body: JSON.stringify({
            content: file.name,
            message_type: getFileType(file.type),
            file_url: data.file_url
          })
        });
      }
    } catch (error) {
      console.error('ファイルアップロードエラー:', error);
    } finally {
      setUploading(false);
    }
  };

  const getFileType = (mimeType) => {
    if (mimeType.startsWith('image/')) return 'image';
    if (mimeType.startsWith('video/')) return 'video';
    return 'file';
  };

  const handleFileSelect = (e) => {
    const file = e.target.files[0];
    if (file) {
      handleFileUpload(file);
    }
  };

  return (
    <div className="file-share-chat">
      <div className="messages-container">
        {messages.map(message => (
          <FileMessageItem key={message.id} message={message} />
        ))}
      </div>

      <div className="message-form">
        <input
          type="file"
          ref={fileInputRef}
          onChange={handleFileSelect}
          accept="image/*,video/*,.pdf,.doc,.docx"
          style={{ display: 'none' }}
        />
        
        <button 
          onClick={() => fileInputRef.current?.click()}
          disabled={uploading}
          className="file-upload-button"
        >
          {uploading ? 'アップロード中...' : '📎 ファイル'}
        </button>
      </div>
    </div>
  );
}

// ファイルメッセージアイテム
function FileMessageItem({ message }) {
  const renderFileContent = () => {
    switch (message.message_type) {
      case 'image':
        return (
          <div className="image-message">
            <img 
              src={message.file_url} 
              alt={message.content}
              className="shared-image"
              onClick={() => window.open(message.file_url, '_blank')}
            />
            <p className="file-name">{message.content}</p>
          </div>
        );
      
      case 'video':
        return (
          <div className="video-message">
            <video 
              src={message.file_url} 
              controls
              className="shared-video"
            />
            <p className="file-name">{message.content}</p>
          </div>
        );
      
      default:
        return (
          <div className="file-message">
            <a 
              href={message.file_url} 
              download={message.content}
              className="file-download"
            >
              📄 {message.content}
            </a>
          </div>
        );
    }
  };

  return (
    <div className="file-message-item">
      {renderFileContent()}
      <div className="message-meta">
        <span className="message-time">
          {new Date(message.created_at).toLocaleTimeString('ja-JP')}

      </div>
    </div>
  );
```

```javascript
// メッセージの遅延読み込み
function PaginatedMessages({ roomId }) {
  const [messages, setMessages] = useState([]);
  const [hasMore, setHasMore] = useState(true);
  const [loading, setLoading] = useState(false);
  const messagesContainerRef = useRef(null);

  const loadMoreMessages = async () => {
    if (loading || !hasMore) return;

    setLoading(true);
    try {
      const oldestMessage = messages[0];
      const response = await fetch(
        `/api/chat/rooms/${roomId}/messages?before=${oldestMessage?.id}&limit=20`
      );
      const data = await response.json();
      
      setMessages(prev => [...data.messages, ...prev]);
      setHasMore(data.hasMore);
    } catch (error) {
      console.error('メッセージ読み込みエラー:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleScroll = (e) => {
    if (e.target.scrollTop === 0 && hasMore) {
      loadMoreMessages();
    }
  };

  return (
    <div 
      ref={messagesContainerRef}
      className="messages-container"
      onScroll={handleScroll}
    >
      {loading && <div>読み込み中...</div>}
      {messages.map(message => (
        <MessageItem key={message.id} message={message} />
      ))}
    </div>
  );
}
```

```javascript
// メッセージのキャッシュ管理
class MessageCache {
  constructor() {
    this.cache = new Map();
    this.maxSize = 1000;
  }

  get(roomId) {
    return this.cache.get(roomId) || [];
  }

  set(roomId, messages) {
    if (this.cache.size >= this.maxSize) {
      const firstKey = this.cache.keys().next().value;
      this.cache.delete(firstKey);
    }
    this.cache.set(roomId, messages);
  }

  addMessage(roomId, message) {
    const messages = this.get(roomId);
    this.set(roomId, [...messages, message]);
  }

  clear(roomId) {
    this.cache.delete(roomId);
  }
}
```
