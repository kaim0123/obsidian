# タブ・アコーディオンの実装方法とベストプラクティス

## 概要

タブ・アコーディオンは、コンテンツを整理して表示するための重要なUI要素です。限られたスペースで多くの情報を効率的に表示し、ユーザーのナビゲーションを向上させることができます。

## タブ・アコーディオンの種類と比較表

| 項目 | 基本タブ | アコーディオン | ドロップダウンメニュー | ステップインジケーター |
|------|---------|---------------|---------------------|---------------------|
| **実装の複雑さ** | ⭐⭐ 中程度 | ⭐⭐ 中程度 | ⭐⭐⭐ 複雑 | ⭐⭐⭐ 複雑 |
| **ユーザビリティ** | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い | ⭐⭐ 中程度 | ⭐⭐ 中程度 |
| **スペース効率** | ⭐⭐ 中程度 | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い | ⭐ 低い |
| **アクセシビリティ** | ⭐⭐ 中程度 | ⭐⭐ 中程度 | ⭐ 低い | ⭐⭐ 中程度 |
| **モバイル対応** | ⭐⭐ 中程度 | ⭐⭐⭐ 優秀 | ⭐⭐⭐ 優秀 | ⭐ 低い |

## 1. 基本タブ

### 概要
複数のコンテンツをタブで切り替える基本的な実装

### 実装例

```javascript
// 基本タブコンポーネント
function Tabs({ tabs, defaultTab = 0, onTabChange }) {
  const [activeTab, setActiveTab] = useState(defaultTab);

  const handleTabClick = (index) => {
    setActiveTab(index);
    onTabChange?.(index);
  };

  return (
    <div className="tabs">
      <div className="tabs-header" role="tablist">
        {tabs.map((tab, index) => (
          <button
            key={index}
            className={`tab-button ${activeTab === index ? 'active' : ''}`}
            onClick={() => handleTabClick(index)}
            role="tab"
            aria-selected={activeTab === index}
            aria-controls={`tab-panel-${index}`}
            id={`tab-${index}`}
          >
            {tab.icon && <span className="tab-icon">{tab.icon}</span>}
            <span className="tab-label">{tab.label}</span>
            {tab.badge && <span className="tab-badge">{tab.badge}</span>}
          </button>
        ))}
      </div>
      
      <div className="tabs-content">
        {tabs.map((tab, index) => (
          <div
            key={index}
            className={`tab-panel ${activeTab === index ? 'active' : ''}`}
            role="tabpanel"
            id={`tab-panel-${index}`}
            aria-labelledby={`tab-${index}`}
            hidden={activeTab !== index}
          >
            {tab.content}
          </div>
        ))}
      </div>
    </div>
  );
}

// 使用例
function App() {
  const tabs = [
    {
      label: 'ホーム',
      icon: '🏠',
      content: <HomeContent />
    },
    {
      label: 'プロフィール',

    },
    {
      label: '設定',
      icon: '⚙️',
      content: <SettingsContent />,
      badge: '3'
    }
  ];

  return (
    <div>
      <Tabs 
        tabs={tabs}

}
```

## 2. アコーディオン

### 概要
コンテンツを折りたたみ可能なアコーディオン

### 実装例

```javascript
// アコーディオンコンポーネント
function Accordion({ items, allowMultiple = false, defaultOpen = [] }) {
  const [openItems, setOpenItems] = useState(new Set(defaultOpen));

  const handleToggle = (index) => {
    setOpenItems(prev => {
      const newSet = new Set(prev);
      if (newSet.has(index)) {
        newSet.delete(index);
      } else {
        if (!allowMultiple) {
          newSet.clear();
        }
        newSet.add(index);
      }
      return newSet;
    });
  };

  return (
    <div className="accordion">
      {items.map((item, index) => (
        <div key={index} className="accordion-item">
          <button
            className="accordion-header"
            onClick={() => handleToggle(index)}
            aria-expanded={openItems.has(index)}
            aria-controls={`accordion-content-${index}`}
            id={`accordion-header-${index}`}
          >
            <span className="accordion-title">{item.title}</span>
            <span className={`accordion-icon ${openItems.has(index) ? 'open' : ''}`}>
              ▼
            </span>
          </button>
          
          <div
            className={`accordion-content ${openItems.has(index) ? 'open' : ''}`}
            id={`accordion-content-${index}`}
            aria-labelledby={`accordion-header-${index}`}
            hidden={!openItems.has(index)}
          >
            <div className="accordion-body">
              {item.content}
            </div>
          </div>
        </div>
      ))}
    </div>
  );
}

// 使用例
```

## 3. ドロップダウンメニュー

### 概要
クリックで展開するドロップダウンメニュー

### 実装例

```javascript
// ドロップダウンメニューコンポーネント
function DropdownMenu({ 
  trigger, 
  items, 
  position = 'bottom-left',
  onItemClick 
}) {
  const [isOpen, setIsOpen] = useState(false);
  const [selectedItem, setSelectedItem] = useState(null);
  const dropdownRef = useRef(null);

  useEffect(() => {
    const handleClickOutside = (event) => {
      if (dropdownRef.current && !dropdownRef.current.contains(event.target)) {
        setIsOpen(false);
      }
    };

    document.addEventListener('mousedown', handleClickOutside);
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, []);

  const handleItemClick = (item) => {
    setSelectedItem(item);
    setIsOpen(false);
    onItemClick?.(item);
  };

  const getPositionClass = () => {
    const positions = {
      'bottom-left': 'dropdown-bottom-left',
      'bottom-right': 'dropdown-bottom-right',
      'top-left': 'dropdown-top-left',
      'top-right': 'dropdown-top-right'
    };
    return positions[position] || positions['bottom-left'];
  };

  return (
    <div className="dropdown" ref={dropdownRef}>
      <div 
        className="dropdown-trigger"
        onClick={() => setIsOpen(!isOpen)}
        role="button"
        tabIndex={0}
        aria-haspopup="true"
        aria-expanded={isOpen}
      >
        {trigger}
      </div>
      
      {isOpen && (
        <div className={`dropdown-menu ${getPositionClass()}`} role="menu">
          {items.map((item, index) => (
            <div
              key={index}
              className={`dropdown-item ${item.disabled ? 'disabled' : ''} ${selectedItem?.value === item.value ? 'selected' : ''}`}
              onClick={() => !item.disabled && handleItemClick(item)}
              role="menuitem"
              tabIndex={item.disabled ? -1 : 0}
            >
              {item.icon && <span className="dropdown-item-icon">{item.icon}</span>}
              <span className="dropdown-item-text">{item.label}</span>
              {item.badge && <span className="dropdown-item-badge">{item.badge}</span>}
            </div>
          ))}
        </div>
      )}
    </div>
  );
}

// 使用例
```

## 4. ステップインジケーター

### 概要
プロセスの進行状況を表示するステップインジケーター

### 実装例

```javascript
// ステップインジケーターコンポーネント
function StepIndicator({ 
  steps, 
  currentStep, 
  onStepClick,
  orientation = 'horizontal' 
}) {
  const handleStepClick = (stepIndex) => {
    if (onStepClick && stepIndex <= currentStep) {
      onStepClick(stepIndex);
    }
  };

  return (
    <div className={`step-indicator step-indicator-${orientation}`}>
      {steps.map((step, index) => {
        const isCompleted = index < currentStep;
        const isCurrent = index === currentStep;
        const isClickable = onStepClick && index <= currentStep;

        return (
          <div key={index} className="step-item">
            <div className="step-connector" />
            
            <button
              className={`step-button ${isCompleted ? 'completed' : ''} ${isCurrent ? 'current' : ''} ${isClickable ? 'clickable' : ''}`}
              onClick={() => handleStepClick(index)}
              disabled={!isClickable}
              aria-label={`ステップ ${index + 1}: ${step.title}`}
            >
              <span className="step-number">
                {isCompleted ? '✓' : index + 1}
              </span>
            </button>
            
            <div className="step-content">
              <h3 className="step-title">{step.title}</h3>
              {step.description && (
                <p className="step-description">{step.description}</p>
              )}
            </div>
          </div>
        );
      })}
    </div>
  );
}

// 使用例
function CheckoutProcess() {
  const [currentStep, setCurrentStep] = useState(0);
  
  const steps = [

    },
    {
      title: '完了',

    }

  const handleStepClick = (stepIndex) => {
    setCurrentStep(stepIndex);

        )}
      </div>
    </div>
  );
}
```
