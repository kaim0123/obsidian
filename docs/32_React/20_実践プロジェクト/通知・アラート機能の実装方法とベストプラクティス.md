# 通知・アラート機能の実装方法とベストプラクティス

## 概要

通知・アラート機能は、ユーザーに重要な情報を伝えるための重要なUI要素です。成功、警告、エラー、情報など、様々な種類の通知を適切に実装することで、ユーザーエクスペリエンスを向上させることができます。

## 通知・アラートの種類と比較表

| 項目 | トースト通知 | インライン通知 | モーダル通知 | システム通知 |
|------|-------------|---------------|-------------|-------------|
| **実装の複雑さ** | ⭐⭐ 中程度 | ⭐ 簡単 | ⭐⭐ 中程度 | ⭐⭐⭐ 複雑 |
| **ユーザビリティ** | ⭐⭐⭐ 高い | ⭐⭐ 中程度 | ⭐⭐ 中程度 | ⭐⭐⭐ 高い |
| **非侵襲性** | ⭐⭐⭐ 高い | ⭐⭐ 中程度 | ⭐ 低い | ⭐⭐⭐ 高い |
| **アクセシビリティ** | ⭐⭐ 中程度 | ⭐⭐⭐ 高い | ⭐⭐ 中程度 | ⭐ 低い |
| **モバイル対応** | ⭐⭐⭐ 優秀 | ⭐⭐ 中程度 | ⭐ 低い | ⭐⭐ 中程度 |

## 1. トースト通知

### 概要
画面の端に表示される一時的な通知

### 実装例

```javascript
// トースト通知管理クラス
class ToastManager {
  constructor() {
    this.toasts = [];
    this.container = this.createContainer();
  }

  createContainer() {
    const container = document.createElement('div');
    container.className = 'toast-container';
    container.setAttribute('aria-live', 'polite');
    container.setAttribute('aria-atomic', 'true');
    document.body.appendChild(container);
    return container;
  }

  show(message, type = 'info', duration = 5000) {
    const toast = {
      id: Date.now().toString(),
      message,
      type,
      duration,
      timestamp: Date.now()
    };

    this.toasts.push(toast);
    this.renderToast(toast);
    this.scheduleRemoval(toast.id, duration);
  }

  renderToast(toast) {
    const toastElement = document.createElement('div');
    toastElement.className = `toast toast-${toast.type}`;
    toastElement.setAttribute('role', 'alert');
    toastElement.innerHTML = `
      <div class="toast-content">
        <span class="toast-icon">${this.getIcon(toast.type)}</span>
        <span class="toast-message">${toast.message}</span>
        <button class="toast-close" aria-label="閉じる">×</button>
      </div>
    `;

    // 閉じるボタンのイベント
    toastElement.querySelector('.toast-close').addEventListener('click', () => {
      this.remove(toast.id);
    });

    this.container.appendChild(toastElement);

    // アニメーション
    setTimeout(() => {
      toastElement.classList.add('toast-show');
    }, 10);
  }

  getIcon(type) {
    const icons = {
      success: '✓',
      error: '✕',
      warning: '⚠',
      info: 'ℹ'
    };
    return icons[type] || icons.info;
  }

  scheduleRemoval(id, duration) {
    setTimeout(() => {
      this.remove(id);
    }, duration);
  }

  remove(id) {
    const toastElement = this.container.querySelector(`[data-id="${id}"]`);
    if (toastElement) {
      toastElement.classList.add('toast-hide');
      setTimeout(() => {
        toastElement.remove();
      }, 300);
    }

    this.toasts = this.toasts.filter(toast => toast.id !== id);
  }
}

// Reactコンポーネント版
import { useState, useEffect, createContext, useContext } from 'react';

const ToastContext = createContext();

export const ToastProvider = ({ children }) => {
  const [toasts, setToasts] = useState([]);

  const showToast = (message, type = 'info', duration = 5000) => {
    const id = Date.now().toString();
    const newToast = { id, message, type, duration };
    
    setToasts(prev => [...prev, newToast]);
    
    setTimeout(() => {
      removeToast(id);
    }, duration);
  };

  const removeToast = (id) => {
    setToasts(prev => prev.filter(toast => toast.id !== id));
  };

  return (
    <ToastContext.Provider value={{ showToast, removeToast }}>
      {children}
      <ToastContainer toasts={toasts} onRemove={removeToast} />
    </ToastContext.Provider>
  );
};

export const useToast = () => {
  const context = useContext(ToastContext);
  if (!context) {
    throw new Error('useToast must be used within a ToastProvider');
  }
  return context;
};

function ToastContainer({ toasts, onRemove }) {
  return (
    <div className="toast-container" aria-live="polite" aria-atomic="true">
      {toasts.map(toast => (
        <Toast key={toast.id} toast={toast} onRemove={onRemove} />
      ))}
    </div>
  );
}

function Toast({ toast, onRemove }) {
  const [isVisible, setIsVisible] = useState(false);

  useEffect(() => {
    setIsVisible(true);
  }, []);

  const handleClose = () => {
    setIsVisible(false);
    setTimeout(() => onRemove(toast.id), 300);
  };

  return (
    <div 
      className={`toast toast-${toast.type} ${isVisible ? 'toast-show' : ''}`}
      role="alert"
    >
      <div className="toast-content">
        <span className="toast-icon">{getIcon(toast.type)}</span>
        <span className="toast-message">{toast.message}</span>
        <button 
          className="toast-close" 
          onClick={handleClose}
          aria-label="閉じる"
        >
          ×
        </button>
      </div>
    </div>
  );
}

function getIcon(type) {
  const icons = {
    success: '✓',
    error: '✕',
    warning: '⚠',
    info: 'ℹ'
  };
  return icons[type] || icons.info;
}
```

## 2. インライン通知

### 概要
コンテンツ内に表示される通知

### 実装例

```javascript
// インライン通知コンポーネント
function InlineAlert({ type, title, message, onClose, dismissible = true }) {
  const [isVisible, setIsVisible] = useState(true);

  const handleClose = () => {
    setIsVisible(false);
    onClose?.();
  };

  if (!isVisible) return null;

  return (
    <div className={`inline-alert alert-${type}`} role="alert">
      <div className="alert-content">
        <div className="alert-icon">
          {getIcon(type)}
        </div>
        <div className="alert-body">
          {title && <h4 className="alert-title">{title}</h4>}
          <p className="alert-message">{message}</p>
        </div>
        {dismissible && (
          <button 
            className="alert-close"
            onClick={handleClose}
            aria-label="閉じる"
          >
            ×
          </button>
        )}
      </div>
    </div>
  );
}

// 使用例
function FormWithValidation() {
  const [errors, setErrors] = useState([]);
  const [success, setSuccess] = useState('');

  const handleSubmit = async (formData) => {
    try {
      await submitForm(formData);
      setSuccess('フォームが正常に送信されました');
      setErrors([]);
    } catch (error) {
      setErrors([error.message]);
      setSuccess('');
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      {errors.map((error, index) => (
        <InlineAlert
          key={index}
          type="error"
          message={error}
          onClose={() => setErrors(prev => prev.filter((_, i) => i !== index))}
        />
      ))}
      
      {success && (
        <InlineAlert
          type="success"
          message={success}
          onClose={() => setSuccess('')}
        />
      )}
      
      {/* フォームフィールド */}
    </form>
  );
}
```

## 3. モーダル通知

### 概要
画面を覆うモーダル形式の通知

### 実装例

```javascript
// モーダル通知コンポーネント
function ModalAlert({ isOpen, type, title, message, onClose, onConfirm }) {
  useEffect(() => {
    if (isOpen) {
      document.body.style.overflow = 'hidden';
    } else {
      document.body.style.overflow = 'unset';
    }

    return () => {
      document.body.style.overflow = 'unset';
    };
  }, [isOpen]);

  useEffect(() => {
    const handleEscape = (e) => {
      if (e.key === 'Escape') {
        onClose();
      }
    };

    if (isOpen) {
      document.addEventListener('keydown', handleEscape);
    }

    return () => {
      document.removeEventListener('keydown', handleEscape);
    };
  }, [isOpen, onClose]);

  if (!isOpen) return null;

  return (
    <div className="modal-overlay" onClick={onClose}>
      <div 
        className="modal-alert"
        onClick={(e) => e.stopPropagation()}
        role="dialog"
        aria-modal="true"
        aria-labelledby="modal-title"
      >
        <div className="modal-header">
          <div className="modal-icon">
            {getIcon(type)}
          </div>
          <h2 id="modal-title" className="modal-title">{title}</h2>
        </div>
        
        <div className="modal-body">
          <p className="modal-message">{message}</p>
        </div>
        
        <div className="modal-footer">
          {type === 'confirm' && (
            <button 
              className="modal-button modal-button-secondary"
              onClick={onClose}
            >
              キャンセル
            </button>
          )}
          <button 
            className={`modal-button modal-button-${type === 'confirm' ? 'primary' : type}`}
            onClick={onConfirm || onClose}
          >
            {getButtonText(type)}
          </button>
        </div>
      </div>
    </div>
  );
}

function getIcon(type) {
  const icons = {
    success: '✓',
    error: '✕',
    warning: '⚠',
    info: 'ℹ',
    confirm: '?'
  };
  return icons[type] || icons.info;
}

function getButtonText(type) {
  const texts = {
    success: 'OK',
    error: 'OK',
    warning: 'OK',
    info: 'OK',
    confirm: '確認'
  };
  return texts[type] || 'OK';
}

// 使用例
function App() {
  const [modal, setModal] = useState({ isOpen: false });

  const showModal = (type, title, message, onConfirm) => {
    setModal({ isOpen: true, type, title, message, onConfirm });
  };

  const closeModal = () => {
    setModal({ isOpen: false });
  };

  const handleDelete = () => {
    showModal('confirm', '削除の確認', 'このアイテムを削除しますか？', () => {
      // 削除処理
      console.log('削除しました');
      closeModal();
    });
  };

  return (
    <div>
      <button onClick={handleDelete}>削除</button>
      
      <ModalAlert
        isOpen={modal.isOpen}
        type={modal.type}
        title={modal.title}
        message={modal.message}
        onClose={closeModal}
        onConfirm={modal.onConfirm}
      />
    </div>
  );
}
```

## 4. システム通知

### 概要
ブラウザの通知APIを使用したシステム通知

### 実装例

```javascript
// システム通知管理クラス
class SystemNotificationManager {
  constructor() {
    this.permission = Notification.permission;
  }

  async requestPermission() {
    if (!('Notification' in window)) {
      throw new Error('このブラウザは通知をサポートしていません');
    }

    if (this.permission === 'default') {
      this.permission = await Notification.requestPermission();
    }

    return this.permission === 'granted';
  }

  show(title, options = {}) {
    if (this.permission !== 'granted') {
      console.warn('通知の許可が得られていません');
      return null;
    }

    const defaultOptions = {
      body: '',
      icon: '/favicon.ico',
      badge: '/badge.png',
      tag: 'default',
      requireInteraction: false,
      silent: false,
      timestamp: Date.now()
    };

    const notification = new Notification(title, { ...defaultOptions, ...options });

    // 自動で閉じる
    if (!options.requireInteraction) {
      setTimeout(() => {
        notification.close();
      }, 5000);
    }

    return notification;
  }

  showSuccess(title, message) {
    return this.show(title, {
      body: message,
      icon: '/icons/success.png',
      tag: 'success'
    });
  }

  showError(title, message) {
    return this.show(title, {
      body: message,
      icon: '/icons/error.png',
      tag: 'error',
      requireInteraction: true
    });
  }

  showWarning(title, message) {
    return this.show(title, {
      body: message,
      icon: '/icons/warning.png',
      tag: 'warning'
    });
  }

  showInfo(title, message) {
    return this.show(title, {
      body: message,
      icon: '/icons/info.png',
      tag: 'info'
    });
  }
}

// React Hook版
export const useSystemNotification = () => {
  const [permission, setPermission] = useState(Notification.permission);

  const requestPermission = async () => {
    if (!('Notification' in window)) {
      throw new Error('このブラウザは通知をサポートしていません');
    }

    const newPermission = await Notification.requestPermission();
    setPermission(newPermission);
    return newPermission === 'granted';
  };

  const showNotification = (title, options = {}) => {
    if (permission !== 'granted') {
      console.warn('通知の許可が得られていません');
      return null;
    }

    const notification = new Notification(title, options);
    
    if (!options.requireInteraction) {
      setTimeout(() => notification.close(), 5000);
    }

    return notification;
  };

  return {
    permission,
    requestPermission,
    showNotification
  };
};

// 使用例
function NotificationDemo() {
  const { permission, requestPermission, showNotification } = useSystemNotification();

  const handleRequestPermission = async () => {
    try {
      await requestPermission();
    } catch (error) {
      alert('通知の許可を取得できませんでした');
    }
  };

  const handleShowNotification = () => {
    showNotification('テスト通知', {
      body: 'これはテスト通知です',
      icon: '/favicon.ico'
    });
  };

  return (
    <div>
      {permission === 'default' && (
        <button onClick={handleRequestPermission}>
          通知の許可をリクエスト
        </button>
      )}
      
      {permission === 'granted' && (
        <button onClick={handleShowNotification}>
          通知を表示
        </button>
      )}
      
      {permission === 'denied' && (
        <p>通知が拒否されています。ブラウザの設定で許可してください。</p>
      )}
    </div>
  );
}
```

## まとめ

| 通知タイプ | 適用場面 | 主なメリット | 主なデメリット |
|-----------|---------|-------------|---------------|
| **トースト通知** | 一般的なアプリ | 非侵襲性、ユーザビリティ | 実装の複雑さ |
| **インライン通知** | フォーム、コンテンツ | アクセシビリティ、シンプル | スペース使用 |
| **モーダル通知** | 重要な確認 | 注意を引く、確実 | 侵襲的 |
| **システム通知** | バックグラウンド処理 | 非侵襲性、永続性 | ブラウザ依存 |

### 選択の指針

1. **一般的な通知** → トースト通知
2. **フォームエラー** → インライン通知
3. **重要な確認** → モーダル通知
4. **バックグラウンド処理** → システム通知
