## おすすめ・関連アイテムとは

おすすめ・関連アイテム機能は、ユーザーの行動履歴や嗜好に基づいて、関連性の高いコンテンツを自動的に提案する機能です。ECサイト、動画サイト、ニュースサイトなどで広く活用されており、ユーザーエクスペリエンスの向上とコンバージョン率の改善に大きく貢献します。

### 主な特徴
- **パーソナライゼーション**: ユーザー個人に最適化された提案
- **関連性**: 現在のコンテンツと関連するアイテムの表示
- **多様性**: 様々なアルゴリズムによる提案の多様化
- **学習機能**: ユーザーの反応から継続的に改善

## フロントエンド実装例

### React での実装
```jsx
import React, { useState, useEffect, useCallback } from 'react';

const RecommendationsSection = ({ currentItem, userId }) => {
  const [recommendations, setRecommendations] = useState([]);
  const [relatedItems, setRelatedItems] = useState([]);
  const [activeFilter, setActiveFilter] = useState('similar');
  const [loading, setLoading] = useState(true);
  const [feedback, setFeedback] = useState({});

  // おすすめアイテムの取得
  const fetchRecommendations = useCallback(async (type = 'similar') => {
    try {
      setLoading(true);
      const response = await fetch(
        `/api/recommendations?type=${type}&userId=${userId}&itemId=${currentItem?.id}`
      );
      const data = await response.json();
      setRecommendations(data.recommendations);
    } catch (error) {
      console.error('おすすめアイテムの取得に失敗:', error);
    } finally {
      setLoading(false);
    }
  }, [userId, currentItem]);

  // 関連アイテムの取得
  const fetchRelatedItems = useCallback(async () => {
    if (!currentItem?.id) return;
    
    try {
      const response = await fetch(`/api/related-items/${currentItem.id}`);
      const data = await response.json();
      setRelatedItems(data.relatedItems);
    } catch (error) {
      console.error('関連アイテムの取得に失敗:', error);
    }
  }, [currentItem]);

  // 初期読み込み
  useEffect(() => {
    fetchRecommendations(activeFilter);
    fetchRelatedItems();
  }, [fetchRecommendations, fetchRelatedItems, activeFilter]);

  // フィルター変更
  const handleFilterChange = (type) => {
    setActiveFilter(type);
    fetchRecommendations(type);
  };

  // フィードバック送信
  const handleFeedback = async (itemId, action) => {
    try {
      await fetch('/api/recommendations/feedback', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          userId,
          itemId,
          action, // 'like', 'dislike', 'click', 'purchase'
          recommendationType: activeFilter
        })
      });

      setFeedback(prev => ({
        ...prev,
        [itemId]: action
      }));
    } catch (error) {
      console.error('フィードバック送信に失敗:', error);
    }
  };

  // おすすめ更新
  const handleRefresh = () => {
    fetchRecommendations(activeFilter);
  };

  if (loading) {
    return <div className="loading">おすすめを読み込み中...</div>;
  }

  return (
    <div>
      {/* おすすめセクション */}
      <section className="recommendations-section">
        <div className="recommendations-header">
          <h2>おすすめアイテム</h2>
          <div className="recommendation-filters">
            <button
              className={`filter-btn ${activeFilter === 'similar' ? 'active' : ''}`}
              onClick={() => handleFilterChange('similar')}
            >
              類似商品
            </button>
            <button
              className={`filter-btn ${activeFilter === 'popular' ? 'active' : ''}`}
              onClick={() => handleFilterChange('popular')}
            >
              人気商品
            </button>
            <button
              className={`filter-btn ${activeFilter === 'personal' ? 'active' : ''}`}
              onClick={() => handleFilterChange('personal')}
            >
              あなたにおすすめ
            </button>
          </div>
        </div>

        <div className="recommendations-grid">
          {recommendations.map(item => (
            <RecommendationItem
              key={item.id}
              item={item}
              onFeedback={handleFeedback}
              feedback={feedback[item.id]}
            />
          ))}
        </div>

        <div className="recommendation-actions">
          <button className="refresh-btn" onClick={handleRefresh}>
            おすすめを更新
          </button>
          <button className="feedback-btn">
            フィードバック
          </button>
        </div>
      </section>

      {/* 関連アイテムセクション */}
      {relatedItems.length > 0 && (
        <section className="related-items-section">
          <h3>この商品を見た人はこんな商品も見ています</h3>
          <div className="related-items-grid">
            {relatedItems.map(item => (
              <RelatedItem key={item.id} item={item} />
            ))}
          </div>
        </section>
      )}
    </div>
  );
};

// おすすめアイテムコンポーネント
const RecommendationItem = ({ item, onFeedback, feedback }) => {
  const handleLike = () => {
    onFeedback(item.id, 'like');
  };

  const handleDislike = () => {
    onFeedback(item.id, 'dislike');
  };

  const handleClick = () => {
    onFeedback(item.id, 'click');
    window.location.href = `/products/${item.id}`;
  };

  return (
    <div className="recommendation-item">
      <img
        src={item.image}
        alt={item.title}
        className="recommendation-item-image"
      />
      <div className="recommendation-item-content">
        <h3 className="recommendation-item-title">{item.title}</h3>
        <div className="recommendation-item-price">¥{item.price.toLocaleString()}</div>
        
        {item.rating && (
          <div className="recommendation-item-rating">
            {[...Array(5)].map((_, i) => (
              <span key={i} className={`star ${i < item.rating ? 'filled' : ''}`}>
                ★
              </span>
            ))}
            <span className="rating-text">({item.reviewCount})</span>
          </div>
        )}

        <div className="recommendation-item-actions">
          <button className="btn-primary" onClick={handleClick}>
            詳細を見る
          </button>
          <button
            className={`btn-secondary ${feedback === 'like' ? 'liked' : ''}`}
            onClick={handleLike}
          >
            ♥
          </button>
          <button
            className={`btn-secondary ${feedback === 'dislike' ? 'disliked' : ''}`}
            onClick={handleDislike}
          >
            ✗
          </button>
        </div>
      </div>
    </div>
  );
};

// 関連アイテムコンポーネント
const RelatedItem = ({ item }) => {
  const handleClick = () => {
    window.location.href = `/products/${item.id}`;
  };

  return (
    <div className="related-item" onClick={handleClick}>
      <img
        src={item.image}
        alt={item.title}
        className="related-item-image"
      />
      <div className="related-item-content">
        <h4 className="related-item-title">{item.title}</h4>
        <div className="related-item-price">¥{item.price.toLocaleString()}</div>
      </div>
    </div>
  );
};

// 使用例
function ProductPage({ product, user }) {
  return (
    <div>
      <h1>{product.name}</h1>
      <p>{product.description}</p>
      
      <RecommendationsSection
        currentItem={product}
        userId={user?.id}
      />
    </div>
  );
}

export default RecommendationsSection;
```

```vue
<script>
import RecommendationItem from './RecommendationItem.vue'
import RelatedItem from './RelatedItem.vue'

export default {
  name: 'RecommendationsSection',
  components: {
    RecommendationItem,
    RelatedItem
  },
  props: {
    currentItem: Object,
    userId: String
  },
  data() {
    return {
      recommendations: [],
      relatedItems: [],
      activeFilter: 'similar',
      loading: true,
      feedback: {},
      filters: [
        { type: 'similar', label: '類似商品' },
        { type: 'popular', label: '人気商品' },
        { type: 'personal', label: 'あなたにおすすめ' }
      ]
    }
  },
  async mounted() {
    await this.fetchRecommendations(this.activeFilter)
    await this.fetchRelatedItems()
  },
  methods: {
    async fetchRecommendations(type = 'similar') {
      try {
        this.loading = true
        const response = await fetch(
          `/api/recommendations?type=${type}&userId=${this.userId}&itemId=${this.currentItem?.id}`
        )
        const data = await response.json()
        this.recommendations = data.recommendations
      } catch (error) {
        console.error('おすすめアイテムの取得に失敗:', error)
      } finally {
        this.loading = false
      }
    },
    
    async fetchRelatedItems() {
      if (!this.currentItem?.id) return
      
      try {
        const response = await fetch(`/api/related-items/${this.currentItem.id}`)
        const data = await response.json()
        this.relatedItems = data.relatedItems
      } catch (error) {
        console.error('関連アイテムの取得に失敗:', error)
      }
    },
    
    handleFilterChange(type) {
      this.activeFilter = type
      this.fetchRecommendations(type)
    },
    
    async handleFeedback(itemId, action) {
      try {
        await fetch('/api/recommendations/feedback', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            userId: this.userId,
            itemId,
            action,
            recommendationType: this.activeFilter
          })
        })

        this.$set(this.feedback, itemId, action)
      } catch (error) {
        console.error('フィードバック送信に失敗:', error)
      }
    },
    
    handleRefresh() {
      this.fetchRecommendations(this.activeFilter)
    }
  }
}
```

## バックエンド実装例

### Node.js + Express での実装
```javascript
const express = require('express');
const router = express.Router();

// おすすめアイテムの取得
router.get('/recommendations', authenticateUser, async (req, res) => {
  try {
    const { type = 'similar', userId, itemId } = req.query;
    
    let recommendations = [];
    
    switch (type) {
      case 'similar':
        recommendations = await getSimilarItems(itemId, userId);
        break;
      case 'popular':
        recommendations = await getPopularItems(userId);
        break;
      case 'personal':
        recommendations = await getPersonalizedRecommendations(userId);
        break;
      default:
        recommendations = await getSimilarItems(itemId, userId);
    }
    
    res.json({ recommendations });
  } catch (error) {
    console.error('おすすめ取得エラー:', error);
    res.status(500).json({ error: 'サーバーエラーが発生しました' });
  }
});

// 類似商品の取得
const getSimilarItems = async (itemId, userId) => {
  // 現在の商品の特徴を取得
  const currentItem = await Item.findByPk(itemId, {
    include: ['categories', 'tags']
  });
  
  if (!currentItem) return [];
  
  // 類似商品を検索
  const similarItems = await Item.findAll({
    where: {
      id: { [Op.ne]: itemId },
      [Op.or]: [
        { categoryId: currentItem.categoryId },
        { '$categories.id$': { [Op.in]: currentItem.categories.map(c => c.id) } },
        { '$tags.id$': { [Op.in]: currentItem.tags.map(t => t.id) } }
      ]
    },
    include: ['categories', 'tags'],
    limit: 10,
    order: [
      // カテゴリ一致度でソート
      [sequelize.literal('CASE WHEN categoryId = ' + currentItem.categoryId + ' THEN 1 ELSE 2 END')],
      ['rating', 'DESC'],
      ['createdAt', 'DESC']
    ]
  });
  
  return similarItems;
};

// 人気商品の取得
const getPopularItems = async (userId) => {
  const popularItems = await Item.findAll({
    include: [
      {
        model: View,
        attributes: [],
        where: {
          createdAt: {
            [Op.gte]: new Date(Date.now() - 30 * 24 * 60 * 60 * 1000) // 30日間
          }
        }
      }
    ],
    attributes: {
      include: [
        [sequelize.fn('COUNT', sequelize.col('Views.id')), 'viewCount']
      ]
    },
    group: ['Item.id'],
    order: [[sequelize.literal('viewCount'), 'DESC']],
    limit: 10
  });
  
  return popularItems;
};

// パーソナライズされたおすすめの取得
const getPersonalizedRecommendations = async (userId) => {
  // ユーザーの行動履歴を分析
  const userBehavior = await analyzeUserBehavior(userId);
  
  // 協調フィルタリング
  const collaborativeRecommendations = await getCollaborativeRecommendations(userId);
  
  // コンテンツベースフィルタリング
  const contentBasedRecommendations = await getContentBasedRecommendations(userId);
  
  // ハイブリッドアプローチで統合
  const recommendations = combineRecommendations(
    collaborativeRecommendations,
    contentBasedRecommendations,
    userBehavior
  );
  
  return recommendations;
};

// 関連アイテムの取得
router.get('/related-items/:itemId', async (req, res) => {
  try {
    const { itemId } = req.params;
    
    // 一緒に閲覧されることが多い商品を取得
    const relatedItems = await getCoViewedItems(itemId);
    
    res.json({ relatedItems });
  } catch (error) {
    console.error('関連アイテム取得エラー:', error);
    res.status(500).json({ error: 'サーバーエラーが発生しました' });
  }
});

// 一緒に閲覧される商品の取得
const getCoViewedItems = async (itemId) => {
  const coViewedItems = await sequelize.query(`
    SELECT 
      i2.id,
      i2.title,
      i2.price,
      i2.image,
      COUNT(*) as co_view_count
    FROM views v1
    JOIN views v2 ON v1.user_id = v2.user_id AND v1.item_id != v2.item_id
    JOIN items i2 ON v2.item_id = i2.id
    WHERE v1.item_id = :itemId
      AND v1.created_at >= DATE_SUB(NOW(), INTERVAL 30 DAY)
      AND v2.created_at >= DATE_SUB(NOW(), INTERVAL 30 DAY)
    GROUP BY i2.id
    ORDER BY co_view_count DESC
    LIMIT 8
  `, {
    replacements: { itemId },
    type: sequelize.QueryTypes.SELECT
  });
  
  return coViewedItems;
};

// フィードバックの処理
router.post('/recommendations/feedback', authenticateUser, async (req, res) => {
  try {
    const { userId, itemId, action, recommendationType } = req.body;
    
    // フィードバックを保存
    await RecommendationFeedback.create({
      userId,
      itemId,
      action,
      recommendationType,
      createdAt: new Date()
    });
    
    // ユーザーの嗜好を更新
    await updateUserPreferences(userId, itemId, action);
    
    res.json({ message: 'フィードバックを受け付けました' });
  } catch (error) {
    console.error('フィードバック処理エラー:', error);
    res.status(500).json({ error: 'サーバーエラーが発生しました' });
  }
});

// ユーザー嗜好の更新
const updateUserPreferences = async (userId, itemId, action) => {
  const item = await Item.findByPk(itemId, {
    include: ['categories', 'tags']
  });
  
  if (!item) return;
  
  // 嗜好スコアの計算
  let preferenceScore = 0;
  switch (action) {
    case 'like':
      preferenceScore = 1;
      break;
    case 'dislike':
      preferenceScore = -1;
      break;
    case 'click':
      preferenceScore = 0.5;
      break;
    case 'purchase':
      preferenceScore = 2;
      break;
  }
  
  // カテゴリ嗜好の更新
```

```javascript
// 類似ユーザーの検索
const findSimilarUsers = async (userId) => {
  const userRatings = await UserRating.findAll({
    where: { userId },
    attributes: ['itemId', 'rating']
  });
  
  const userRatingMap = new Map();
  userRatings.forEach(rating => {
    userRatingMap.set(rating.itemId, rating.rating);
  });
  
  // 他のユーザーとの類似度を計算
  const similarities = await sequelize.query(`
    SELECT 
      u2.id as userId,
      SUM((r1.rating - r1_avg.avg_rating) * (r2.rating - r2_avg.avg_rating)) / 
      (SQRT(SUM(POW(r1.rating - r1_avg.avg_rating, 2))) * 
       SQRT(SUM(POW(r2.rating - r2_avg.avg_rating, 2)))) as similarity
    FROM user_ratings r1
    JOIN user_ratings r2 ON r1.itemId = r2.itemId
    JOIN users u2 ON r2.userId = u2.id
    JOIN (
      SELECT userId, AVG(rating) as avg_rating
      FROM user_ratings
      GROUP BY userId
    ) r1_avg ON r1.userId = r1_avg.userId
    JOIN (
      SELECT userId, AVG(rating) as avg_rating
      FROM user_ratings
      GROUP BY userId
    ) r2_avg ON r2.userId = r2_avg.userId
    WHERE r1.userId = :userId
      AND r2.userId != :userId
    GROUP BY u2.id
    HAVING similarity > 0.3
    ORDER BY similarity DESC
    LIMIT 20
  `, {
    replacements: { userId },
    type: sequelize.QueryTypes.SELECT
  });
  
  return similarities;
};
```

```javascript
// レコメンデーションの統合
const combineRecommendations = (recommendations, weights) => {
  const itemScores = new Map();
  
  // 各アルゴリズムの結果をスコア化
  Object.entries(recommendations).forEach(([algorithm, items]) => {
    const weight = weights[algorithm] || 0;
    
    items.forEach((item, index) => {
      const score = (items.length - index) / items.length * weight;
      
      if (itemScores.has(item.id)) {
        itemScores.set(item.id, itemScores.get(item.id) + score);
      } else {
        itemScores.set(item.id, score);
      }
    });
  });
  
  // スコア順にソート
  const sortedItems = Array.from(itemScores.entries())
    .sort((a, b) => b[1] - a[1])
    .map(([itemId, score]) => ({ itemId, score }));
  
  return sortedItems;
};
```

### 2. A/Bテストの実装
```javascript
// A/Bテスト機能
class RecommendationABTest {
  constructor() {
    this.variants = {
      'control': { weight: 0.5, algorithm: 'collaborative' },
      'variant_a': { weight: 0.3, algorithm: 'hybrid' },
      'variant_b': { weight: 0.2, algorithm: 'content_based' }
    };
  }
  
  getVariant(userId) {
    const hash = this.hashUserId(userId);
    const random = hash % 100;
    
    let cumulative = 0;
    for (const [variant, config] of Object.entries(this.variants)) {
      cumulative += config.weight * 100;
      if (random < cumulative) {
        return variant;
      }
    }
    
    return 'control';
  }
  
  hashUserId(userId) {
    let hash = 0;
    for (let i = 0; i < userId.length; i++) {
      const char = userId.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash; // 32bit整数に変換
    }
    return Math.abs(hash);
  }
  
  async getRecommendations(userId) {
    const variant = this.getVariant(userId);
    const algorithm = this.variants[variant].algorithm;
    
    // アルゴリズムに応じてレコメンデーションを取得
    const recommendations = await this.getRecommendationsByAlgorithm(userId, algorithm);
    
    // バリアント情報を記録
    await this.recordVariantUsage(userId, variant);
    
    return recommendations;
  }
}
```
