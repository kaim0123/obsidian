# 同期機能の実装方法とベストプラクティス

## 🔄 概要

同期機能は、複数のデバイスやクライアント間でデータの整合性を保ち、リアルタイムで情報を共有するための重要な機能です。適切な同期機能により、ユーザーはどのデバイスからでも最新のデータにアクセスでき、一貫した体験を得ることができます。

## 🎯 主要機能

### 1. リアルタイム同期
- **WebSocket 通信** - 双方向リアルタイム通信
- **Server-Sent Events** - サーバーからの一方向通信
- **ポーリング** - 定期的なデータ取得
- **プッシュ通知** - 変更の即座な通知

### 2. データ同期
- **差分同期** - 変更部分のみの同期
- **競合解決** - データ競合の処理
- **バージョン管理** - データのバージョン制御
- **整合性チェック** - データの整合性検証

### 3. 状態管理
- **楽観的更新** - UI の即座な更新
- **悲観的更新** - サーバー確認後の更新
- **ハイブリッド更新** - 状況に応じた更新戦略
- **ロールバック** - エラー時の状態復元

### 4. 同期トポロジー（システム構成の視点）

旧・概要ノートで整理されていた分類をここに統合します。

- **マスター・スレーブ** — 単一の正（マスター）へ書き込み、他ノードは追従
- **マルチマスター** — 複数ノードが書き込み可能（競合解決・バージョン設計が必須）
- **ピアツーピア（P2P）** — クライアント同士やエッジ間で直接同期

また、タイミングの切り口では **バッチ同期**（一定間隔での一括反映）もよく用いられます。実装のヒントは後述の「バッチ処理」の節も参照してください。

### WebSocket を使用したリアルタイム同期

```javascript
// WebSocket 接続管理クラス
class WebSocketManager {
  constructor(url, options = {}) {
    this.url = url;
    this.options = options;
    this.ws = null;
    this.reconnectAttempts = 0;
    this.maxReconnectAttempts = 5;
    this.reconnectInterval = 1000;
    this.listeners = new Map();
    this.isConnecting = false;
  }

  // 接続の確立
  connect() {
    if (this.isConnecting || (this.ws && this.ws.readyState === WebSocket.OPEN)) {
      return;
    }

    this.isConnecting = true;

    try {
      this.ws = new WebSocket(this.url);

      this.ws.onopen = () => {
        console.log('WebSocket 接続が確立されました');
        this.isConnecting = false;
        this.reconnectAttempts = 0;
        this.emit('connected');
      };

      this.ws.onmessage = (event) => {
        try {
          const data = JSON.parse(event.data);
          this.handleMessage(data);
        } catch (error) {
          console.error('メッセージの解析に失敗:', error);
        }
      };

      this.ws.onclose = (event) => {
        console.log('WebSocket 接続が閉じられました:', event.code, event.reason);
        this.isConnecting = false;
        this.emit('disconnected', event);
        
        if (!event.wasClean && this.reconnectAttempts < this.maxReconnectAttempts) {
          this.scheduleReconnect();
        }
      };

      this.ws.onerror = (error) => {
        console.error('WebSocket エラー:', error);
        this.isConnecting = false;
        this.emit('error', error);
      };

    } catch (error) {
      console.error('WebSocket 接続エラー:', error);
      this.isConnecting = false;
      this.emit('error', error);
    }
  }

  // メッセージの処理
  handleMessage(data) {
    const { type, payload } = data;
    
    switch (type) {
      case 'data_update':
        this.emit('dataUpdate', payload);
        break;
      case 'user_joined':
        this.emit('userJoined', payload);
        break;
      case 'user_left':
        this.emit('userLeft', payload);
        break;
      case 'error':
        this.emit('error', payload);
        break;
      default:
        console.warn('未知のメッセージタイプ:', type);
    }
  }

  // メッセージの送信
  send(type, payload) {
    if (this.ws && this.ws.readyState === WebSocket.OPEN) {
      const message = { type, payload, timestamp: Date.now() };
      this.ws.send(JSON.stringify(message));
    } else {
      console.warn('WebSocket が接続されていません');
    }
  }

  // 再接続のスケジュール
  scheduleReconnect() {
    this.reconnectAttempts++;
    const delay = this.reconnectInterval * Math.pow(2, this.reconnectAttempts - 1);
    
    setTimeout(() => {
      console.log(`再接続を試行中... (${this.reconnectAttempts}/${this.maxReconnectAttempts})`);
      this.connect();
    }, delay);
  }

  // イベントリスナーの登録
  on(event, callback) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, []);
    }
    this.listeners.get(event).push(callback);
  }

  // イベントリスナーの削除
  off(event, callback) {
    if (this.listeners.has(event)) {
      const callbacks = this.listeners.get(event);
      const index = callbacks.indexOf(callback);
      if (index > -1) {
        callbacks.splice(index, 1);
      }
    }
  }

  // イベントの発生
  emit(event, data) {
    if (this.listeners.has(event)) {
      this.listeners.get(event).forEach(callback => {
        try {
          callback(data);
        } catch (error) {
          console.error('イベントコールバックエラー:', error);
        }
      });
    }
  }

  // 接続の切断
  disconnect() {
    if (this.ws) {
      this.ws.close(1000, '正常終了');
      this.ws = null;
    }
  }
}

// 使用例
const wsManager = new WebSocketManager('ws://localhost:8080');

wsManager.on('dataUpdate', (data) => {
  console.log('データが更新されました:', data);
  // UI の更新処理
});

wsManager.on('userJoined', (user) => {
  console.log('ユーザーが参加しました:', user);
  // ユーザーリストの更新
});

wsManager.connect();
```

### React Hook を使用した同期状態管理

```javascript
import { useState, useEffect, useCallback, useRef } from 'react';

// 同期状態管理フック
function useSyncState(initialState, syncConfig) {
  const [state, setState] = useState(initialState);
  const [isSyncing, setIsSyncing] = useState(false);
  const [syncError, setSyncError] = useState(null);
  const [lastSyncTime, setLastSyncTime] = useState(null);
  const wsManagerRef = useRef(null);
  const pendingChangesRef = useRef(new Set());

  // WebSocket 接続の初期化
  useEffect(() => {
    if (syncConfig.websocket) {
      wsManagerRef.current = new WebSocketManager(syncConfig.websocket.url);
      
      wsManagerRef.current.on('dataUpdate', handleRemoteUpdate);
      wsManagerRef.current.on('connected', () => {
        console.log('同期接続が確立されました');
        setSyncError(null);
      });
      wsManagerRef.current.on('disconnected', () => {
        console.log('同期接続が切断されました');
      });
      wsManagerRef.current.on('error', (error) => {
        console.error('同期エラー:', error);
        setSyncError(error);
      });

      wsManagerRef.current.connect();

      return () => {
        wsManagerRef.current?.disconnect();
      };
    }
  }, [syncConfig.websocket]);

  // リモート更新の処理
  const handleRemoteUpdate = useCallback((data) => {
    const { id, changes, timestamp } = data;
    
    // 自分の変更でない場合のみ適用
    if (!pendingChangesRef.current.has(id)) {
      setState(prevState => {
        if (syncConfig.mergeStrategy) {
          return syncConfig.mergeStrategy(prevState, changes);
        }
        return { ...prevState, ...changes };
      });
      setLastSyncTime(timestamp);
    }
  }, [syncConfig.mergeStrategy]);

  // 状態の更新
  const updateState = useCallback(async (changes, options = {}) => {
    const { optimistic = true, sync = true } = options;
    
    // 楽観的更新
    if (optimistic) {
      setState(prevState => ({ ...prevState, ...changes }));
    }

    // 同期の実行
    if (sync && wsManagerRef.current) {
      setIsSyncing(true);
      setSyncError(null);

      try {
        const changeId = `change_${Date.now()}`;
        pendingChangesRef.current.add(changeId);

        // WebSocket で変更を送信
        wsManagerRef.current.send('data_update', {
          id: changeId,
          changes,
          timestamp: Date.now(),
        });

        // サーバーからの確認を待つ
        await new Promise((resolve, reject) => {
          const timeout = setTimeout(() => {
            reject(new Error('同期タイムアウト'));
          }, 5000);

          const handleConfirm = (data) => {
            if (data.id === changeId) {
              clearTimeout(timeout);
              pendingChangesRef.current.delete(changeId);
              resolve();
            }
          };

          wsManagerRef.current.on('dataUpdate', handleConfirm);
        });

        setLastSyncTime(Date.now());
      } catch (error) {
        console.error('同期エラー:', error);
        setSyncError(error);
        
        // エラー時のロールバック
        if (optimistic) {
          setState(prevState => {
            const reverted = { ...prevState };
            Object.keys(changes).forEach(key => {
              // 前の状態に戻す（実際の実装ではより複雑なロールバックが必要）
              delete reverted[key];
            });
            return reverted;
          });
        }
      } finally {
        setIsSyncing(false);
      }
    }
  }, []);

  // 手動同期
  const sync = useCallback(async () => {
    if (wsManagerRef.current) {
      setIsSyncing(true);
      setSyncError(null);

      try {
        wsManagerRef.current.send('sync_request', {
          timestamp: lastSyncTime,
        });
        setLastSyncTime(Date.now());
      } catch (error) {
        setSyncError(error);
      } finally {
        setIsSyncing(false);
      }
    }
  }, [lastSyncTime]);

  return {
    state,
    updateState,
    isSyncing,
    syncError,
    lastSyncTime,
    sync,
  };
}

// 使用例
function SyncComponent() {
  const { state, updateState, isSyncing, syncError, sync } = useSyncState(
    { count: 0, message: '' },
    {
      websocket: {
        url: 'ws://localhost:8080',
      },
      mergeStrategy: (prevState, changes) => {
        // カスタムマージ戦略
        return { ...prevState, ...changes };
      },
    }
  );

  const handleIncrement = () => {
    updateState({ count: state.count + 1 });
  };

  const handleMessageChange = (message) => {
    updateState({ message });
  };

  return (
    <div>
      <div>
        <span>カウント: {state.count}</span>
        <button onClick={handleIncrement} disabled={isSyncing}>
          {isSyncing ? '同期中...' : 'インクリメント'}
        </button>
      </div>
      
      <div>
        <input
          value={state.message}
          onChange={(e) => handleMessageChange(e.target.value)}
          placeholder="メッセージを入力"
        />
      </div>

      {syncError && (
        <div className="error">
          同期エラー: {syncError.message}
          <button onClick={sync}>再試行</button>
        </div>
      )}

      <button onClick={sync} disabled={isSyncing}>
        手動同期
      </button>
    </div>
  );
}
```

### 競合解決の実装

```javascript
// 競合解決クラス
class ConflictResolver {
  constructor(strategies = {}) {
    this.strategies = {
      lastWriteWins: this.lastWriteWins.bind(this),
      firstWriteWins: this.firstWriteWins.bind(this),
      merge: this.merge.bind(this),
      userChoice: this.userChoice.bind(this),
      ...strategies,
    };
  }

  // 最後の書き込みが勝つ
  lastWriteWins(conflict) {
    const { local, remote } = conflict;
    return remote.timestamp > local.timestamp ? remote : local;
  }

  // 最初の書き込みが勝つ
  firstWriteWins(conflict) {
    const { local, remote } = conflict;
    return remote.timestamp < local.timestamp ? remote : local;
  }

  // マージ戦略
  merge(conflict) {
    const { local, remote } = conflict;
    const merged = { ...local, ...remote };
    
    // 配列の場合は結合
    Object.keys(merged).forEach(key => {
      if (Array.isArray(local[key]) && Array.isArray(remote[key])) {
        merged[key] = [...new Set([...local[key], ...remote[key]])];
      }
    });
    
    return merged;
  }

  // ユーザー選択
  userChoice(conflict, userChoice) {
    return userChoice === 'local' ? conflict.local : conflict.remote;
  }

  // 競合の解決
  resolve(conflict, strategy = 'lastWriteWins', userChoice = null) {
    if (!this.strategies[strategy]) {
      throw new Error(`未知の競合解決戦略: ${strategy}`);
    }

    return this.strategies[strategy](conflict, userChoice);
  }
}

// 競合解決フック
function useConflictResolution() {
  const [conflicts, setConflicts] = useState([]);
  const resolver = useRef(new ConflictResolver());

  const addConflict = useCallback((conflict) => {
    setConflicts(prev => [...prev, { ...conflict, id: Date.now() }]);
  }, []);

  const resolveConflict = useCallback((conflictId, strategy, userChoice = null) => {
    const conflict = conflicts.find(c => c.id === conflictId);
    if (!conflict) return;

    const resolved = resolver.current.resolve(conflict, strategy, userChoice);
    
    setConflicts(prev => prev.filter(c => c.id !== conflictId));
    
    return resolved;
  }, [conflicts]);

  const clearConflicts = useCallback(() => {
    setConflicts([]);
  }, []);

  return {
    conflicts,
    addConflict,
    resolveConflict,
    clearConflicts,
  };
}
```

### 差分同期の実装

```javascript
// 差分同期クラス
class DiffSync {
  constructor() {
    this.patches = new Map();
  }

  // オブジェクトの差分を計算
  calculateDiff(oldObj, newObj) {
    const diff = {};
    
    // 変更されたプロパティを検出
    Object.keys(newObj).forEach(key => {
      if (oldObj[key] !== newObj[key]) {
        diff[key] = newObj[key];
      }
    });

    // 削除されたプロパティを検出
    Object.keys(oldObj).forEach(key => {
      if (!(key in newObj)) {
        diff[key] = null; // 削除マーカー
      }
    });

    return diff;
  }

  // パッチの適用
  applyPatch(obj, patch) {
    const result = { ...obj };
    
    Object.keys(patch).forEach(key => {
      if (patch[key] === null) {
        delete result[key];
      } else {
        result[key] = patch[key];
      }
    });

    return result;
  }

  // パッチの保存
  savePatch(id, patch) {
    this.patches.set(id, {
      patch,
      timestamp: Date.now(),
    });
  }

  // パッチの取得
  getPatch(id) {
    return this.patches.get(id);
  }

  // パッチの削除
  removePatch(id) {
    this.patches.delete(id);
  }

  // 同期の実行
  sync(localObj, remoteObj, lastSyncTime) {
    const localDiff = this.calculateDiff(remoteObj, localObj);
    const remoteDiff = this.calculateDiff(localObj, remoteObj);

    // 競合の検出
    const conflicts = this.detectConflicts(localDiff, remoteDiff);
    
    if (conflicts.length > 0) {
      return { conflicts, needsResolution: true };
    }

    // 競合がない場合はマージ
    const merged = this.applyPatch(localObj, remoteDiff);
    return { merged, needsResolution: false };
  }

  // 競合の検出
  detectConflicts(localDiff, remoteDiff) {
    const conflicts = [];
    
    Object.keys(localDiff).forEach(key => {
      if (key in remoteDiff && localDiff[key] !== remoteDiff[key]) {
        conflicts.push({
          key,
          local: localDiff[key],
          remote: remoteDiff[key],
        });
      }
    });

    return conflicts;
  }
}

// 差分同期フック
function useDiffSync(initialData) {
  const [data, setData] = useState(initialData);
  const [lastSyncTime, setLastSyncTime] = useState(Date.now());
  const diffSync = useRef(new DiffSync());

  const updateData = useCallback((newData) => {
    setData(prevData => {
      const diff = diffSync.current.calculateDiff(prevData, newData);
      diffSync.current.savePatch(`local_${Date.now()}`, diff);
      return newData;
    });
  }, []);

  const syncWithRemote = useCallback(async (remoteData) => {
    const syncResult = diffSync.current.sync(data, remoteData, lastSyncTime);
    
    if (syncResult.needsResolution) {
      // 競合解決が必要
      return { conflicts: syncResult.conflicts, data };
    } else {
      // 同期完了
      setData(syncResult.merged);
      setLastSyncTime(Date.now());
      return { conflicts: [], data: syncResult.merged };
    }
  }, [data, lastSyncTime]);

  return {
    data,
    updateData,
    syncWithRemote,
    lastSyncTime,
  };
}
```

```javascript
function SyncStatusIndicator({ isSyncing, lastSyncTime, syncError }) {
  const [timeSinceSync, setTimeSinceSync] = useState('');

    if (lastSyncTime) {
      const updateTime = () => {

            <span className="spinner">🔄</span>
```

```javascript
class BatchProcessor {
  constructor(batchSize = 10, batchTimeout = 1000) {
    this.batchSize = batchSize;
    this.batchTimeout = batchTimeout;
    this.queue = [];
    this.timeoutId = null;
  }

  add(item) {
    this.queue.push(item);
    
    if (this.queue.length >= this.batchSize) {
      this.processBatch();
    } else if (!this.timeoutId) {
      this.timeoutId = setTimeout(() => {
        this.processBatch();
      }, this.batchTimeout);
    }
  }

  processBatch() {
    if (this.queue.length === 0) return;

    const batch = this.queue.splice(0, this.batchSize);
    
    if (this.timeoutId) {
      clearTimeout(this.timeoutId);
      this.timeoutId = null;
    }

    // バッチ処理の実行
    this.onBatch(batch);
  }

  onBatch(batch) {
    // サブクラスで実装
  }
}
```

```javascript
function useDebounce(value, delay) {
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => {
      clearTimeout(handler);
    };
  }, [value, delay]);

  return debouncedValue;
}

// 使用例
function SyncInput({ value, onChange, onSync }) {
  const debouncedValue = useDebounce(value, 500);

  useEffect(() => {
    if (debouncedValue !== value) {
      onSync(debouncedValue);
    }
  }, [debouncedValue, value, onSync]);

  return (
    <input
      value={value}
      onChange={(e) => onChange(e.target.value)}
    />
  );
}
```
