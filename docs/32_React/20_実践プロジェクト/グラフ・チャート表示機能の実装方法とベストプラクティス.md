## 1. Chart.js

### 概要
最も人気のあるJavaScriptチャートライブラリの一つ

### 実装例

```javascript
// Chart.jsの基本的な実装
import Chart from 'chart.js/auto';

function BasicChart({ data, type = 'bar' }) {
  const chartRef = useRef(null);
  const chartInstance = useRef(null);

  useEffect(() => {
    if (chartRef.current) {
      // 既存のチャートを破棄
      if (chartInstance.current) {
        chartInstance.current.destroy();
      }

      // 新しいチャートを作成
      chartInstance.current = new Chart(chartRef.current, {
        type: type,
        data: {
          labels: data.labels,
          datasets: [{
            label: data.datasetLabel,
            data: data.values,
            backgroundColor: data.backgroundColor || [
              'rgba(255, 99, 132, 0.2)',
              'rgba(54, 162, 235, 0.2)',
              'rgba(255, 205, 86, 0.2)',
              'rgba(75, 192, 192, 0.2)',
              'rgba(153, 102, 255, 0.2)',
            ],
            borderColor: data.borderColor || [
              'rgba(255, 99, 132, 1)',
              'rgba(54, 162, 235, 1)',
              'rgba(255, 205, 86, 1)',
              'rgba(75, 192, 192, 1)',
              'rgba(153, 102, 255, 1)',
            ],
            borderWidth: 1
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: {
            legend: {
              position: 'top',
            },
            title: {
              display: true,
              text: data.title || 'グラフタイトル'
            }
          },
          scales: {
            y: {
              beginAtZero: true
            }
          }
        }
      });
    }

    return () => {
      if (chartInstance.current) {
        chartInstance.current.destroy();
      }
    };
  }, [data, type]);

  return (
    <div className="chart-container">
      <canvas ref={chartRef}></canvas>
    </div>
  );
}
```

### 高度なChart.js実装

```javascript
// 複数のデータセットを持つ高度なグラフ
function AdvancedChart({ data }) {
  const chartRef = useRef(null);
  const chartInstance = useRef(null);

  useEffect(() => {
    if (chartRef.current && data) {
      if (chartInstance.current) {
        chartInstance.current.destroy();
      }

      chartInstance.current = new Chart(chartRef.current, {
        type: 'line',
        data: {
          labels: data.labels,
          datasets: data.datasets.map((dataset, index) => ({
            label: dataset.label,
            data: dataset.data,
            borderColor: dataset.color,
            backgroundColor: dataset.color + '20',
            borderWidth: 2,
            fill: dataset.fill || false,
            tension: 0.4,
            pointRadius: 4,
            pointHoverRadius: 6,
            pointBackgroundColor: dataset.color,
            pointBorderColor: '#fff',
            pointBorderWidth: 2
          }))
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          interaction: {
            intersect: false,
            mode: 'index'
          },
          plugins: {
            legend: {
              position: 'top',
              labels: {
                usePointStyle: true,
                padding: 20
              }
            },
            tooltip: {
              backgroundColor: 'rgba(0, 0, 0, 0.8)',
              titleColor: '#fff',
              bodyColor: '#fff',
              borderColor: '#fff',
              borderWidth: 1,
              cornerRadius: 8,
              displayColors: true,
              callbacks: {
                title: function(context) {
                  return `期間: ${context[0].label}`;
                },
                label: function(context) {
                  return `${context.dataset.label}: ${context.parsed.y.toLocaleString()}`;
                }
              }
            }
          },
          scales: {
            x: {
              display: true,
              title: {
                display: true,
                text: '期間'
              },
              grid: {
                display: false
              }
            },
            y: {
              display: true,
              title: {
                display: true,
                text: '値'
              },
              beginAtZero: true,
              grid: {
                color: 'rgba(0, 0, 0, 0.1)'
              },
              ticks: {
                callback: function(value) {
                  return value.toLocaleString();
                }
              }
            }
          },
          animation: {
            duration: 1000,
            easing: 'easeInOutQuart'
          }
        }
      });
    }

    return () => {
      if (chartInstance.current) {
        chartInstance.current.destroy();
      }
    };
  }, [data]);

  return (
    <div className="advanced-chart-container">
      <canvas ref={chartRef}></canvas>
    </div>
  );
}
```

- ✅ **豊富なチャートタイプ**: 8種類の基本的なチャートタイプ
- ✅ **アニメーション**: デフォルトでアニメーション対応
- ✅ **レスポンシブ**: 自動的にレスポンシブ対応

### デメリット

- ❌ **Reactとの統合**: 直接的なReactコンポーネントではない
- ❌ **パフォーマンス**: 大量データでは重くなる

### 使用場面

```javascript
// D3.jsの基本的な実装
import * as d3 from 'd3';
import { useEffect, useRef } from 'react';

function D3BarChart({ data, width = 800, height = 400 }) {
  const svgRef = useRef(null);

  useEffect(() => {
    if (!data || !svgRef.current) return;

    const svg = d3.select(svgRef.current);
    svg.selectAll("*").remove(); // 既存の要素をクリア

    const margin = { top: 20, right: 30, bottom: 40, left: 40 };
    const chartWidth = width - margin.left - margin.right;
    const chartHeight = height - margin.top - margin.bottom;

    // スケールの設定
    const xScale = d3.scaleBand()
      .domain(data.map(d => d.name))
      .range([0, chartWidth])
      .padding(0.1);

    const yScale = d3.scaleLinear()
      .domain([0, d3.max(data, d => d.value)])
      .range([chartHeight, 0]);

    // SVG要素の作成
    const g = svg.append("g")
      .attr("transform", `translate(${margin.left},${margin.top})`);

    // バーの描画
    g.selectAll(".bar")
      .data(data)
      .enter().append("rect")
      .attr("class", "bar")
      .attr("x", d => xScale(d.name))
      .attr("width", xScale.bandwidth())
      .attr("y", chartHeight)
      .attr("height", 0)
      .attr("fill", "#8884d8")
      .transition()
      .duration(1000)
      .attr("y", d => yScale(d.value))
      .attr("height", d => chartHeight - yScale(d.value));

    // X軸の描画
    g.append("g")
      .attr("transform", `translate(0,${chartHeight})`)
      .call(d3.axisBottom(xScale))
      .selectAll("text")
      .style("text-anchor", "end")
      .attr("dx", "-.8em")
      .attr("dy", ".15em")
      .attr("transform", "rotate(-45)");

    // Y軸の描画
    g.append("g")
      .call(d3.axisLeft(yScale));

    // ツールチップの追加
    const tooltip = d3.select("body").append("div")
      .attr("class", "tooltip")
      .style("opacity", 0)
      .style("position", "absolute")
      .style("background", "rgba(0, 0, 0, 0.8)")
      .style("color", "white")
      .style("padding", "8px")
      .style("border-radius", "4px")
      .style("font-size", "12px");

    g.selectAll(".bar")
      .on("mouseover", function(event, d) {
        tooltip.transition()
          .duration(200)
          .style("opacity", .9);
        tooltip.html(`${d.name}: ${d.value.toLocaleString()}`)
          .style("left", (event.pageX + 10) + "px")
          .style("top", (event.pageY - 28) + "px");
      })
      .on("mouseout", function(d) {
        tooltip.transition()
          .duration(500)
          .style("opacity", 0);
      });

    return () => {
      d3.select("body").selectAll(".tooltip").remove();
    };
  }, [data, width, height]);

  return (
    <div className="d3-chart-container">
      <svg ref={svgRef} width={width} height={height}></svg>
    </div>
  );
}
```

### 高度なD3.js実装

```javascript
// 複雑なインタラクティブなチャート
function AdvancedD3Chart({ data }) {
  const svgRef = useRef(null);

  useEffect(() => {
    if (!data || !svgRef.current) return;

    const svg = d3.select(svgRef.current);
    svg.selectAll("*").remove();

    const width = 800;
    const height = 500;
    const margin = { top: 20, right: 80, bottom: 30, left: 50 };
    const chartWidth = width - margin.left - margin.right;
    const chartHeight = height - margin.top - margin.bottom;

    // スケールの設定
    const xScale = d3.scaleTime()
      .domain(d3.extent(data, d => new Date(d.date)))
      .range([0, chartWidth]);

    const yScale = d3.scaleLinear()
      .domain(d3.extent(data, d => d.value))
      .range([chartHeight, 0]);

    // ラインジェネレーター
    const line = d3.line()
      .x(d => xScale(new Date(d.date)))
      .y(d => yScale(d.value))
      .curve(d3.curveMonotoneX);

    // エリアジェネレーター
    const area = d3.area()
      .x(d => xScale(new Date(d.date)))
      .y0(chartHeight)
      .y1(d => yScale(d.value))
      .curve(d3.curveMonotoneX);

    const g = svg.append("g")
      .attr("transform", `translate(${margin.left},${margin.top})`);

    // エリアの描画
    g.append("path")
      .datum(data)
      .attr("fill", "rgba(136, 132, 216, 0.2)")
      .attr("d", area);

    // ラインの描画
    g.append("path")
      .datum(data)
      .attr("fill", "none")
      .attr("stroke", "#8884d8")
      .attr("stroke-width", 2)
      .attr("d", line);

    // データポイントの描画
    g.selectAll(".dot")
      .data(data)
      .enter().append("circle")
      .attr("class", "dot")
      .attr("cx", d => xScale(new Date(d.date)))
      .attr("cy", d => yScale(d.value))
      .attr("r", 4)
      .attr("fill", "#8884d8")
      .attr("stroke", "#fff")
      .attr("stroke-width", 2)
      .on("mouseover", function(event, d) {
        d3.select(this)
          .transition()
          .duration(200)
          .attr("r", 6);
      })
      .on("mouseout", function(d) {
        d3.select(this)
          .transition()
          .duration(200)
          .attr("r", 4);
      });

    // X軸の描画
    g.append("g")
      .attr("transform", `translate(0,${chartHeight})`)
      .call(d3.axisBottom(xScale).tickFormat(d3.timeFormat("%Y-%m")));

    // Y軸の描画
    g.append("g")
      .call(d3.axisLeft(yScale).tickFormat(d => d.toLocaleString()));

    // グリッドラインの追加
    g.append("g")
      .attr("class", "grid")
      .attr("transform", `translate(0,${chartHeight})`)
      .call(d3.axisBottom(xScale)
        .tickSize(-chartHeight)
        .tickFormat("")
      )
      .style("stroke-dasharray", "3,3")
      .style("opacity", 0.3);

    g.append("g")
      .attr("class", "grid")
      .call(d3.axisLeft(yScale)
        .tickSize(-chartWidth)
        .tickFormat("")
      )
      .style("stroke-dasharray", "3,3")
      .style("opacity", 0.3);

  }, [data]);

  return (
    <div className="advanced-d3-chart">
      <svg ref={svgRef} width={800} height={500}></svg>
    </div>
  );
}
```

// 大量データの最適化
function OptimizedChart({ rawData }) {
  const [processedData, setProcessedData] = useState([]);

  useEffect(() => {
    // データの前処理
    const optimized = useMemo(() => {
      if (rawData.length > 1000) {
        // 大量データの場合はサンプリング
        const step = Math.ceil(rawData.length / 100);
        return rawData.filter((_, index) => index % step === 0);
      }
      return rawData;
    }, [rawData]);

    setProcessedData(optimized);
  }, [rawData]);

  return (
    <Chart data={processedData} />
  );
}

// チャートの遅延読み込み
function LazyChart({ data, type }) {
  const [isVisible, setIsVisible] = useState(false);
  const chartRef = useRef(null);

  useEffect(() => {
    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          setIsVisible(true);
          observer.disconnect();
        }
      },
      { threshold: 0.1 }
    );

    if (chartRef.current) {
      observer.observe(chartRef.current);
    }

    return () => observer.disconnect();
  }, []);

  return (
    <div ref={chartRef} className="chart-container">
      {isVisible ? (
        <Chart data={data} type={type} />
      ) : (
        <div className="chart-placeholder">読み込み中...</div>
      )}
    </div>
  );
}

## まとめ

| ライブラリ | 適用場面 | 主なメリット | 主なデメリット |
|-----------|---------|-------------|---------------|
| **Chart.js** | シンプルなグラフ | 実装が簡単、軽量 | カスタマイズ性が限定的 |
| **Recharts** | Reactアプリケーション | React専用、宣言的 | バンドルサイズが大きい |
| **D3.js** | 高度なビジュアライゼーション | 最高の柔軟性、高性能 | 学習コストが高い |
| **ApexCharts** | 高品質なチャート | 美しいデザイン、豊富な機能 | 商用利用は有料 |

### 選択の指針

1. **シンプルなグラフ** → Chart.js
2. **Reactアプリケーション** → Recharts
3. **高度なカスタマイズ** → D3.js
4. **高品質なビジネスチャート** → ApexCharts
5. **大量データ** → D3.js + 仮想化
6. **モバイル対応** → Recharts または ApexCharts
