# レビュー・評価機能の実装方法とベストプラクティス

## 概要

レビュー・評価機能は、ユーザーが商品やサービスに対する意見や評価を共有し、他のユーザーの参考にできる重要な機能です。星評価、数値評価、テキストレビューなど、様々な評価手法とその実装方法を詳しく解説します。

## レビュー・評価機能の種類と比較表

| 項目 | 星評価 | 数値評価 | テキストレビュー | 総合評価システム |
|------|-------|----------|-----------------|-----------------|
| **実装の複雑さ** | ⭐ 簡単 | ⭐⭐ 中程度 | ⭐⭐ 中程度 | ⭐⭐⭐ 複雑 |
| **ユーザビリティ** | ⭐⭐⭐ 高い | ⭐⭐ 中程度 | ⭐⭐ 中程度 | ⭐⭐⭐ 高い |
| **データの価値** | ⭐⭐ 中程度 | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い |
| **分析の容易さ** | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い | ⭐⭐ 中程度 | ⭐⭐⭐ 高い |
| **スパム対策** | ⭐⭐ 中程度 | ⭐⭐ 中程度 | ⭐⭐ 中程度 | ⭐⭐⭐ 高い |
| **モバイル対応** | ⭐⭐⭐ 高い | ⭐⭐⭐ 高い | ⭐⭐ 中程度 | ⭐⭐⭐ 高い |

### 推奨使用場面

| 評価タイプ | 最適な使用場面 | 例 |
|-----------|----------------|-----|
| **星評価** | 直感的な評価 | ECサイト、レストラン、映画 |

## 1. 星評価機能

### 概要
直感的で分かりやすい星による評価システム

### データベース設計

```sql
-- レビューテーブル
CREATE TABLE reviews (
  id INT PRIMARY KEY AUTO_INCREMENT,
  user_id INT NOT NULL,
  item_id INT NOT NULL,
  item_type ENUM('product', 'service', 'restaurant', 'movie') NOT NULL,
  rating INT NOT NULL CHECK (rating >= 1 AND rating <= 5),
  title VARCHAR(200),
  content TEXT,
  is_verified BOOLEAN DEFAULT FALSE,
  helpful_count INT DEFAULT 0,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  
  FOREIGN KEY (user_id) REFERENCES users(id),
  INDEX idx_user_id (user_id),
  INDEX idx_item_id (item_id),
  INDEX idx_item_type (item_type),
  INDEX idx_rating (rating),
  UNIQUE KEY unique_user_item (user_id, item_id, item_type)
);

-- レビューの役立つ評価
CREATE TABLE review_helpful (
  id INT PRIMARY KEY AUTO_INCREMENT,
  review_id INT NOT NULL,
  user_id INT NOT NULL,
  is_helpful BOOLEAN NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  
  FOREIGN KEY (review_id) REFERENCES reviews(id) ON DELETE CASCADE,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  UNIQUE KEY unique_user_review (user_id, review_id),
  INDEX idx_review_id (review_id)
);
```

```javascript
// 星評価コンポーネント
function StarRating({ rating, onRatingChange, readOnly = false, size = 'medium' }) {
  const [hoverRating, setHoverRating] = useState(0);
  const [currentRating, setCurrentRating] = useState(rating || 0);

  const handleStarClick = (starRating) => {
    if (readOnly) return;
    
    setCurrentRating(starRating);
    onRatingChange?.(starRating);
  };

  const handleStarHover = (starRating) => {
    if (readOnly) return;
    setHoverRating(starRating);
  };

  const handleStarLeave = () => {
    if (readOnly) return;
    setHoverRating(0);
  };

  const getStarSize = () => {
    switch (size) {
      case 'small': return '16px';
      case 'large': return '32px';
      default: return '24px';
    }
  };

  return (
    <div className="star-rating">
      {[1, 2, 3, 4, 5].map((star) => {
        const isFilled = star <= (hoverRating || currentRating);
        return (
          <button
            key={star}
            type="button"
            className={`star ${isFilled ? 'filled' : 'empty'}`}
            onClick={() => handleStarClick(star)}
            onMouseEnter={() => handleStarHover(star)}
            onMouseLeave={handleStarLeave}
            disabled={readOnly}
            style={{ fontSize: getStarSize() }}
          >
            ★
          </button>
        );
      })}
      <span className="rating-text">
        {currentRating > 0 && `${currentRating}/5`}
      </span>
    </div>
  );
}

// レビュー投稿コンポーネント
function ReviewForm({ itemId, itemType, onReviewSubmit }) {
  const [rating, setRating] = useState(0);
  const [title, setTitle] = useState('');
  const [content, setContent] = useState('');
  const [isVerified, setIsVerified] = useState(false);
  const [user] = useAuth();

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (rating === 0) {
      alert('評価を選択してください');
      return;
    }

    try {
      const response = await fetch('/api/reviews', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${user.token}`
        },
        body: JSON.stringify({
          item_id: itemId,
          item_type: itemType,
          rating,
          title: title.trim(),
          content: content.trim(),
          is_verified: isVerified
        })
      });

      if (response.ok) {
        const newReview = await response.json();
        onReviewSubmit(newReview);
        
        // フォームをリセット
        setRating(0);
        setTitle('');
        setContent('');
        setIsVerified(false);
        
        alert('レビューを投稿しました');
      }
    } catch (error) {
      console.error('レビュー投稿エラー:', error);
      alert('レビューの投稿に失敗しました');
    }
  };

  return (
    <form onSubmit={handleSubmit} className="review-form">
      <h3>レビューを投稿</h3>
      
      <div className="rating-section">
        <label>評価 *</label>
        <StarRating 
          rating={rating} 
          onRatingChange={setRating}
          size="large"
        />
      </div>

      <div className="title-section">
        <label htmlFor="review-title">タイトル</label>
        <input
          id="review-title"
          type="text"
          value={title}
          onChange={(e) => setTitle(e.target.value)}
          placeholder="レビューのタイトルを入力..."
          maxLength={200}
        />
      </div>

      <div className="content-section">
        <label htmlFor="review-content">レビュー内容</label>
        <textarea
          id="review-content"
          value={content}
          onChange={(e) => setContent(e.target.value)}
          placeholder="商品やサービスの詳細な感想を入力..."
          rows={5}
          maxLength={2000}
        />
        <span className="char-count">{content.length}/2000</span>
      </div>

      <div className="verification-section">
        <label>
          <input
            type="checkbox"
            checked={isVerified}
            onChange={(e) => setIsVerified(e.target.checked)}
          />
          実際に購入・利用した商品・サービスです
        </label>
      </div>

      <button type="submit" disabled={rating === 0}>
        レビューを投稿
      </button>
    </form>
  );
}

// レビュー一覧コンポーネント
function ReviewList({ itemId, itemType }) {
  const [reviews, setReviews] = useState([]);
  const [averageRating, setAverageRating] = useState(0);
  const [ratingDistribution, setRatingDistribution] = useState({});
  const [loading, setLoading] = useState(true);
  const [sortBy, setSortBy] = useState('newest');

  const fetchReviews = useCallback(async () => {
    try {
      const response = await fetch(
        `/api/reviews?item_id=${itemId}&item_type=${itemType}&sort=${sortBy}`
      );
      const data = await response.json();
      
      setReviews(data.reviews);
      setAverageRating(data.averageRating);
      setRatingDistribution(data.ratingDistribution);
    } catch (error) {
      console.error('レビュー取得エラー:', error);
    } finally {
      setLoading(false);
    }
  }, [itemId, itemType, sortBy]);

  useEffect(() => {
    fetchReviews();
  }, [fetchReviews]);

  if (loading) return <div>読み込み中...</div>;

  return (
    <div className="review-list">
      <div className="review-summary">
        <div className="average-rating">
          <span className="rating-number">{averageRating.toFixed(1)}</span>
          <StarRating rating={Math.round(averageRating)} readOnly size="large" />
          <span className="review-count">({reviews.length}件のレビュー)</span>
        </div>

        <div className="rating-distribution">
          {[5, 4, 3, 2, 1].map((star) => (
            <div key={star} className="rating-bar">
              <span className="star-label">{star}★</span>
              <div className="bar-container">
                <div 
                  className="bar-fill"
                  style={{ 
                    width: `${(ratingDistribution[star] || 0) / reviews.length * 100}%` 
                  }}
                />
              </div>
              <span className="count">({ratingDistribution[star] || 0})</span>
            </div>
          ))}
        </div>
      </div>

      <div className="review-controls">
        <select value={sortBy} onChange={(e) => setSortBy(e.target.value)}>
          <option value="newest">新しい順</option>
          <option value="oldest">古い順</option>
          <option value="highest">評価の高い順</option>
          <option value="lowest">評価の低い順</option>
          <option value="most_helpful">役立った順</option>
        </select>
      </div>

      <div className="reviews">
        {reviews.map(review => (
          <ReviewItem 
            key={review.id} 
            review={review}
            onHelpfulChange={fetchReviews}
          />
        ))}
      </div>
    </div>
  );
}

// 個別レビューアイテム
function ReviewItem({ review, onHelpfulChange }) {
  const [user] = useAuth();
  const [helpfulStatus, setHelpfulStatus] = useState(null);

  const handleHelpful = async (isHelpful) => {
    if (!user) {
      alert('ログインが必要です');
      return;
    }

    try {
      const response = await fetch(`/api/reviews/${review.id}/helpful`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${user.token}`
        },
        body: JSON.stringify({ is_helpful: isHelpful })
      });

      if (response.ok) {
        setHelpfulStatus(isHelpful);
        onHelpfulChange();
      }
    } catch (error) {
      console.error('役立つ評価エラー:', error);
    }
  };

  return (
    <div className="review-item">
      <div className="review-header">
        <div className="reviewer-info">
          <img 
            src={review.user.avatar || '/default-avatar.png'} 
            alt={review.user.name}
            className="reviewer-avatar"
          />
          <div className="reviewer-details">
            <span className="reviewer-name">{review.user.name}</span>
            <div className="review-meta">
              <StarRating rating={review.rating} readOnly size="small" />
              <span className="review-date">
                {new Date(review.created_at).toLocaleDateString('ja-JP')}
              </span>
              {review.is_verified && (
                <span className="verified-badge">購入済み</span>
              )}
            </div>
          </div>
        </div>
      </div>

      <div className="review-content">
        {review.title && (
          <h4 className="review-title">{review.title}</h4>
        )}
        <p className="review-text">{review.content}</p>
      </div>

      <div className="review-actions">
        <div className="helpful-section">
          <span>このレビューは役に立ちましたか？</span>
          <button 
            onClick={() => handleHelpful(true)}
            className={`helpful-button ${helpfulStatus === true ? 'active' : ''}`}
          >
            はい ({review.helpful_count})
          </button>
          <button 
            onClick={() => handleHelpful(false)}
            className={`helpful-button ${helpfulStatus === false ? 'active' : ''}`}
          >
            いいえ
          </button>
        </div>
      </div>
    </div>
  );
}
```

```javascript
// 数値評価コンポーネント
function NumericRating({ 
  rating, 
  onRatingChange, 
  readOnly = false, 
  min = 1, 
  max = 10,
  step = 1 
}) {
  const [currentRating, setCurrentRating] = useState(rating || min);

  const handleRatingChange = (newRating) => {
    if (readOnly) return;
    
    setCurrentRating(newRating);
    onRatingChange?.(newRating);
  };

  return (
    <div className="numeric-rating">
      <div className="rating-input">
        <input
          type="range"
          min={min}
          max={max}
          step={step}
          value={currentRating}
          onChange={(e) => handleRatingChange(parseInt(e.target.value))}
          disabled={readOnly}
          className="rating-slider"
        />
        <div className="rating-labels">
          <span>{min}</span>
          <span>{max}</span>
        </div>
      </div>
      
      <div className="rating-display">
        <span className="rating-value">{currentRating}</span>
        <span className="rating-max">/{max}</span>
      </div>
    </div>
  );
}

// 複数項目評価コンポーネント
function MultiCriteriaRating({ criteria, onRatingChange, readOnly = false }) {
  const [ratings, setRatings] = useState({});

  const handleCriteriaChange = (criterionId, rating) => {
    if (readOnly) return;
    
    const newRatings = { ...ratings, [criterionId]: rating };
    setRatings(newRatings);
    onRatingChange?.(newRatings);
  };

  return (
    <div className="multi-criteria-rating">
      {criteria.map(criterion => (
        <div key={criterion.id} className="criterion-rating">
          <label className="criterion-label">
            {criterion.name}
            {criterion.description && (
              <span className="criterion-description">({criterion.description})</span>
            )}
          </label>
          <NumericRating
            rating={ratings[criterion.id]}
            onRatingChange={(rating) => handleCriteriaChange(criterion.id, rating)}
            readOnly={readOnly}
            min={criterion.min || 1}
            max={criterion.max || 10}
          />
        </div>
      ))}
    </div>
  );
}
```

```javascript
// テキストレビューコンポーネント
function TextReview({ review, onReviewSubmit, readOnly = false }) {
  const [content, setContent] = useState(review?.content || '');
  const [title, setTitle] = useState(review?.title || '');
  const [tags, setTags] = useState(review?.tags || []);
  const [user] = useAuth();

  const handleSubmit = async (e) => {
    e.preventDefault();
    if (!content.trim()) {
      alert('レビュー内容を入力してください');
      return;
    }

    try {
      const response = await fetch('/api/reviews', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${user.token}`
        },
        body: JSON.stringify({
          title: title.trim(),
          content: content.trim(),
          tags: tags,
          item_id: review.item_id,
          item_type: review.item_type
        })
      });

      if (response.ok) {
        const newReview = await response.json();
        onReviewSubmit(newReview);
        alert('レビューを投稿しました');
      }
    } catch (error) {
      console.error('レビュー投稿エラー:', error);
    }
  };

  const addTag = (tag) => {
    if (!tags.includes(tag) && tags.length < 5) {
      setTags([...tags, tag]);
    }
  };

  const removeTag = (tagToRemove) => {
    setTags(tags.filter(tag => tag !== tagToRemove));
  };

  const predefinedTags = [
    '高品質', 'コスパが良い', '使いやすい', 'デザインが良い', 
    '軽い', '重い', '大きい', '小さい', '速い', '遅い'
  ];

  if (readOnly) {
    return (
      <div className="text-review-readonly">
        {review.title && <h4>{review.title}</h4>}
        <p className="review-content">{review.content}</p>
        {review.tags && review.tags.length > 0 && (
          <div className="review-tags">
            {review.tags.map(tag => (
              <span key={tag} className="tag">{tag}</span>
            ))}
          </div>
        )}
      </div>
    );
  }

  return (
    <form onSubmit={handleSubmit} className="text-review-form">
      <h3>詳細レビューを投稿</h3>
      
      <div className="title-section">
        <label htmlFor="review-title">タイトル</label>
        <input
          id="review-title"
          type="text"
          value={title}
          onChange={(e) => setTitle(e.target.value)}
          placeholder="レビューのタイトルを入力..."
          maxLength={200}
        />
      </div>

      <div className="content-section">
        <label htmlFor="review-content">レビュー内容 *</label>
        <textarea
          id="review-content"
          value={content}
          onChange={(e) => setContent(e.target.value)}
          placeholder="商品やサービスの詳細な感想、使用感、おすすめポイントなどを入力..."
          rows={8}
          maxLength={5000}
        />
        <div className="char-count">{content.length}/5000</div>
      </div>

      <div className="tags-section">
        <label>タグ（最大5個）</label>
        <div className="tag-input">
          <input
            type="text"
            placeholder="タグを入力..."
            onKeyPress={(e) => {
              if (e.key === 'Enter') {
                e.preventDefault();
                const tag = e.target.value.trim();
                if (tag) {
                  addTag(tag);
                  e.target.value = '';
                }
              }
            }}
          />
        </div>
        
        <div className="predefined-tags">
          {predefinedTags.map(tag => (
            <button
              key={tag}
              type="button"
              onClick={() => addTag(tag)}
              disabled={tags.includes(tag) || tags.length >= 5}
              className="tag-button"
            >
              {tag}
            </button>
          ))}
        </div>

        <div className="selected-tags">
          {tags.map(tag => (
            <span key={tag} className="selected-tag">
              {tag}
              <button
                type="button"
                onClick={() => removeTag(tag)}
                className="remove-tag"
              >
                ×
              </button>
            </span>
          ))}
        </div>
      </div>

      <button type="submit" disabled={!content.trim()}>
        レビューを投稿
      </button>
    </form>
  );
```

### 1. 評価のキャッシュ

```javascript
// 評価データのキャッシュ
class RatingCache {
  constructor() {
    this.cache = new Map();
    this.ttl = 5 * 60 * 1000; // 5分
  }

  get(key) {
    const item = this.cache.get(key);
    if (item && Date.now() - item.timestamp < this.ttl) {
      return item.data;
    }
    return null;
  }

  set(key, data) {
    this.cache.set(key, {
      data,
      timestamp: Date.now()
    });
  }

  clear(key) {
    if (key) {
      this.cache.delete(key);
    } else {
      this.cache.clear();
    }
  }
}

const ratingCache = new RatingCache();
```

```javascript
// 評価の集計処理
async function calculateRatingStats(itemId, itemType) {
  const cacheKey = `rating_stats_${itemId}_${itemType}`;
  const cached = ratingCache.get(cacheKey);
  
  if (cached) {
    return cached;
  }

  try {
    const response = await fetch(`/api/ratings/stats?item_id=${itemId}&item_type=${itemType}`);
    const stats = await response.json();
    
    ratingCache.set(cacheKey, stats);
    return stats;
  } catch (error) {
    console.error('評価統計取得エラー:', error);
    return null;
  }
}
```

```javascript
// 評価のリアルタイム更新
function useRealtimeRatings(itemId, itemType) {
  const [ratings, setRatings] = useState([]);
  const [ws, setWs] = useState(null);

  useEffect(() => {
    const websocket = new WebSocket(`ws://localhost:8080/ratings/${itemId}`);
    
    websocket.onmessage = (event) => {
      const data = JSON.parse(event.data);
      
      if (data.type === 'new_rating') {
        setRatings(prev => [data.rating, ...prev]);
        ratingCache.clear(`rating_stats_${itemId}_${itemType}`);
      } else if (data.type === 'rating_updated') {
        setRatings(prev => prev.map(rating => 
          rating.id === data.rating.id ? data.rating : rating
        ));
        ratingCache.clear(`rating_stats_${itemId}_${itemType}`);
      }
    };

    setWs(websocket);

    return () => {
      websocket.close();
    };
  }, [itemId, itemType]);

  return ratings;
}
```
