## 1. WebSocket

### 概要
双方向のリアルタイム通信を実現するプロトコル

### 実装例

```javascript
// WebSocketの基本的な実装
class WebSocketManager {
  constructor(url, options = {}) {
    this.url = url;
    this.options = {
      reconnectInterval: 5000,
      maxReconnectAttempts: 5,
      heartbeatInterval: 30000,
      ...options
    };
    
    this.ws = null;
    this.reconnectAttempts = 0;
    this.heartbeatTimer = null;
    this.listeners = new Map();
    this.isConnecting = false;
  }

  connect() {
    if (this.isConnecting || (this.ws && this.ws.readyState === WebSocket.OPEN)) {
      return;
    }

    this.isConnecting = true;
    
    try {
      this.ws = new WebSocket(this.url);
      
      this.ws.onopen = (event) => {
        console.log('WebSocket接続が確立されました');
        this.isConnecting = false;
        this.reconnectAttempts = 0;
        this.startHeartbeat();
        this.emit('open', event);
      };
      
      this.ws.onmessage = (event) => {
        try {
          const data = JSON.parse(event.data);
          
          // ハートビートメッセージの処理
          if (data.type === 'heartbeat') {
            return;
          }
          
          this.emit('message', data);
          this.emit(data.type, data);
        } catch (error) {
          console.error('メッセージの解析エラー:', error);
          this.emit('error', error);
        }
      };
      
      this.ws.onclose = (event) => {
        console.log('WebSocket接続が閉じられました:', event.code, event.reason);
        this.isConnecting = false;
        this.stopHeartbeat();
        this.emit('close', event);
        
        // 自動再接続
        if (!event.wasClean && this.reconnectAttempts < this.options.maxReconnectAttempts) {
          this.scheduleReconnect();
        }
      };
      
      this.ws.onerror = (error) => {
        console.error('WebSocketエラー:', error);
        this.isConnecting = false;
        this.emit('error', error);
      };
      
    } catch (error) {
      console.error('WebSocket接続エラー:', error);
      this.isConnecting = false;
      this.emit('error', error);
    }
  }

  disconnect() {
    this.stopHeartbeat();
    if (this.ws) {
      this.ws.close(1000, '正常終了');
      this.ws = null;
    }
  }

  send(data) {
    if (this.ws && this.ws.readyState === WebSocket.OPEN) {
      const message = typeof data === 'string' ? data : JSON.stringify(data);
      this.ws.send(message);
      return true;
    } else {
      console.warn('WebSocketが接続されていません');
      return false;
    }
  }

  on(event, callback) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, []);
    }
    this.listeners.get(event).push(callback);
  }

  off(event, callback) {
    if (this.listeners.has(event)) {
      const callbacks = this.listeners.get(event);
      const index = callbacks.indexOf(callback);
      if (index > -1) {
        callbacks.splice(index, 1);
      }
    }
  }

  emit(event, data) {
    if (this.listeners.has(event)) {
      this.listeners.get(event).forEach(callback => {
        try {
          callback(data);
        } catch (error) {
          console.error('イベントリスナーエラー:', error);
        }
      });
    }
  }

  startHeartbeat() {
    this.heartbeatTimer = setInterval(() => {
      if (this.ws && this.ws.readyState === WebSocket.OPEN) {
        this.send({ type: 'heartbeat', timestamp: Date.now() });
      }
    }, this.options.heartbeatInterval);
  }

  stopHeartbeat() {
    if (this.heartbeatTimer) {
      clearInterval(this.heartbeatTimer);
      this.heartbeatTimer = null;
    }
  }

  scheduleReconnect() {
    this.reconnectAttempts++;
    const delay = this.options.reconnectInterval * Math.pow(2, this.reconnectAttempts - 1);
    
    console.log(`${delay}ms後に再接続を試行します (${this.reconnectAttempts}/${this.options.maxReconnectAttempts})`);
    
    setTimeout(() => {
      this.connect();
    }, delay);
  }

  getReadyState() {
    return this.ws ? this.ws.readyState : WebSocket.CLOSED;
  }

  isConnected() {
    return this.ws && this.ws.readyState === WebSocket.OPEN;
  }
}

// Reactでの使用例
function WebSocketComponent() {
  const [wsManager, setWsManager] = useState(null);
  const [messages, setMessages] = useState([]);
  const [connectionStatus, setConnectionStatus] = useState('disconnected');
  const [inputMessage, setInputMessage] = useState('');

  useEffect(() => {
    const ws = new WebSocketManager('ws://localhost:8080/chat');
    
    ws.on('open', () => {
      setConnectionStatus('connected');
    });
    
    ws.on('close', () => {
      setConnectionStatus('disconnected');
    });
    
    ws.on('error', (error) => {
      setConnectionStatus('error');
      console.error('WebSocketエラー:', error);
    });
    
    ws.on('message', (data) => {
      setMessages(prev => [...prev, data]);
    });
    
    ws.connect();
    setWsManager(ws);
    
    return () => {
      ws.disconnect();
    };
  }, []);

  const sendMessage = () => {
    if (wsManager && inputMessage.trim()) {
      wsManager.send({
        type: 'message',
        content: inputMessage,
        timestamp: Date.now()
      });
      setInputMessage('');
    }
  };

  return (
    <div className="websocket-component">
      <div className="connection-status">
        接続状態: <span className={connectionStatus}>{connectionStatus}</span>
      </div>
      
      <div className="messages">
        {messages.map((msg, index) => (
          <div key={index} className="message">
            <span className="timestamp">
              {new Date(msg.timestamp).toLocaleTimeString()}
            </span>
            <span className="content">{msg.content}</span>
          </div>
        ))}
      </div>
      
      <div className="input-area">
        <input
          type="text"
          value={inputMessage}
          onChange={(e) => setInputMessage(e.target.value)}
          onKeyPress={(e) => e.key === 'Enter' && sendMessage()}
          placeholder="メッセージを入力..."
        />
        <button onClick={sendMessage} disabled={!wsManager?.isConnected()}>
          送信
        </button>
      </div>
    </div>
  );
}
```

- ✅ **双方向通信**: クライアントとサーバー間の双方向通信
- ✅ **低遅延**: リアルタイム性が高い
- ✅ **効率的**: オーバーヘッドが少ない
- ✅ **柔軟性**: 様々な用途に適用可能

### デメリット
- ❌ **接続管理**: 接続の管理が複雑
- ❌ **プロキシ問題**: 一部のプロキシで問題が発生
- ❌ **スケーラビリティ**: 大量の接続でサーバー負荷が高い
- ❌ **ブラウザ制限**: 一部のブラウザで制限がある

### 使用場面
- 💬 **チャットアプリ**: リアルタイムメッセージング
- 🎮 **オンラインゲーム**: リアルタイムゲームプレイ

## 2. Server-Sent Events (SSE)

### 概要
サーバーからクライアントへの一方向リアルタイム通信

### 実装例

```javascript
// Server-Sent Eventsの実装
class SSEManager {
  constructor(url, options = {}) {
    this.url = url;
    this.options = {
      reconnectInterval: 3000,
      maxReconnectAttempts: 10,
      ...options
    };
    
    this.eventSource = null;
    this.reconnectAttempts = 0;
    this.listeners = new Map();
    this.isConnecting = false;
  }

  connect() {
    if (this.isConnecting || (this.eventSource && this.eventSource.readyState === EventSource.OPEN)) {
      return;
    }

    this.isConnecting = true;
    
    try {
      this.eventSource = new EventSource(this.url);
      
      this.eventSource.onopen = (event) => {
        console.log('SSE接続が確立されました');
        this.isConnecting = false;
        this.reconnectAttempts = 0;
        this.emit('open', event);
      };
      
      this.eventSource.onmessage = (event) => {
        try {
          const data = JSON.parse(event.data);
          this.emit('message', data);
        } catch (error) {
          console.error('メッセージの解析エラー:', error);
          this.emit('error', error);
        }
      };
      
      this.eventSource.onerror = (error) => {
        console.error('SSEエラー:', error);
        this.isConnecting = false;
        this.emit('error', error);
        
        // 自動再接続
        if (this.reconnectAttempts < this.options.maxReconnectAttempts) {
          this.scheduleReconnect();
        }
      };
      
      // カスタムイベントのリスナー
      this.eventSource.addEventListener('notification', (event) => {
        try {
          const data = JSON.parse(event.data);
          this.emit('notification', data);
        } catch (error) {
          console.error('通知データの解析エラー:', error);
        }
      });
      
      this.eventSource.addEventListener('update', (event) => {
        try {
          const data = JSON.parse(event.data);
          this.emit('update', data);
        } catch (error) {
          console.error('更新データの解析エラー:', error);
        }
      });
      
    } catch (error) {
      console.error('SSE接続エラー:', error);
      this.isConnecting = false;
      this.emit('error', error);
    }
  }

  disconnect() {
    if (this.eventSource) {
      this.eventSource.close();
      this.eventSource = null;
    }
  }

  on(event, callback) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, []);
    }
    this.listeners.get(event).push(callback);
  }

  off(event, callback) {
    if (this.listeners.has(event)) {
      const callbacks = this.listeners.get(event);
      const index = callbacks.indexOf(callback);
      if (index > -1) {
        callbacks.splice(index, 1);
      }
    }
  }

  emit(event, data) {
    if (this.listeners.has(event)) {
      this.listeners.get(event).forEach(callback => {
        try {
          callback(data);
        } catch (error) {
          console.error('イベントリスナーエラー:', error);
        }
      });
    }
  }

  scheduleReconnect() {
    this.reconnectAttempts++;
    const delay = this.options.reconnectInterval * Math.pow(2, this.reconnectAttempts - 1);
    
    console.log(`${delay}ms後に再接続を試行します (${this.reconnectAttempts}/${this.options.maxReconnectAttempts})`);
    
    setTimeout(() => {
      this.connect();
    }, delay);
  }

  getReadyState() {
    return this.eventSource ? this.eventSource.readyState : EventSource.CLOSED;
  }

  isConnected() {
    return this.eventSource && this.eventSource.readyState === EventSource.OPEN;
  }
}

// Reactでの使用例
function SSEDashboard() {
  const [sseManager, setSseManager] = useState(null);
```

## 3. ポーリング

### 概要
定期的にサーバーにリクエストを送信してデータを取得する手法

### 実装例

```javascript
// ポーリングの実装
class PollingManager {
  constructor(fetchFunction, options = {}) {
    this.fetchFunction = fetchFunction;
    this.options = {
      interval: 5000,
      maxRetries: 3,
      retryDelay: 1000,
      exponentialBackoff: true,
      ...options
    };
    
    this.intervalId = null;
    this.isPolling = false;
    this.retryCount = 0;
    this.listeners = new Map();
  }

  start() {
    if (this.isPolling) {
      return;
    }

    this.isPolling = true;
    this.retryCount = 0;
    this.poll();
  }

  stop() {
    this.isPolling = false;
    if (this.intervalId) {
      clearTimeout(this.intervalId);
      this.intervalId = null;
    }
  }

  async poll() {
    if (!this.isPolling) {
      return;
    }

    try {
      const data = await this.fetchFunction();
      this.retryCount = 0;
      this.emit('data', data);
      this.emit('success', data);
    } catch (error) {
      console.error('ポーリングエラー:', error);
      this.emit('error', error);
      
      if (this.retryCount < this.options.maxRetries) {
        this.retryCount++;
        const delay = this.options.exponentialBackoff 
          ? this.options.retryDelay * Math.pow(2, this.retryCount - 1)
          : this.options.retryDelay;
        
        console.log(`${delay}ms後にリトライします (${this.retryCount}/${this.options.maxRetries})`);
        
        this.intervalId = setTimeout(() => {
          this.poll();
        }, delay);
        return;
      } else {
        this.emit('maxRetriesReached', error);
      }
    }

    // 次のポーリングをスケジュール
    this.intervalId = setTimeout(() => {
      this.poll();
    }, this.options.interval);
  }

  on(event, callback) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, []);
    }
    this.listeners.get(event).push(callback);
  }

  off(event, callback) {
    if (this.listeners.has(event)) {
      const callbacks = this.listeners.get(event);
      const index = callbacks.indexOf(callback);
      if (index > -1) {
        callbacks.splice(index, 1);
      }
    }
  }

  emit(event, data) {
    if (this.listeners.has(event)) {
      this.listeners.get(event).forEach(callback => {
        try {
          callback(data);
        } catch (error) {
          console.error('イベントリスナーエラー:', error);
        }
      });
    }
  }

  updateInterval(newInterval) {
    this.options.interval = newInterval;
  }

  isActive() {
    return this.isPolling;
  }
}

// Reactでの使用例
function PollingComponent() {
  const [pollingManager, setPollingManager] = useState(null);
  const [data, setData] = useState(null);
  const [isPolling, setIsPolling] = useState(false);
  const [error, setError] = useState(null);

  const fetchData = async () => {
    const response = await fetch('/api/data');
    if (!response.ok) {
      throw new Error(`HTTP error! status: ${response.status}`);
    }
    return response.json();
  };

  useEffect(() => {
    const polling = new PollingManager(fetchData, {
      interval: 3000,
      maxRetries: 3
    });
    
    polling.on('data', (newData) => {
      setData(newData);
      setError(null);
    });
    
    polling.on('error', (err) => {
      setError(err.message);
    });
    
    polling.on('maxRetriesReached', (err) => {
      setError('最大リトライ回数に達しました');
      setIsPolling(false);
    });
    
    setPollingManager(polling);
    
    return () => {
      polling.stop();
    };
  }, []);

  const startPolling = () => {
    if (pollingManager) {
      pollingManager.start();
      setIsPolling(true);
    }
  };

  const stopPolling = () => {
    if (pollingManager) {
      pollingManager.stop();
      setIsPolling(false);
    }
  };

  return (
    <div className="polling-component">
      <div className="controls">
        <button onClick={startPolling} disabled={isPolling}>
          ポーリング開始
        </button>
        <button onClick={stopPolling} disabled={!isPolling}>
          ポーリング停止
        </button>
      </div>
      
      <div className="status">
        状態: {isPolling ? 'ポーリング中' : '停止中'}
      </div>
      
      {error && (
        <div className="error">
          エラー: {error}
        </div>
      )}
      
      {data && (
        <div className="data">
          <h3>取得データ</h3>
          <pre>{JSON.stringify(data, null, 2)}</pre>
        </div>
      )}
    </div>
  );
}
```

### デメリット
- ❌ **リアルタイム性が低い**: ポーリング間隔による遅延

- ❌ **ネットワーク使用量**: 不要なリクエストでネットワークを消費
- ❌ **バッテリー消費**: モバイルデバイスでバッテリーを消費

### 使用場面
- 📊 **ステータス確認**: 定期的なステータス確認
- 🔄 **同期処理**: データの同期

## 4. Push通知

### 概要
モバイルデバイスやブラウザにプッシュ通知を送信する機能

### 実装例

```javascript
// Push通知の実装
class PushNotificationManager {
  constructor() {
    this.registration = null;
    this.subscription = null;
    this.isSupported = 'serviceWorker' in navigator && 'PushManager' in window;
  }

  async initialize() {
    if (!this.isSupported) {
      throw new Error('Push通知はサポートされていません');
    }

    try {
      // Service Workerの登録
      this.registration = await navigator.serviceWorker.register('/sw.js');
      console.log('Service Workerが登録されました');
      
      // 通知の許可を要求
      const permission = await Notification.requestPermission();
      if (permission !== 'granted') {
        throw new Error('通知の許可が得られませんでした');
      }
      
      return true;
    } catch (error) {
      console.error('Push通知の初期化エラー:', error);
      throw error;
    }
  }

  async subscribe() {
    if (!this.registration) {
      throw new Error('Service Workerが登録されていません');
    }

    try {
      // VAPIDキーを取得
      const response = await fetch('/api/vapid-public-key');
      const vapidPublicKey = await response.text();
      
      // プッシュ購読
      this.subscription = await this.registration.pushManager.subscribe({
        userVisibleOnly: true,
        applicationServerKey: this.urlBase64ToUint8Array(vapidPublicKey)
      });
      
      // サーバーに購読情報を送信
      await this.sendSubscriptionToServer(this.subscription);
      
      console.log('Push通知の購読が完了しました');
      return this.subscription;
    } catch (error) {
      console.error('Push通知の購読エラー:', error);
      throw error;
    }
  }

  async unsubscribe() {
    if (!this.subscription) {
      return;
    }

    try {
      const success = await this.subscription.unsubscribe();
      if (success) {
        // サーバーから購読情報を削除
        await this.removeSubscriptionFromServer(this.subscription);
        this.subscription = null;
        console.log('Push通知の購読を解除しました');
      }
      return success;
    } catch (error) {
      console.error('Push通知の購読解除エラー:', error);
      throw error;
    }
  }

  async sendSubscriptionToServer(subscription) {
    const response = await fetch('/api/subscribe', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(subscription)
    });
    
    if (!response.ok) {
      throw new Error('サーバーへの購読情報送信に失敗しました');
    }
  }

  async removeSubscriptionFromServer(subscription) {
    const response = await fetch('/api/unsubscribe', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(subscription)
    });
    
    if (!response.ok) {
      throw new Error('サーバーからの購読情報削除に失敗しました');
    }
  }

  urlBase64ToUint8Array(base64String) {
    const padding = '='.repeat((4 - base64String.length % 4) % 4);
    const base64 = (base64String + padding)
      .replace(/-/g, '+')
      .replace(/_/g, '/');

    const rawData = window.atob(base64);
    const outputArray = new Uint8Array(rawData.length);

    for (let i = 0; i < rawData.length; ++i) {
      outputArray[i] = rawData.charCodeAt(i);
    }
    return outputArray;
  }

  isSubscribed() {
    return this.subscription !== null;
  }

  getSubscription() {
    return this.subscription;
  }
}

function PushNotificationComponent() {
  const [pushManager, setPushManager] = useState(null);
  const [isSubscribed, setIsSubscribed] = useState(false);
  const [isSupported, setIsSupported] = useState(false);
  const [error, setError] = useState(null);

  useEffect(() => {
    const manager = new PushNotificationManager();
    setPushManager(manager);
    setIsSupported(manager.isSupported);
    
    if (manager.isSupported) {
      manager.initialize()
        .then(() => {
          setIsSubscribed(manager.isSubscribed());
        })
        .catch(err => {
          setError(err.message);
        });
    }
  }, []);

  const subscribe = async () => {
    try {
      await pushManager.subscribe();
      setIsSubscribed(true);
      setError(null);
    } catch (err) {
      setError(err.message);
    }
  };

  const unsubscribe = async () => {
    try {
      await pushManager.unsubscribe();
      setIsSubscribed(false);
      setError(null);
    } catch (err) {
      setError(err.message);
    }
  };

  if (!isSupported) {
    return (
      <div className="push-notification">
        <p>このブラウザはPush通知をサポートしていません</p>
      </div>
    );
  }

  return (
    <div className="push-notification">
      <h3>Push通知設定</h3>
      
      {error && (
        <div className="error">
          エラー: {error}
        </div>
      )}
      
      <div className="status">
        状態: {isSubscribed ? '購読中' : '未購読'}
      </div>
      
      <div className="controls">
        {!isSubscribed ? (
          <button onClick={subscribe}>
            Push通知を有効にする
          </button>
        ) : (
          <button onClick={unsubscribe}>
            Push通知を無効にする

      </div>

  );
```

- ❌ **配信制限**: ブラウザやOSの配信制限がある

```javascript
class ConnectionPool {
  constructor(maxConnections = 10) {
    this.maxConnections = maxConnections;
    this.connections = new Map();
    this.availableConnections = [];
  }

  getConnection(key) {
    if (this.connections.has(key)) {
      return this.connections.get(key);
    }

    if (this.availableConnections.length > 0) {
      const connection = this.availableConnections.pop();
      this.connections.set(key, connection);
      return connection;
    }

    if (this.connections.size < this.maxConnections) {
      const connection = this.createConnection();
      this.connections.set(key, connection);
      return connection;
    }

    throw new Error('接続プールが満杯です');
  }

  releaseConnection(key) {
    const connection = this.connections.get(key);
    if (connection) {
      this.connections.delete(key);
      this.availableConnections.push(connection);
    }
  }
```

```javascript
class DataOptimizer {
  constructor() {
    this.lastData = new Map();
  }

  getDiff(newData, key) {
    const lastData = this.lastData.get(key);
    if (!lastData) {
      this.lastData.set(key, newData);
      return newData;
    }

    const diff = this.calculateDiff(lastData, newData);
    if (Object.keys(diff).length > 0) {
      this.lastData.set(key, newData);
      return diff;
    }

    return null; // 変更なし
  }

  calculateDiff(oldData, newData) {
    const diff = {};
    
    for (const key in newData) {
      if (oldData[key] !== newData[key]) {
        diff[key] = newData[key];
      }
    }
    
    return diff;
  }
}
```

```javascript
// メッセージのバッファリング
class MessageBuffer {
  constructor(maxSize = 100, flushInterval = 1000) {
    this.buffer = [];
    this.maxSize = maxSize;
    this.flushInterval = flushInterval;
    this.flushTimer = null;
    this.flushCallback = null;
  }

  add(message) {
    this.buffer.push(message);
    
    if (this.buffer.length >= this.maxSize) {
      this.flush();
    } else if (!this.flushTimer) {
      this.flushTimer = setTimeout(() => {
        this.flush();
      }, this.flushInterval);
    }
  }

  flush() {
    if (this.buffer.length > 0 && this.flushCallback) {
      this.flushCallback([...this.buffer]);
      this.buffer = [];
    }
    
    if (this.flushTimer) {
      clearTimeout(this.flushTimer);
      this.flushTimer = null;
    }
  }

  onFlush(callback) {
    this.flushCallback = callback;
  }
}
```

## まとめ

| 手法 | 適用場面 | 主なメリット | 主なデメリット |
|-----|---------|-------------|---------------|
| **WebSocket** | 双方向リアルタイム通信 | 低遅延、双方向通信 | 接続管理が複雑 |
| **Server-Sent Events** | 一方向リアルタイム更新 | 実装が簡単、自動再接続 | 一方向通信のみ |
| **ポーリング** | シンプルな定期更新 | 実装が簡単、信頼性 | リアルタイム性が低い |
| **Push通知** | モバイル通知 | 高い到達率、即座の配信 | 実装が複雑 |

### 選択の指針

1. **双方向リアルタイム通信** → WebSocket
2. **一方向リアルタイム更新** → Server-Sent Events
3. **シンプルな定期更新** → ポーリング
4. **モバイル通知** → Push通知
5. **複数の要件** → 組み合わせパターン
