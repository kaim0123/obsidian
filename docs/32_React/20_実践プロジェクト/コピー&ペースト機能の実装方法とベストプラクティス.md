## 概要

コピー&ペースト機能の実装には主に3つの方法があります。それぞれの特徴、メリット・デメリット、使用場面を詳しく解説します。

| 実装方法 | 最適な使用場面 | 例 |

## 1. Clipboard API

### 概要
モダンブラウザでサポートされている標準的なクリップボード操作API

### 実装例

```javascript
import { useState, useCallback } from 'react';

function ClipboardAPIExample() {
  const [copiedText, setCopiedText] = useState('');
  const [error, setError] = useState('');
  const [clipboardData, setClipboardData] = useState('');

  // テキストのコピー
  const copyText = useCallback(async (text) => {
    try {
      if (!navigator.clipboard) {
        throw new Error('Clipboard APIがサポートされていません');
      }

      await navigator.clipboard.writeText(text);
      setCopiedText(text);
      setError('');
      
      // 成功メッセージの表示
      setTimeout(() => setCopiedText(''), 2000);
      
    } catch (err) {
      setError(`コピーに失敗しました: ${err.message}`);
    }
  }, []);

  // テキストのペースト
  const pasteText = useCallback(async () => {
    try {
      if (!navigator.clipboard) {
        throw new Error('Clipboard APIがサポートされていません');
      }

      const text = await navigator.clipboard.readText();
      setClipboardData(text);
      setError('');
      
    } catch (err) {
      setError(`ペーストに失敗しました: ${err.message}`);
    }
  }, []);

  // 画像のコピー
  const copyImage = useCallback(async (imageBlob) => {
    try {
      if (!navigator.clipboard) {
        throw new Error('Clipboard APIがサポートされていません');
      }

      const clipboardItem = new ClipboardItem({
        'image/png': imageBlob
      });

      await navigator.clipboard.write([clipboardItem]);
      setError('');
      
    } catch (err) {
      setError(`画像のコピーに失敗しました: ${err.message}`);
    }
  }, []);

  // 画像のペースト
  const pasteImage = useCallback(async () => {
    try {
      if (!navigator.clipboard) {
        throw new Error('Clipboard APIがサポートされていません');
      }

      const clipboardItems = await navigator.clipboard.read();
      
      for (const clipboardItem of clipboardItems) {
        if (clipboardItem.types.includes('image/png')) {
          const imageBlob = await clipboardItem.getType('image/png');
          // 画像の処理
          const imageUrl = URL.createObjectURL(imageBlob);
          console.log('ペーストされた画像:', imageUrl);
        }
      }
      
      setError('');
      
    } catch (err) {
      setError(`画像のペーストに失敗しました: ${err.message}`);
    }
  }, []);

  // 複数データ形式のコピー
  const copyMultipleFormats = useCallback(async (text, html, imageBlob) => {
    try {
      if (!navigator.clipboard) {
        throw new Error('Clipboard APIがサポートされていません');
      }

      const clipboardItem = new ClipboardItem({
        'text/plain': new Blob([text], { type: 'text/plain' }),
        'text/html': new Blob([html], { type: 'text/html' }),
        'image/png': imageBlob
      });

      await navigator.clipboard.write([clipboardItem]);
      setError('');
      
    } catch (err) {
      setError(`複数形式のコピーに失敗しました: ${err.message}`);
    }
  }, []);

  return (
    <div style={{ padding: '20px' }}>
      <h2>Clipboard API Example</h2>

      {/* エラーメッセージ */}
      {error && (
        <div style={{ 
          padding: '10px', 
          backgroundColor: '#ffebee', 
          color: '#c62828',
          borderRadius: '4px',
          marginBottom: '20px'
        }}>
          {error}
        </div>
      )}

      {/* 成功メッセージ */}
      {copiedText && (
        <div style={{ 
          padding: '10px', 
          backgroundColor: '#e8f5e8', 
          color: '#2e7d32',
          borderRadius: '4px',
          marginBottom: '20px'
        }}>
          コピーしました: {copiedText}
        </div>
      )}

      {/* テキスト操作 */}
      <div style={{ marginBottom: '20px' }}>
        <h3>テキスト操作</h3>
        <button 
          onClick={() => copyText('Hello, World!')}
          style={{ 
            marginRight: '10px',
            padding: '8px 16px',
            backgroundColor: '#2196f3',
            color: 'white',
            border: 'none',
            borderRadius: '4px'
          }}
        >
          テキストをコピー
        </button>
        <button 
          onClick={pasteText}
          style={{ 
            padding: '8px 16px',
            backgroundColor: '#4caf50',
            color: 'white',
            border: 'none',
            borderRadius: '4px'
          }}
        >
          テキストをペースト
        </button>
        {clipboardData && (
          <div style={{ 
            marginTop: '10px',
            padding: '10px',
            backgroundColor: '#f5f5f5',
            borderRadius: '4px'
          }}>
            ペーストされたテキスト: {clipboardData}
          </div>
        )}
      </div>

      {/* 画像操作 */}
      <div style={{ marginBottom: '20px' }}>
        <h3>画像操作</h3>
        <button 
          onClick={() => {
            // 画像のコピー例（実際の実装では画像データが必要）
            console.log('画像コピー機能');
          }}
          style={{ 
            marginRight: '10px',
            padding: '8px 16px',
            backgroundColor: '#ff9800',
            color: 'white',
            border: 'none',
            borderRadius: '4px'
          }}
        >
          画像をコピー
        </button>
        <button 
          onClick={pasteImage}
          style={{ 
            padding: '8px 16px',
            backgroundColor: '#9c27b0',
            color: 'white',
            border: 'none',
            borderRadius: '4px'
          }}
        >
          画像をペースト
        </button>
      </div>

      {/* カスタムテキスト入力 */}
      <div style={{ marginBottom: '20px' }}>
        <h3>カスタムテキスト</h3>
        <textarea
          id="customText"
          placeholder="コピーしたいテキストを入力..."
          style={{
            width: '100%',
            height: '100px',
            padding: '10px',
            border: '1px solid #ddd',
            borderRadius: '4px',
            resize: 'vertical'
          }}
        />
        <br />
        <button 
          onClick={() => {
            const text = document.getElementById('customText').value;
            if (text) copyText(text);
          }}
          style={{ 
            marginTop: '10px',
            padding: '8px 16px',
            backgroundColor: '#607d8b',
            color: 'white',
            border: 'none',
            borderRadius: '4px'
          }}
        >
          入力したテキストをコピー
        </button>
      </div>
    </div>
  );
}
```

### メリット
- ✅ **セキュリティ**: セキュアなクリップボード操作
- ✅ **多様なデータ形式**: テキスト、画像、ファイルに対応
- ✅ **モダン対応**: 最新のWeb標準
- ✅ **アクセシビリティ**: スクリーンリーダー対応
- ✅ **非同期処理**: Promiseベースの処理

### デメリット
- ❌ **ブラウザ対応**: 古いブラウザでは未対応
- ❌ **HTTPS必須**: セキュアコンテキストが必要
- ❌ **実装の複雑さ**: エラーハンドリングが複雑
- ❌ **学習コスト**: APIの理解に時間がかかる

### 使用場面
- 📱 **モダンWebアプリ**: 最新ブラウザをターゲット
- 🖼️ **画像編集ツール**: 画像のコピー&ペースト
- 📊 **データ分析ツール**: 複雑なデータのコピー
- 🎨 **デザインツール**: デザイン要素のコピー

## 2. execCommand

### 概要
レガシーブラウザでも動作する従来のクリップボード操作

### 実装例

```javascript
function ExecCommandExample() {
  const [error, setError] = useState('');
  const [success, setSuccess] = useState('');

  // テキストのコピー
  const copyTextWithExec = (text) => {
    try {
      // 一時的なテキストエリアを作成
      const textArea = document.createElement('textarea');
      textArea.value = text;
      textArea.style.position = 'fixed';
      textArea.style.left = '-999999px';
      textArea.style.top = '-999999px';
      
      document.body.appendChild(textArea);
      textArea.focus();
      textArea.select();
      
      // execCommandでコピー
      const successful = document.execCommand('copy');
      document.body.removeChild(textArea);
      
      if (successful) {
        setSuccess('テキストをコピーしました');
        setError('');
        setTimeout(() => setSuccess(''), 2000);
      } else {
        throw new Error('コピーコマンドが失敗しました');
      }
      
    } catch (err) {
      setError(`コピーに失敗しました: ${err.message}`);
      setSuccess('');
    }
  };

  // 選択範囲のコピー
  const copySelection = () => {
    try {
      const selectedText = window.getSelection().toString();
      
      if (!selectedText) {
        setError('コピーするテキストが選択されていません');
        return;
      }

      const successful = document.execCommand('copy');
      
      if (successful) {
        setSuccess('選択したテキストをコピーしました');
        setError('');
        setTimeout(() => setSuccess(''), 2000);
      } else {
        throw new Error('コピーコマンドが失敗しました');
      }
      
    } catch (err) {
      setError(`コピーに失敗しました: ${err.message}`);
      setSuccess('');
    }
  };

  // ペースト（制限あり）
  const pasteText = () => {
    try {
      // execCommandのpasteは制限が多いため、
      // 通常はClipboard APIを使用するか、
      // フォーカスされた要素に直接ペースト
      const textArea = document.createElement('textarea');
      textArea.style.position = 'fixed';
      textArea.style.left = '-999999px';
      textArea.style.top = '-999999px';
      
      document.body.appendChild(textArea);
      textArea.focus();
      
      // ペーストイベントを監視
      textArea.addEventListener('paste', (e) => {
        const pastedText = e.clipboardData.getData('text');
        console.log('ペーストされたテキスト:', pastedText);
        setSuccess(`ペーストしました: ${pastedText}`);
        setTimeout(() => setSuccess(''), 2000);
      });
      
      // ペーストコマンドを実行
      const successful = document.execCommand('paste');
      
      if (!successful) {
        throw new Error('ペーストコマンドが失敗しました');
      }
      
      // クリーンアップ
      setTimeout(() => {
        document.body.removeChild(textArea);
      }, 100);
      
    } catch (err) {
      setError(`ペーストに失敗しました: ${err.message}`);
      setSuccess('');
    }
  };

  // カット操作
  const cutText = () => {
    try {
      const textArea = document.getElementById('cutTextArea');
      if (!textArea) {
        setError('テキストエリアが見つかりません');
        return;
      }

      textArea.focus();
      textArea.select();
      
      const successful = document.execCommand('cut');
      
      if (successful) {
        setSuccess('テキストをカットしました');
        setError('');
        setTimeout(() => setSuccess(''), 2000);
      } else {
        throw new Error('カットコマンドが失敗しました');
      }
      
    } catch (err) {
      setError(`カットに失敗しました: ${err.message}`);
      setSuccess('');
    }
  };

  return (
    <div style={{ padding: '20px' }}>
      <h2>execCommand Example</h2>

      {/* エラーメッセージ */}
      {error && (
        <div style={{ 
          padding: '10px', 
          backgroundColor: '#ffebee', 
          color: '#c62828',
          borderRadius: '4px',
          marginBottom: '20px'
        }}>
          {error}
        </div>
      )}

      {/* 成功メッセージ */}
      {success && (
        <div style={{ 
          padding: '10px', 
          backgroundColor: '#e8f5e8', 
          color: '#2e7d32',
          borderRadius: '4px',
          marginBottom: '20px'
        }}>
          {success}
        </div>
      )}

      {/* テキスト操作 */}
      <div style={{ marginBottom: '20px' }}>
        <h3>テキスト操作</h3>
        <button 
          onClick={() => copyTextWithExec('Hello, World!')}
          style={{ 
            marginRight: '10px',
            padding: '8px 16px',
            backgroundColor: '#2196f3',
            color: 'white',
            border: 'none',
            borderRadius: '4px'
          }}
        >
          テキストをコピー
        </button>
        <button 
          onClick={copySelection}
          style={{ 
            marginRight: '10px',
            padding: '8px 16px',
            backgroundColor: '#4caf50',
            color: 'white',
            border: 'none',
            borderRadius: '4px'
          }}
        >
          選択範囲をコピー
        </button>
        <button 
          onClick={pasteText}
          style={{ 
            marginRight: '10px',
            padding: '8px 16px',
            backgroundColor: '#ff9800',
            color: 'white',
            border: 'none',
            borderRadius: '4px'
          }}
        >
          ペースト
        </button>
        <button 
          onClick={cutText}
          style={{ 
            padding: '8px 16px',
            backgroundColor: '#f44336',
            color: 'white',
            border: 'none',
            borderRadius: '4px'
          }}
        >
          カット
        </button>
      </div>

      {/* テキストエリア */}
      <div style={{ marginBottom: '20px' }}>
        <h3>テキストエリア</h3>
        <textarea
          id="cutTextArea"
          placeholder="ここにテキストを入力してカット操作を試してください..."
          style={{
            width: '100%',
            height: '100px',
            padding: '10px',
            border: '1px solid #ddd',
            borderRadius: '4px',
            resize: 'vertical'
          }}
        />
      </div>

      {/* 選択可能なテキスト */}
      <div style={{ marginBottom: '20px' }}>
        <h3>選択可能なテキスト</h3>
        <p style={{ 
          padding: '10px',
          backgroundColor: '#f5f5f5',
          borderRadius: '4px',
          userSelect: 'text'
        }}>
          このテキストを選択してから「選択範囲をコピー」ボタンを押してください。
          このテキストは選択可能です。
        </p>
      </div>
    </div>
  );
}
```

### メリット
- ✅ **ブラウザ対応**: 古いブラウザでも動作
- ✅ **実装が簡単**: 基本的なコマンドで実装可能
- ✅ **即座に利用可能**: 特別な設定が不要

### 使用場面

- 📝 **シンプルなコピー**: 基本的なテキストコピー
- 🔧 **プロトタイプ**: 簡単な実装でテスト

## 3. カスタム実装

### 概要
独自のロジックでクリップボード機能を実装する方法

### 実装例

```javascript
function CustomCopyPasteExample() {
  const [error, setError] = useState('');
  const [success, setSuccess] = useState('');
  const [customData, setCustomData] = useState('');

  // カスタムデータのコピー
  const copyCustomData = (data) => {
    try {
      // データをJSON形式でシリアライズ
      const serializedData = JSON.stringify(data);
      
      // カスタムイベントを発火
      const customEvent = new CustomEvent('customCopy', {
        detail: {
          data: serializedData,
          timestamp: Date.now(),
          type: 'custom'
        }
      });
      
      document.dispatchEvent(customEvent);
      
      // ローカルストレージにも保存（バックアップ）
      localStorage.setItem('customClipboard', serializedData);
      
      setSuccess('カスタムデータをコピーしました');
      setError('');
      setTimeout(() => setSuccess(''), 2000);
      
    } catch (err) {
      setError(`カスタムコピーに失敗しました: ${err.message}`);
    }
  };

  // カスタムデータのペースト
  const pasteCustomData = () => {
    try {
      // カスタムイベントを監視
      const handleCustomPaste = (event) => {
        const { data } = event.detail;
        const parsedData = JSON.parse(data);
        setCustomData(JSON.stringify(parsedData, null, 2));
        setSuccess('カスタムデータをペーストしました');
        setTimeout(() => setSuccess(''), 2000);
      };
      
      document.addEventListener('customPaste', handleCustomPaste);
      
      // ローカルストレージから復元
      const storedData = localStorage.getItem('customClipboard');
      if (storedData) {
        const parsedData = JSON.parse(storedData);
        setCustomData(JSON.stringify(parsedData, null, 2));
        setSuccess('ローカルストレージからデータを復元しました');
        setTimeout(() => setSuccess(''), 2000);
      }
      
      // イベントリスナーをクリーンアップ
      setTimeout(() => {
        document.removeEventListener('customPaste', handleCustomPaste);
      }, 1000);
      
    } catch (err) {
      setError(`カスタムペーストに失敗しました: ${err.message}`);
    }
  };

  // キーボードショートカット
  useEffect(() => {
    const handleKeyDown = (event) => {
      // Ctrl+C (コピー)
      if (event.ctrlKey && event.key === 'c') {
        event.preventDefault();
        copyCustomData({
          text: 'カスタムコピーされたテキスト',
          metadata: {
            source: 'keyboard',
            timestamp: Date.now()
          }
        });
      }
      
      // Ctrl+V (ペースト)
      if (event.ctrlKey && event.key === 'v') {
        event.preventDefault();
        pasteCustomData();
      }
    };

    document.addEventListener('keydown', handleKeyDown);
    
    return () => {
      document.removeEventListener('keydown', handleKeyDown);
    };
  }, []);

  // ドラッグ&ドロップでのコピー
  const handleDragStart = (event, data) => {
    event.dataTransfer.setData('application/json', JSON.stringify(data));
    event.dataTransfer.effectAllowed = 'copy';
  };

  const handleDrop = (event) => {
    event.preventDefault();
    const data = event.dataTransfer.getData('application/json');
    if (data) {
      const parsedData = JSON.parse(data);
      setCustomData(JSON.stringify(parsedData, null, 2));
      setSuccess('ドラッグ&ドロップでデータをコピーしました');
      setTimeout(() => setSuccess(''), 2000);
    }
  };

  const handleDragOver = (event) => {
    event.preventDefault();
    event.dataTransfer.dropEffect = 'copy';
  };

  return (
    <div style={{ padding: '20px' }}>
      <h2>カスタム実装 Example</h2>

      {/* エラーメッセージ */}
      {error && (
        <div style={{ 
          padding: '10px', 
          backgroundColor: '#ffebee', 
          color: '#c62828',
          borderRadius: '4px',
          marginBottom: '20px'
        }}>
          {error}
        </div>
      )}

      {/* 成功メッセージ */}
      {success && (
        <div style={{ 
          padding: '10px', 
          backgroundColor: '#e8f5e8', 
          color: '#2e7d32',
          borderRadius: '4px',
          marginBottom: '20px'
        }}>
          {success}
        </div>
      )}

      {/* 操作ボタン */}
      <div style={{ marginBottom: '20px' }}>
        <h3>カスタム操作</h3>
        <button 
          onClick={() => copyCustomData({
            message: 'Hello, Custom World!',
            timestamp: Date.now(),
            metadata: { source: 'button' }
          })}
          style={{ 
            marginRight: '10px',
            padding: '8px 16px',
            backgroundColor: '#2196f3',
            color: 'white',
            border: 'none',
            borderRadius: '4px'
          }}
        >
          カスタムデータをコピー
        </button>
        <button 
          onClick={pasteCustomData}
          style={{ 
            padding: '8px 16px',
            backgroundColor: '#4caf50',
            color: 'white',
            border: 'none',
            borderRadius: '4px'
          }}
        >
          カスタムデータをペースト
        </button>
      </div>

      {/* キーボードショートカット説明 */}
      <div style={{ marginBottom: '20px' }}>
        <h3>キーボードショートカット</h3>
        <p>
          <strong>Ctrl+C:</strong> カスタムデータをコピー<br />
          <strong>Ctrl+V:</strong> カスタムデータをペースト
        </p>
      </div>

      {/* ドラッグ&ドロップエリア */}
      <div style={{ marginBottom: '20px' }}>
        <h3>ドラッグ&ドロップ</h3>
        <div
          onDrop={handleDrop}
          onDragOver={handleDragOver}
          style={{
            border: '2px dashed #ccc',
            borderRadius: '8px',
            padding: '40px',
            textAlign: 'center',
            backgroundColor: '#f9f9f9'
          }}
        >
          ここにドラッグ&ドロップでデータをコピー
        </div>
      </div>

      {/* ドラッグ可能なアイテム */}
      <div style={{ marginBottom: '20px' }}>
        <h3>ドラッグ可能なアイテム</h3>
        <div
          draggable
          onDragStart={(e) => handleDragStart(e, {
            message: 'ドラッグされたデータ',
            timestamp: Date.now()
          })}
          style={{
            padding: '10px',
            backgroundColor: '#e3f2fd',
            borderRadius: '4px',
            cursor: 'move',
            userSelect: 'none'
          }}
        >
          このアイテムをドラッグしてください
        </div>
      </div>

      {/* ペーストされたデータ表示 */}
      {customData && (
        <div style={{ marginBottom: '20px' }}>
          <h3>ペーストされたデータ</h3>
          <pre style={{
            backgroundColor: '#f5f5f5',
            padding: '10px',
            borderRadius: '4px',
            overflow: 'auto',
            maxHeight: '200px'
          }}>
            {customData}
          </pre>
        </div>
      )}
    </div>
  );
}
```

## 組み合わせパターン

実際のアプリケーションでは、複数の方法を組み合わせて使用することが多いです。

### パターン1: フォールバック実装

```javascript
function FallbackCopyPaste() {
  const copyText = async (text) => {
    try {
      // モダンブラウザ: Clipboard API
      if (navigator.clipboard) {
        await navigator.clipboard.writeText(text);
      } else {
        // レガシーブラウザ: execCommand
        const textArea = document.createElement('textarea');
        textArea.value = text;
        document.body.appendChild(textArea);
        textArea.select();
        document.execCommand('copy');
        document.body.removeChild(textArea);
      }
    } catch (error) {
      // カスタム実装にフォールバック
      localStorage.setItem('fallbackClipboard', text);
    }
  };
}
```

### パターン2: 用途別の使い分け

```javascript
function MultiMethodCopyPaste() {
  // テキスト: Clipboard API
  const copyText = async (text) => {
    await navigator.clipboard.writeText(text);
  };

  // 画像: Clipboard API
  const copyImage = async (imageBlob) => {
    await navigator.clipboard.write([new ClipboardItem({
      'image/png': imageBlob
    })]);
  };

  // 複雑なデータ: カスタム実装
  const copyComplexData = (data) => {
    localStorage.setItem('complexData', JSON.stringify(data));
  };
}
```

## まとめ

| 実装方法 | 適用場面 | 主なメリット | 主なデメリット |
|---------|---------|-------------|---------------|
| **Clipboard API** | モダンブラウザ | セキュリティ、多様なデータ形式 | ブラウザ対応、HTTPS必須 |
| **execCommand** | レガシーブラウザ | 実装が簡単、ブラウザ対応 | 非推奨、モバイル困難 |
| **カスタム実装** | 特殊な要件 | 完全な制御、柔軟性 | 実装が複雑、メンテナンス |

### 選択の指針

1. **モダンブラウザ** → Clipboard API
2. **レガシー対応** → execCommand
3. **特殊な要件** → カスタム実装
4. **堅牢性重視** → フォールバック実装
