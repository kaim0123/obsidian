# オフライン対応の実装方法とベストプラクティス

## 📱 概要

オフライン対応は、ネットワーク接続が不安定な環境や完全にオフラインの状況でも、ユーザーがアプリケーションを継続して使用できるようにする重要な機能です。適切なオフライン対応により、ユーザーエクスペリエンスの向上とデータの安全性を確保できます。

## 🎯 主要機能

### 1. オフライン検出
- **ネットワーク状態の監視** - 接続状態のリアルタイム検出
- **接続品質の測定** - ネットワーク速度の監視
- **オフライン状態の通知** - ユーザーへの状態表示
- **自動再接続** - 接続復旧時の自動同期

### 2. データの永続化
- **ローカルストレージ** - ブラウザでのデータ保存
- **IndexedDB** - 大容量データの保存
- **Service Worker キャッシュ** - リソースのキャッシュ
- **SQLite** - 構造化データの保存

### 3. 同期機能
- **差分同期** - 変更部分のみの同期
- **競合解決** - データ競合の処理
- **バックグラウンド同期** - 接続復旧時の自動同期
- **手動同期** - ユーザー主導の同期

### ネットワーク状態の監視

```javascript
import { useState, useEffect } from 'react';

// ネットワーク状態の監視フック
function useNetworkStatus() {
  const [isOnline, setIsOnline] = useState(navigator.onLine);
  const [connectionType, setConnectionType] = useState(null);
  const [connectionSpeed, setConnectionSpeed] = useState(null);

  useEffect(() => {
    // オンライン/オフライン状態の監視
    const handleOnline = () => {
      setIsOnline(true);
      console.log('ネットワーク接続が復旧しました');
    };

    const handleOffline = () => {
      setIsOnline(false);
      console.log('ネットワーク接続が切断されました');
    };

    // 接続情報の取得
    const updateConnectionInfo = () => {
      if ('connection' in navigator) {
        const connection = navigator.connection;
        setConnectionType(connection.effectiveType);
        setConnectionSpeed(connection.downlink);
      }
    };

    // イベントリスナーの登録
    window.addEventListener('online', handleOnline);
    window.addEventListener('offline', handleOffline);
    window.addEventListener('load', updateConnectionInfo);

    // 初期状態の設定
    updateConnectionInfo();

    // クリーンアップ
    return () => {
      window.removeEventListener('online', handleOnline);
      window.removeEventListener('offline', handleOffline);
      window.removeEventListener('load', updateConnectionInfo);
    };
  }, []);

  return {
    isOnline,
    connectionType,
    connectionSpeed,
  };
}

// 使用例
function NetworkStatusIndicator() {
  const { isOnline, connectionType, connectionSpeed } = useNetworkStatus();

  return (
    <div className={`network-status ${isOnline ? 'online' : 'offline'}`}>
      <div className="status-indicator">
        {isOnline ? '🟢 オンライン' : '🔴 オフライン'}
      </div>
      {isOnline && connectionType && (
        <div className="connection-info">
          <span>接続タイプ: {connectionType}</span>
          <span>速度: {connectionSpeed} Mbps</span>
        </div>
      )}
    </div>
  );
}
```

### IndexedDB を使用したデータ永続化

```javascript
// IndexedDB の管理クラス
class OfflineDatabase {
  constructor(dbName, version) {
    this.dbName = dbName;
    this.version = version;
    this.db = null;
  }

  // データベースの初期化
  async init() {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(this.dbName, this.version);

      request.onerror = () => reject(request.error);
      request.onsuccess = () => {
        this.db = request.result;
        resolve(this.db);
      };

      request.onupgradeneeded = (event) => {
        const db = event.target.result;
        
        // ストアの作成
        if (!db.objectStoreNames.contains('users')) {
          const userStore = db.createObjectStore('users', { keyPath: 'id' });
          userStore.createIndex('email', 'email', { unique: true });
        }

        if (!db.objectStoreNames.contains('posts')) {
          const postStore = db.createObjectStore('posts', { keyPath: 'id' });
          postStore.createIndex('userId', 'userId', { unique: false });
          postStore.createIndex('createdAt', 'createdAt', { unique: false });
        }

        if (!db.objectStoreNames.contains('syncQueue')) {
          const syncStore = db.createObjectStore('syncQueue', { keyPath: 'id', autoIncrement: true });
          syncStore.createIndex('type', 'type', { unique: false });
          syncStore.createIndex('timestamp', 'timestamp', { unique: false });
        }
      };
    });
  }

  // データの保存
  async save(storeName, data) {
    const transaction = this.db.transaction([storeName], 'readwrite');
    const store = transaction.objectStore(storeName);
    
    return new Promise((resolve, reject) => {
      const request = store.put(data);
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }

  // データの取得
  async get(storeName, key) {
    const transaction = this.db.transaction([storeName], 'readonly');
    const store = transaction.objectStore(storeName);
    
    return new Promise((resolve, reject) => {
      const request = store.get(key);
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }

  // 全データの取得
  async getAll(storeName) {
    const transaction = this.db.transaction([storeName], 'readonly');
    const store = transaction.objectStore(storeName);
    
    return new Promise((resolve, reject) => {
      const request = store.getAll();
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }

  // データの削除
  async delete(storeName, key) {
    const transaction = this.db.transaction([storeName], 'readwrite');
    const store = transaction.objectStore(storeName);
    
    return new Promise((resolve, reject) => {
      const request = store.delete(key);
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }

  // 同期キューへの追加
  async addToSyncQueue(operation) {
    const syncItem = {
      type: operation.type, // 'create', 'update', 'delete'
      storeName: operation.storeName,
      data: operation.data,
      timestamp: Date.now(),
      synced: false,
    };

    return this.save('syncQueue', syncItem);
  }

  // 同期キューの取得
  async getSyncQueue() {
    return this.getAll('syncQueue');
  }

  // 同期済みアイテムの削除
  async removeSyncedItems() {
    const transaction = this.db.transaction(['syncQueue'], 'readwrite');
    const store = transaction.objectStore('syncQueue');
    const index = store.index('timestamp');
    
    return new Promise((resolve, reject) => {
      const request = index.openCursor();
      request.onsuccess = (event) => {
        const cursor = event.target.result;
        if (cursor) {
          if (cursor.value.synced) {
            cursor.delete();
          }
          cursor.continue();
        } else {
          resolve();
        }
      };
      request.onerror = () => reject(request.error);
    });
  }
}

// 使用例
const offlineDB = new OfflineDatabase('MyAppDB', 1);
await offlineDB.init();
```

### オフライン対応のデータ管理フック

```javascript
import { useState, useEffect, useCallback } from 'react';
import { useNetworkStatus } from './useNetworkStatus';

function useOfflineData(storeName, apiEndpoint) {
  const [data, setData] = useState([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);
  const { isOnline } = useNetworkStatus();

  // データの読み込み
  const loadData = useCallback(async () => {
    setLoading(true);
    setError(null);

    try {
      if (isOnline) {
        // オンライン時はAPIから取得
        const response = await fetch(apiEndpoint);
        if (!response.ok) {
          throw new Error('データの取得に失敗しました');
        }
        const apiData = await response.json();
        setData(apiData);
        
        // ローカルストレージにも保存
        await offlineDB.save(storeName, apiData);
      } else {
        // オフライン時はローカルストレージから取得
        const localData = await offlineDB.getAll(storeName);
        setData(localData);
      }
    } catch (err) {
      setError(err.message);
      
      // エラー時はローカルデータをフォールバック
      try {
        const localData = await offlineDB.getAll(storeName);
        setData(localData);
      } catch (localErr) {
        console.error('ローカルデータの取得にも失敗:', localErr);
      }
    } finally {
      setLoading(false);
    }
  }, [isOnline, storeName, apiEndpoint]);

  // データの追加
  const addData = useCallback(async (newItem) => {
    const tempId = `temp_${Date.now()}`;
    const itemWithId = { ...newItem, id: tempId, isOffline: !isOnline };

    // ローカルに即座に追加
    setData(prev => [...prev, itemWithId]);
    await offlineDB.save(storeName, itemWithId);

    if (isOnline) {
      try {
        // オンライン時はAPIに送信
        const response = await fetch(apiEndpoint, {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
          },
          body: JSON.stringify(newItem),
        });

        if (response.ok) {
          const savedItem = await response.json();
          // ローカルデータを更新
          setData(prev => prev.map(item => 
            item.id === tempId ? savedItem : item
          ));
          await offlineDB.save(storeName, savedItem);
        }
      } catch (err) {
        // エラー時は同期キューに追加
        await offlineDB.addToSyncQueue({
          type: 'create',
          storeName,
          data: newItem,
        });
      }
    } else {
      // オフライン時は同期キューに追加
      await offlineDB.addToSyncQueue({
        type: 'create',
        storeName,
        data: newItem,
      });
    }
  }, [isOnline, storeName, apiEndpoint]);

  // データの更新
  const updateData = useCallback(async (id, updates) => {
    // ローカルに即座に更新
    setData(prev => prev.map(item => 
      item.id === id ? { ...item, ...updates } : item
    ));

    const updatedItem = { ...data.find(item => item.id === id), ...updates };
    await offlineDB.save(storeName, updatedItem);

    if (isOnline) {
      try {
        const response = await fetch(`${apiEndpoint}/${id}`, {
          method: 'PUT',
          headers: {
            'Content-Type': 'application/json',
          },
          body: JSON.stringify(updates),
        });

        if (!response.ok) {
          throw new Error('更新に失敗しました');
        }
      } catch (err) {
        await offlineDB.addToSyncQueue({
          type: 'update',
          storeName,
          data: { id, ...updates },
        });
      }
    } else {
      await offlineDB.addToSyncQueue({
        type: 'update',
        storeName,
        data: { id, ...updates },
      });
    }
  }, [data, isOnline, storeName, apiEndpoint]);

  // データの削除
  const deleteData = useCallback(async (id) => {
    // ローカルから削除
    setData(prev => prev.filter(item => item.id !== id));
    await offlineDB.delete(storeName, id);

    if (isOnline) {
      try {
        const response = await fetch(`${apiEndpoint}/${id}`, {
          method: 'DELETE',
        });

        if (!response.ok) {
          throw new Error('削除に失敗しました');
        }
      } catch (err) {
        await offlineDB.addToSyncQueue({
          type: 'delete',
          storeName,
          data: { id },
        });
      }
    } else {
      await offlineDB.addToSyncQueue({
        type: 'delete',
        storeName,
        data: { id },
      });
    }
  }, [isOnline, storeName, apiEndpoint]);

  // 初期データの読み込み
  useEffect(() => {
    loadData();
  }, [loadData]);

  return {
    data,
    loading,
    error,
    addData,
    updateData,
    deleteData,
    refresh: loadData,
  };
}
```

### バックグラウンド同期の実装

```javascript
// バックグラウンド同期サービス
class BackgroundSyncService {
  constructor(offlineDB) {
    this.offlineDB = offlineDB;
    this.isSyncing = false;
    this.syncInterval = null;
  }

  // 同期の開始
  startSync(interval = 30000) { // 30秒間隔
    if (this.syncInterval) {
      clearInterval(this.syncInterval);
    }

    this.syncInterval = setInterval(() => {
      this.performSync();
    }, interval);

    // 初回同期
    this.performSync();
  }

  // 同期の停止
  stopSync() {
    if (this.syncInterval) {
      clearInterval(this.syncInterval);
      this.syncInterval = null;
    }
  }

  // 同期の実行
  async performSync() {
    if (this.isSyncing) return;

    this.isSyncing = true;

    try {
      const syncQueue = await this.offlineDB.getSyncQueue();
      const pendingItems = syncQueue.filter(item => !item.synced);

      for (const item of pendingItems) {
        await this.syncItem(item);
      }

      // 同期済みアイテムのクリーンアップ
      await this.offlineDB.removeSyncedItems();

      console.log('同期が完了しました');
    } catch (error) {
      console.error('同期エラー:', error);
    } finally {
      this.isSyncing = false;
    }
  }

  // 個別アイテムの同期
  async syncItem(item) {
    try {
      let response;

      switch (item.type) {
        case 'create':
          response = await fetch(`/api/${item.storeName}`, {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json',
            },
            body: JSON.stringify(item.data),
          });
          break;

        case 'update':
          response = await fetch(`/api/${item.storeName}/${item.data.id}`, {
            method: 'PUT',
            headers: {
              'Content-Type': 'application/json',
            },
            body: JSON.stringify(item.data),
          });
          break;

        case 'delete':
          response = await fetch(`/api/${item.storeName}/${item.data.id}`, {
            method: 'DELETE',
          });
          break;

        default:
          throw new Error(`未知の同期タイプ: ${item.type}`);
      }

      if (response.ok) {
        // 同期成功
        item.synced = true;
        await this.offlineDB.save('syncQueue', item);
      } else {
        throw new Error(`同期に失敗: ${response.status}`);
      }
    } catch (error) {
      console.error(`アイテムの同期に失敗:`, error);
      throw error;
    }
  }
}

// 使用例
const syncService = new BackgroundSyncService(offlineDB);
syncService.startSync();
```

### Service Worker でのオフライン対応

```javascript
// service-worker.js
const CACHE_NAME = 'offline-cache-v1';
const API_CACHE_NAME = 'api-cache-v1';

// キャッシュするリソース
const urlsToCache = [
  '/',
  '/static/js/bundle.js',
  '/static/css/main.css',
  '/manifest.json',
  '/offline.html',
];

// インストール時のキャッシュ
self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME)
      .then((cache) => {
        return cache.addAll(urlsToCache);
      })
  );
});

// フェッチ時の処理
self.addEventListener('fetch', (event) => {
  const { request } = event;
  const url = new URL(request.url);

  // API リクエストの処理
  if (url.pathname.startsWith('/api/')) {
    event.respondWith(handleApiRequest(request));
  } else {
    // 静的リソースの処理
    event.respondWith(handleStaticRequest(request));
  }
});

// API リクエストの処理
async function handleApiRequest(request) {
  try {
    // ネットワークから取得を試行
    const response = await fetch(request);
    
    if (response.ok) {
      // 成功時はキャッシュに保存
      const cache = await caches.open(API_CACHE_NAME);
      cache.put(request, response.clone());
    }
    
    return response;
  } catch (error) {
    // ネットワークエラー時はキャッシュから取得
    const cachedResponse = await caches.match(request);
    
    if (cachedResponse) {
      return cachedResponse;
    }
    
    // キャッシュにもない場合はオフラインページを返す
    return new Response(
      JSON.stringify({ 
        error: 'オフラインです',
        offline: true 
      }),
      {
        status: 503,
        headers: { 'Content-Type': 'application/json' }
      }
    );
  }
}

// 静的リソースの処理
async function handleStaticRequest(request) {
  try {
    // キャッシュから取得を試行
    const cachedResponse = await caches.match(request);
    
    if (cachedResponse) {
      return cachedResponse;
    }
    
    // ネットワークから取得
    const response = await fetch(request);
    
    if (response.ok) {
      const cache = await caches.open(CACHE_NAME);
      cache.put(request, response.clone());
    }
    
    return response;
  } catch (error) {
    // オフライン時のフォールバック
    if (request.destination === 'document') {
      return caches.match('/offline.html');
    }
    
    return new Response('オフラインです', { status: 503 });
  }
}

// バックグラウンド同期
self.addEventListener('sync', (event) => {
  if (event.tag === 'background-sync') {
    event.waitUntil(doBackgroundSync());
  }
});

async function doBackgroundSync() {
  try {
    // 同期キューからデータを取得して同期
    const response = await fetch('/api/sync');
    if (response.ok) {
      console.log('バックグラウンド同期が完了しました');
    }
  } catch (error) {
    console.error('バックグラウンド同期エラー:', error);
  }
}
```

```javascript
class DataCompression {
  // データの圧縮
  static compress(data) {
    const jsonString = JSON.stringify(data);
    // 簡易的な圧縮（実際の実装ではLZ-string等を使用）
    return btoa(jsonString);
  }

  // データの展開
  static decompress(compressedData) {
    try {
      const jsonString = atob(compressedData);
      return JSON.parse(jsonString);

  }
```
