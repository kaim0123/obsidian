# ダークモード・ライトモード切り替えの実装方法とベストプラクティス

## 概要

ダークモード・ライトモード切り替えは、ユーザーの好みや環境に応じてアプリケーションのテーマを動的に変更する機能です。目の疲労軽減、バッテリー節約、アクセシビリティ向上など、多くのメリットをもたらす重要な機能です。

### Reactコンポーネントでの実装

```javascript
// テーマコンテキストの作成
import { createContext, useContext, useEffect, useState } from 'react';

const ThemeContext = createContext();

export const useTheme = () => {
  const context = useContext(ThemeContext);
  if (!context) {
    throw new Error('useTheme must be used within a ThemeProvider');
  }
  return context;
};

export const ThemeProvider = ({ children }) => {
  const [theme, setTheme] = useState(() => {
    // ローカルストレージからテーマを取得
    const savedTheme = localStorage.getItem('theme');
    if (savedTheme) {
      return savedTheme;
    }
    
    // システム設定をチェック
    if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) {
      return 'dark';
    }
    
    return 'light';
  });

  useEffect(() => {
    // テーマをローカルストレージに保存
    localStorage.setItem('theme', theme);
    
    // HTML要素にdata-theme属性を設定
    document.documentElement.setAttribute('data-theme', theme);
  }, [theme]);

  const toggleTheme = () => {
    setTheme(prevTheme => prevTheme === 'light' ? 'dark' : 'light');
  };

  const value = {
    theme,
    toggleTheme,
    isDark: theme === 'dark'
  };

  return (
    <ThemeContext.Provider value={value}>
      {children}
    </ThemeContext.Provider>
  );
};

// テーマ切り替えボタンコンポーネント
function ThemeToggle() {
  const { theme, toggleTheme, isDark } = useTheme();

  return (
    <button 
      className="theme-toggle"
      onClick={toggleTheme}
      aria-label={`${isDark ? 'ライト' : 'ダーク'}モードに切り替え`}
    >
      {isDark ? '☀️' : '🌙'}
    </button>
  );
}

// 使用例
function App() {
  return (
    <ThemeProvider>
      <div className="app">
        <header>
          <h1>My App</h1>
          <ThemeToggle />
        </header>
        <main>
          <div className="container">
            <div className="card">
              <h2>Welcome</h2>
              <p>This is a theme-aware application.</p>
            </div>
          </div>
        </main>
      </div>
    </ThemeProvider>
  );
}
```

### JavaScriptでの実装

```javascript
// テーマ管理クラス
class ThemeManager {
  constructor() {
    this.currentTheme = this.getStoredTheme() || this.getSystemTheme();
    this.applyTheme(this.currentTheme);
    this.setupSystemThemeListener();
  }

  getStoredTheme() {
    return localStorage.getItem('theme');
  }

  getSystemTheme() {
    if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) {
      return 'dark';
    }
    return 'light';
  }

  applyTheme(theme) {
    document.documentElement.className = `${theme}-theme theme-transition`;
    this.currentTheme = theme;
    localStorage.setItem('theme', theme);
    
    // カスタムイベントを発火
    window.dispatchEvent(new CustomEvent('themeChanged', { 
      detail: { theme, isDark: theme === 'dark' } 
    }));
  }

  toggleTheme() {
    const newTheme = this.currentTheme === 'light' ? 'dark' : 'light';
    this.applyTheme(newTheme);
  }

  setupSystemThemeListener() {
    if (window.matchMedia) {
      const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)');
      mediaQuery.addListener((e) => {
        // ユーザーが手動でテーマを設定していない場合のみシステム設定に従う
        if (!localStorage.getItem('theme')) {
          this.applyTheme(e.matches ? 'dark' : 'light');
        }
      });
    }
  }
}

// テーマ切り替えボタン
class ThemeToggle {
  constructor(themeManager) {
    this.themeManager = themeManager;
    this.button = this.createButton();
    this.setupEventListeners();
  }

  createButton() {
    const button = document.createElement('button');
    button.className = 'theme-toggle-button';
    button.setAttribute('aria-label', 'テーマを切り替え');
    button.innerHTML = this.getIcon();
    return button;
  }

  getIcon() {
    return this.themeManager.currentTheme === 'dark' ? '☀️' : '🌙';
  }

  setupEventListeners() {
    this.button.addEventListener('click', () => {
      this.themeManager.toggleTheme();
      this.updateIcon();
    });

    // テーマ変更イベントをリッスン
    window.addEventListener('themeChanged', () => {
      this.updateIcon();
    });
  }

  updateIcon() {
    this.button.innerHTML = this.getIcon();
  }

  render(container) {
    container.appendChild(this.button);
  }
}

// 使用例
document.addEventListener('DOMContentLoaded', () => {
  const themeManager = new ThemeManager();
  const themeToggle = new ThemeToggle(themeManager);
  
  // ヘッダーにテーマ切り替えボタンを追加
  const header = document.querySelector('header');
  if (header) {
    themeToggle.render(header);
  }

  // テーマ変更イベントをリッスン
  window.addEventListener('themeChanged', (event) => {
    console.log('Theme changed to:', event.detail.theme);
    // 必要に応じて他のコンポーネントを更新
  });
});
```

## 4. システム設定連動

### 概要
ユーザーのOS設定に基づいて自動的にテーマを切り替える方法

### 実装例

```javascript
// システム設定連動のテーマ管理
class SystemThemeManager {
  constructor() {
    this.mediaQuery = window.matchMedia('(prefers-color-scheme: dark)');
    this.currentTheme = this.getInitialTheme();
    this.setupSystemListener();
    this.applyTheme(this.currentTheme);
  }

  getInitialTheme() {
    // ユーザーが手動でテーマを設定している場合はそれを優先
    const storedTheme = localStorage.getItem('theme');
    if (storedTheme) {
      return storedTheme;
    }

    // システム設定に従う
    return this.mediaQuery.matches ? 'dark' : 'light';
  }

  setupSystemListener() {
    this.mediaQuery.addListener((e) => {
      // ユーザーが手動でテーマを設定していない場合のみシステム設定に従う
      if (!localStorage.getItem('theme')) {
        this.applyTheme(e.matches ? 'dark' : 'light');
      }
    });
  }

  applyTheme(theme) {
    document.documentElement.setAttribute('data-theme', theme);
    this.currentTheme = theme;
    
    // 手動設定の場合のみローカルストレージに保存
    if (localStorage.getItem('theme')) {
      localStorage.setItem('theme', theme);
    }
  }

  setManualTheme(theme) {
    localStorage.setItem('theme', theme);
    this.applyTheme(theme);
  }

  resetToSystemTheme() {
    localStorage.removeItem('theme');
    const systemTheme = this.mediaQuery.matches ? 'dark' : 'light';
    this.applyTheme(systemTheme);
  }

  isSystemTheme() {
    return !localStorage.getItem('theme');
  }
}

// React Hook版
import { useState, useEffect } from 'react';

export const useSystemTheme = () => {
  const [theme, setTheme] = useState(() => {
    const storedTheme = localStorage.getItem('theme');
    if (storedTheme) {
      return storedTheme;
    }
    
    if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) {
      return 'dark';
    }
    
    return 'light';
  });

  const [isSystemTheme, setIsSystemTheme] = useState(() => {
    return !localStorage.getItem('theme');
  });

  useEffect(() => {
    const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)');
    
    const handleChange = (e) => {
      if (isSystemTheme) {
        setTheme(e.matches ? 'dark' : 'light');
      }
    };

    mediaQuery.addListener(handleChange);
    return () => mediaQuery.removeListener(handleChange);
  }, [isSystemTheme]);

  const setManualTheme = (newTheme) => {
    setTheme(newTheme);
    setIsSystemTheme(false);
    localStorage.setItem('theme', newTheme);
  };

  const resetToSystemTheme = () => {
    const systemTheme = window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light';
    setTheme(systemTheme);
    setIsSystemTheme(true);
    localStorage.removeItem('theme');
  };

  return {
    theme,
    isSystemTheme,
    setManualTheme,
    resetToSystemTheme,
    isDark: theme === 'dark'
  };
};

// 使用例
function SystemThemeToggle() {
  const { theme, isSystemTheme, setManualTheme, resetToSystemTheme, isDark } = useSystemTheme();

  return (
    <div className="theme-controls">
      <button 
        onClick={() => setManualTheme(isDark ? 'light' : 'dark')}
        className="theme-toggle"
      >
        {isDark ? '☀️' : '🌙'}
      </button>
      
      {!isSystemTheme && (
        <button 
          onClick={resetToSystemTheme}
          className="system-theme-button"
        >
          システム設定に従う
        </button>
      )}
      
      <span className="theme-status">
        {isSystemTheme ? 'システム設定に従っています' : '手動設定'}
      </span>
    </div>
  );
}
```

### メリット
- ✅ **ユーザビリティ**: ユーザーの設定に自動的に従う
- ✅ **アクセシビリティ**: システム設定を尊重
- ✅ **パフォーマンス**: 軽量で高速
- ✅ **直感的**: ユーザーが期待する動作

### デメリット
- ❌ **制御の限界**: ユーザーが細かい制御をできない
- ❌ **ブラウザ対応**: 古いブラウザでは対応していない

### 使用場面
- 📱 **PWA**: プログレッシブWebアプリ
- 🖥️ **デスクトップアプリ**: Electronアプリ
- 🎨 **デザインシステム**: システム設定を重視するアプリ

const themes = {
  light: {
    name: 'ライト',
    colors: {
      primary: '#007bff',
      secondary: '#6c757d',
      background: '#ffffff',
      surface: '#f8f9fa',

    }
  },
  dark: {
    name: 'ダーク',
    colors: {
      primary: '#4dabf7',
      secondary: '#adb5bd',
      background: '#1a1a1a',
      surface: '#2d2d2d',
      text: '#ffffff'
    }
  },
  blue: {
    name: 'ブルー',
    colors: {
      primary: '#1e3a8a',
      secondary: '#3b82f6',
      background: '#dbeafe',
      surface: '#bfdbfe',
      text: '#1e40af'
    }
  },
  green: {
    name: 'グリーン',
    colors: {
      primary: '#059669',
      secondary: '#10b981',
      background: '#ecfdf5',
      surface: '#d1fae5',
      text: '#047857'
    }
  }
};

class MultiThemeManager {
  constructor() {
    this.currentTheme = this.getStoredTheme() || 'light';
    this.applyTheme(this.currentTheme);
  }

  getStoredTheme() {
    return localStorage.getItem('theme');
  }

  applyTheme(themeName) {
    const theme = themes[themeName];
    if (!theme) return;

    const root = document.documentElement;
    Object.entries(theme.colors).forEach(([key, value]) => {
      root.style.setProperty(`--color-${key}`, value);
    });

    this.currentTheme = themeName;
    localStorage.setItem('theme', themeName);
  }

  getAvailableThemes() {
    return Object.entries(themes).map(([key, theme]) => ({
      key,
      name: theme.name
    }));
  }
}

### 3. テーマの永続化

```javascript
// テーマの永続化と同期
class PersistentThemeManager {
  constructor() {
    this.currentTheme = this.getStoredTheme() || this.getSystemTheme();
    this.applyTheme(this.currentTheme);
    this.setupStorageSync();
  }

  getStoredTheme() {
    return localStorage.getItem('theme');
  }

  getSystemTheme() {
    if (window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches) {
      return 'dark';
    }
    return 'light';
  }

  applyTheme(theme) {
    document.documentElement.setAttribute('data-theme', theme);
    this.currentTheme = theme;
    localStorage.setItem('theme', theme);
    
    // 他のタブと同期
    this.broadcastThemeChange(theme);
  }

  setupStorageSync() {
    // 他のタブでのテーマ変更を検知
    window.addEventListener('storage', (e) => {
      if (e.key === 'theme' && e.newValue !== this.currentTheme) {
        this.currentTheme = e.newValue;
        document.documentElement.setAttribute('data-theme', e.newValue);
      }
    });
  }

  broadcastThemeChange(theme) {
    // 他のタブにテーマ変更を通知
    localStorage.setItem('theme', theme);
  }
}
```

## まとめ

| 実装方法 | 適用場面 | 主なメリット | 主なデメリット |
|----------|---------|-------------|---------------|
| **CSS変数** | モダンなWebアプリ | パフォーマンス、メンテナンス性 | 古いブラウザ対応 |
| **CSS-in-JS** | 複雑なスタイリング | 型安全性、動的スタイル | バンドルサイズ、学習コスト |
| **クラス切り替え** | シンプルなサイト | シンプル、ブラウザ対応 | 柔軟性の制限 |
| **システム設定連動** | ネイティブアプリ風 | ユーザビリティ、アクセシビリティ | 制御の限界 |

### 選択の指針

1. **モダンなWebアプリ** → CSS変数
2. **複雑なスタイリング** → CSS-in-JS
3. **シンプルなサイト** → クラス切り替え
4. **ネイティブアプリ風** → システム設定連動
