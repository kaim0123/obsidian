# セッション管理の実装方法とベストプラクティス

## 🔐 概要

セッション管理は、ユーザーの認証状態を維持し、アプリケーション全体で一貫したユーザー体験を提供するための重要な機能です。適切なセッション管理により、セキュリティを確保しながら、ユーザーの利便性を向上させることができます。

## 🎯 主要機能

### 1. 認証状態の管理
- **ログイン状態の維持** - ユーザーの認証情報の管理
- **セッションの有効期限** - 自動的なログアウト機能
- **複数デバイス対応** - デバイス間でのセッション同期
- **セッションの無効化** - ログアウト時の適切な処理

### 2. ユーザー情報の管理
- **プロフィール情報** - ユーザーデータの保存と更新
- **権限管理** - ロールベースのアクセス制御
- **設定情報** - ユーザー固有の設定の保存
- **履歴管理** - アクティビティログの記録

### 3. セキュリティ機能
- **CSRF対策** - クロスサイトリクエストフォージェリの防止
- **XSS対策** - クロスサイトスクリプティングの防止
- **セッションハイジャック対策** - 不正アクセスの検出
- **多要素認証** - 追加のセキュリティ層

### JWT ベースのセッション管理

```javascript
// JWT トークンの管理
class SessionManager {
  constructor() {
    this.tokenKey = 'authToken';
    this.refreshTokenKey = 'refreshToken';
  }

  // トークンの保存
  setTokens(accessToken, refreshToken) {
    localStorage.setItem(this.tokenKey, accessToken);
    localStorage.setItem(this.refreshTokenKey, refreshToken);
  }

  // アクセストークンの取得
  getAccessToken() {
    return localStorage.getItem(this.tokenKey);
  }

  // リフレッシュトークンの取得
  getRefreshToken() {
    return localStorage.getItem(this.refreshTokenKey);
  }

  // トークンの削除
  clearTokens() {
    localStorage.removeItem(this.tokenKey);
    localStorage.removeItem(this.refreshTokenKey);
  }

  // トークンの有効性チェック
  isTokenValid(token) {
    if (!token) return false;
    
    try {
      const payload = JSON.parse(atob(token.split('.')[1]));
      const currentTime = Date.now() / 1000;
      return payload.exp > currentTime;
    } catch (error) {
      return false;
    }
  }

  // トークンの自動更新
  async refreshAccessToken() {
    const refreshToken = this.getRefreshToken();
    if (!refreshToken) {
      throw new Error('リフレッシュトークンがありません');
    }

    try {
      const response = await fetch('/api/auth/refresh', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({ refreshToken }),
      });

      if (!response.ok) {
        throw new Error('トークンの更新に失敗しました');
      }

      const { accessToken, refreshToken: newRefreshToken } = await response.json();
      this.setTokens(accessToken, newRefreshToken);
      return accessToken;
    } catch (error) {
      this.clearTokens();
      throw error;
    }
  }
}
```

### React Context を使用したセッション管理

```javascript
import React, { createContext, useContext, useReducer, useEffect } from 'react';

// セッション状態の型定義
const initialState = {
  user: null,
  isAuthenticated: false,
  isLoading: true,
  error: null,
};

// アクションタイプ
const SESSION_ACTIONS = {
  LOGIN_START: 'LOGIN_START',
  LOGIN_SUCCESS: 'LOGIN_SUCCESS',
  LOGIN_FAILURE: 'LOGIN_FAILURE',
  LOGOUT: 'LOGOUT',
  SET_LOADING: 'SET_LOADING',
  UPDATE_USER: 'UPDATE_USER',
};

// リデューサー
function sessionReducer(state, action) {
  switch (action.type) {
    case SESSION_ACTIONS.LOGIN_START:
      return {
        ...state,
        isLoading: true,
        error: null,
      };
    case SESSION_ACTIONS.LOGIN_SUCCESS:
      return {
        ...state,
        user: action.payload.user,
        isAuthenticated: true,
        isLoading: false,
        error: null,
      };
    case SESSION_ACTIONS.LOGIN_FAILURE:
      return {
        ...state,
        user: null,
        isAuthenticated: false,
        isLoading: false,
        error: action.payload.error,
      };
    case SESSION_ACTIONS.LOGOUT:
      return {
        ...state,
        user: null,
        isAuthenticated: false,
        isLoading: false,
        error: null,
      };
    case SESSION_ACTIONS.SET_LOADING:
      return {
        ...state,
        isLoading: action.payload,
      };
    case SESSION_ACTIONS.UPDATE_USER:
      return {
        ...state,
        user: { ...state.user, ...action.payload },
      };
    default:
      return state;
  }
}

// セッションコンテキスト
const SessionContext = createContext();

// セッションプロバイダー
export function SessionProvider({ children }) {
  const [state, dispatch] = useReducer(sessionReducer, initialState);
  const sessionManager = new SessionManager();

  // 初期化時の認証状態チェック
  useEffect(() => {
    const initializeSession = async () => {
      try {
        const token = sessionManager.getAccessToken();
        if (token && sessionManager.isTokenValid(token)) {
          // トークンが有効な場合、ユーザー情報を取得
          const user = await fetchUserProfile(token);
          dispatch({
            type: SESSION_ACTIONS.LOGIN_SUCCESS,
            payload: { user },
          });
        } else {
          // トークンが無効な場合、リフレッシュを試行
          try {
            await sessionManager.refreshAccessToken();
            const newToken = sessionManager.getAccessToken();
            const user = await fetchUserProfile(newToken);
            dispatch({
              type: SESSION_ACTIONS.LOGIN_SUCCESS,
              payload: { user },
            });
          } catch (error) {
            // リフレッシュも失敗した場合、ログアウト
            sessionManager.clearTokens();
            dispatch({ type: SESSION_ACTIONS.LOGOUT });
          }
        }
      } catch (error) {
        dispatch({
          type: SESSION_ACTIONS.LOGIN_FAILURE,
          payload: { error: error.message },
        });
      }
    };

    initializeSession();
  }, []);

  // ログイン関数
  const login = async (credentials) => {
    dispatch({ type: SESSION_ACTIONS.LOGIN_START });
    
    try {
      const response = await fetch('/api/auth/login', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify(credentials),
      });

      if (!response.ok) {
        throw new Error('ログインに失敗しました');
      }

      const { accessToken, refreshToken, user } = await response.json();
      sessionManager.setTokens(accessToken, refreshToken);
      
      dispatch({
        type: SESSION_ACTIONS.LOGIN_SUCCESS,
        payload: { user },
      });
    } catch (error) {
      dispatch({
        type: SESSION_ACTIONS.LOGIN_FAILURE,
        payload: { error: error.message },
      });
    }
  };

  // ログアウト関数
  const logout = async () => {
    try {
      const token = sessionManager.getAccessToken();
      if (token) {
        await fetch('/api/auth/logout', {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${token}`,
          },
        });
      }
    } catch (error) {
      console.error('ログアウトエラー:', error);
    } finally {
      sessionManager.clearTokens();
      dispatch({ type: SESSION_ACTIONS.LOGOUT });
    }
  };

  // ユーザー情報更新関数
  const updateUser = (userData) => {
    dispatch({
      type: SESSION_ACTIONS.UPDATE_USER,
      payload: userData,
    });
  };

  const value = {
    ...state,
    login,
    logout,
    updateUser,
  };

  return (
    <SessionContext.Provider value={value}>
      {children}
    </SessionContext.Provider>
  );
}

// セッションフック
export function useSession() {
  const context = useContext(SessionContext);
  if (!context) {
    throw new Error('useSession must be used within a SessionProvider');
  }
  return context;
}
```

### HTTP インターセプターでの自動トークン更新

```javascript
// Axios インターセプターの設定
import axios from 'axios';

const apiClient = axios.create({
  baseURL: process.env.REACT_APP_API_BASE_URL,
});

// リクエストインターセプター
apiClient.interceptors.request.use(
  (config) => {
    const token = sessionManager.getAccessToken();
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

// レスポンスインターセプター
apiClient.interceptors.response.use(
  (response) => {
    return response;
  },
  async (error) => {
    const originalRequest = error.config;

    if (error.response?.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true;

      try {
        const newToken = await sessionManager.refreshAccessToken();
        originalRequest.headers.Authorization = `Bearer ${newToken}`;
        return apiClient(originalRequest);
      } catch (refreshError) {
        // リフレッシュに失敗した場合、ログアウト
        sessionManager.clearTokens();
        window.location.href = '/login';
        return Promise.reject(refreshError);
      }
    }

    return Promise.reject(error);
  }
);
```

## 🔒 セキュリティ機能

### 1. CSRF トークンの実装

```javascript
class CSRFProtection {
  constructor() {
    this.tokenKey = 'csrfToken';
  }

  // CSRF トークンの取得
  async getCSRFToken() {
    let token = localStorage.getItem(this.tokenKey);
    
    if (!token) {
      try {
        const response = await fetch('/api/csrf-token');
        const data = await response.json();
        token = data.csrfToken;
        localStorage.setItem(this.tokenKey, token);
      } catch (error) {
        console.error('CSRF トークンの取得に失敗:', error);
      }
    }
    
    return token;
  }

  // リクエストにCSRFトークンを追加
  async addCSRFToken(config) {
    const token = await this.getCSRFToken();
    if (token) {
      config.headers['X-CSRF-Token'] = token;
    }
    return config;
  }
}
```

### 2. セッションタイムアウトの管理

```javascript
class SessionTimeout {
  constructor(timeoutDuration = 30 * 60 * 1000) { // 30分
    this.timeoutDuration = timeoutDuration;
    this.timeoutId = null;
    this.lastActivity = Date.now();
    this.warningTime = 5 * 60 * 1000; // 5分前の警告
  }

  // アクティビティの記録
  recordActivity() {
    this.lastActivity = Date.now();
    this.resetTimeout();
  }

  // タイムアウトのリセット
  resetTimeout() {
    if (this.timeoutId) {
      clearTimeout(this.timeoutId);
    }

    this.timeoutId = setTimeout(() => {
      this.handleTimeout();
    }, this.timeoutDuration);

    // 警告タイマーの設定
    setTimeout(() => {
      this.showWarning();
    }, this.timeoutDuration - this.warningTime);
  }

  // タイムアウト警告の表示
  showWarning() {
    const remainingTime = Math.ceil(this.warningTime / 1000 / 60);
    const shouldExtend = confirm(
      `セッションが${remainingTime}分後に期限切れになります。セッションを延長しますか？`
    );

    if (shouldExtend) {
      this.recordActivity();
    }
  }

  // タイムアウト処理
  handleTimeout() {
    sessionManager.clearTokens();
    alert('セッションが期限切れになりました。再度ログインしてください。');
    window.location.href = '/login';
  }

  // イベントリスナーの設定
  setupActivityListeners() {
    const events = ['mousedown', 'mousemove', 'keypress', 'scroll', 'touchstart'];
    
    events.forEach(event => {
      document.addEventListener(event, () => {
        this.recordActivity();
      }, true);
    });
  }
}
```

function ProtectedRoute({ children }) {
  const { isAuthenticated, isLoading } = useSession();

  if (isLoading) {
    return <LoadingSpinner />;
  }

  if (!isAuthenticated) {
    return <Navigate to="/login" replace />;
  }

  return children;
}
