## 🔐 認証・認可パターン

### 基本的な認証ミドルウェア

```typescript
const authMiddleware = (req, res, next) => {
  // トークンを取得
  const token = req.headers.authorization?.replace('Bearer ', '');
  
  if (!token) {
    return res.status(401).json({ 
      error: '認証トークンが必要です' 
    });
  }
  
  // トークンを検証（簡易版）
  try {
    // 実際の実装では、JWTの検証などを行う
    const decoded = verifyToken(token);
    req.user = decoded; // ユーザー情報をリクエストに追加
    next();
  } catch (error) {
    return res.status(401).json({ 
      error: '無効なトークンです' 
    });
  }
};

// 保護されたルートに適用
app.get('/api/dashboard', authMiddleware, (req, res) => {
  res.json({ 
    message: 'ダッシュボードデータ',
    user: req.user 
  });
});
```

const adminOnly = (req, res, next) => {

### 複数のロールを許可

```typescript
const hasRole = (...allowedRoles) => {
  return (req, res, next) => {
    if (!req.user) {
      return res.status(401).json({ error: '認証が必要です' });
    }
    
    if (!allowedRoles.includes(req.user.role)) {
      return res.status(403).json({ 
        error: '権限が不足しています' 
      });
    }
    
    next();
  };
};

// 使用例: 管理者または編集者のみ
app.put('/api/posts/:id', 
  authMiddleware,
  hasRole('admin', 'editor'),
  (req, res) => {
    // 投稿を更新
  }
);
```

## 📝 ログ記録パターン

### 詳細なログ記録

```typescript
const detailedLogger = (req, res, next) => {
  const start = Date.now();
  
  // レスポンスが終わった後にログを記録
  res.on('finish', () => {
    const duration = Date.now() - start;
    const log = {
      timestamp: new Date().toISOString(),
      method: req.method,
      path: req.path,
      statusCode: res.statusCode,
      duration: `${duration}ms`,
      ip: req.ip,
      userAgent: req.get('user-agent')
    };
    
    console.log(JSON.stringify(log));
  });
  
  next();
};

app.use(detailedLogger);
```

### レート制限

```typescript
// 簡易的なレート制限（実際には専用ライブラリを使用推奨）
const rateLimitMap = new Map();

const rateLimiter = (maxRequests = 10, windowMs = 60000) => {
  return (req, res, next) => {
    const ip = req.ip;
    const now = Date.now();
    
    // IPごとのリクエスト履歴を取得
    const requests = rateLimitMap.get(ip) || [];
    
    // 時間窓外のリクエストを削除
    const recentRequests = requests.filter(
      time => now - time < windowMs
    );
    
    if (recentRequests.length >= maxRequests) {
      return res.status(429).json({ 
        error: 'リクエストが多すぎます。しばらく待ってから再試行してください。' 
      });
    }
    
    // 現在のリクエストを記録
    recentRequests.push(now);
    rateLimitMap.set(ip, recentRequests);
    
    next();
  };
};

// 1分間に10回まで
app.use('/api', rateLimiter(10, 60000));
```

### CORS設定

```typescript
const corsMiddleware = (req, res, next) => {
  const origin = req.headers.origin;
  const allowedOrigins = [
    'https://example.com',
    'https://app.example.com'
  ];
  
  if (allowedOrigins.includes(origin)) {
    res.setHeader('Access-Control-Allow-Origin', origin);
  }
  
  res.setHeader('Access-Control-Allow-Methods', 'GET, POST, PUT, DELETE, OPTIONS');
  res.setHeader('Access-Control-Allow-Headers', 'Content-Type, Authorization');
  res.setHeader('Access-Control-Allow-Credentials', 'true');
  
  // OPTIONSリクエスト（プリフライト）の処理
  if (req.method === 'OPTIONS') {
    return res.sendStatus(200);
  }
  
  next();
};

app.use(corsMiddleware);
```

## 🔄 データ変換パターン

### リクエストボディの検証

```typescript
const validateBody = (schema) => {
  return (req, res, next) => {
    try {
      // スキーマで検証（例: Joi, Zodなど）
      const validated = schema.parse(req.body);
      req.body = validated; // 検証済みデータに置き換え
      next();
    } catch (error) {
      return res.status(400).json({ 
        error: 'リクエストデータが無効です',
        details: error.errors 
      });
    }
  };
};

// 使用例
const userSchema = {
  name: { type: 'string', required: true },
  email: { type: 'string', required: true, format: 'email' }
};

app.post('/api/users', 
  validateBody(userSchema),
  (req, res) => {
    // req.bodyは検証済み
    res.json({ message: 'ユーザー作成成功' });
  }
);
```

### リクエストデータの正規化

```typescript
const normalizeRequest = (req, res, next) => {
  // 文字列のトリム
  if (req.body && typeof req.body === 'object') {
    Object.keys(req.body).forEach(key => {
      if (typeof req.body[key] === 'string') {
        req.body[key] = req.body[key].trim();
      }
    });
  }
  
  // クエリパラメータの正規化
  if (req.query) {
    Object.keys(req.query).forEach(key => {
      if (req.query[key] === '') {
        delete req.query[key];
      }
    });
  }
  
  next();
};

app.use(normalizeRequest);
```

## ⚠️ エラーハンドリングパターン

### グローバルエラーハンドラー

```typescript
const errorHandler = (error, req, res, next) => {
  // エラーの種類に応じてステータスコードを設定
  let statusCode = 500;
  let message = 'サーバーエラーが発生しました';
  
  if (error.name === 'ValidationError') {
    statusCode = 400;
    message = 'バリデーションエラー';
  } else if (error.name === 'UnauthorizedError') {
    statusCode = 401;
    message = '認証が必要です';
  } else if (error.name === 'ForbiddenError') {
    statusCode = 403;
    message = '権限が不足しています';
  } else if (error.name === 'NotFoundError') {
    statusCode = 404;
    message = 'リソースが見つかりません';
  }
  
  // 開発環境では詳細なエラー情報を返す
  const response = {
    error: message,
    ...(process.env.NODE_ENV === 'development' && {
      details: error.message,
      stack: error.stack
    })
  };
  
  res.status(statusCode).json(response);
};

// 最後にエラーハンドラーを登録
app.use(errorHandler);
```

## 🎨 カスタムミドルウェアの作成

### 再利用可能なミドルウェアファクトリー

```typescript
// 設定を受け取るミドルウェアファクトリー
const createAuthMiddleware = (options = {}) => {
  const {
    tokenHeader = 'authorization',
    tokenPrefix = 'Bearer ',
    errorMessage = '認証が必要です'
  } = options;
  
  return (req, res, next) => {
    const header = req.headers[tokenHeader];
    const token = header?.replace(tokenPrefix, '');
    
    if (!token) {
      return res.status(401).json({ error: errorMessage });
    }
    
    // トークン検証処理
    req.token = token;
    next();
  };
};

// 使用例
const apiAuth = createAuthMiddleware({
  tokenHeader: 'x-api-key',
  tokenPrefix: '',
  errorMessage: 'APIキーが必要です'
});

app.use('/api', apiAuth);
```
