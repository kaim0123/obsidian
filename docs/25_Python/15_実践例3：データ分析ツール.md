## Step 1：データ分析クラスの作成

`analyzer.py` を作成します：

```python
# analyzer.py
import csv
from typing import List, Dict, Optional
from collections import Counter

class DataAnalyzer:
    """データ分析ツール"""
    def __init__(self):
        self.data: List[Dict] = []
        self.columns: List[str] = []
    
    def load_csv(self, filename: str):
        """CSVファイルを読み込む"""
        try:
            with open(filename, "r", encoding="utf-8") as file:
                reader = csv.DictReader(file)
                self.data = list(reader)
                if self.data:
                    self.columns = list(self.data[0].keys())
                print(f"データを読み込みました: {len(self.data)}件")
        except FileNotFoundError:
            print(f"{filename} が見つかりません")
        except Exception as e:
            print(f"エラーが発生しました: {e}")
    
    def get_basic_stats(self, column: str):
        """基本統計情報を取得"""
        if not self.data or column not in self.columns:
            return None
        
        values = []
        for row in self.data:
            try:
                value = float(row[column])
                values.append(value)
            except ValueError:
                continue
        
        if not values:
            return None
        
        stats = {
            "件数": len(values),
            "合計": sum(values),
            "平均": sum(values) / len(values),
            "最小値": min(values),
            "最大値": max(values),
            "中央値": sorted(values)[len(values) // 2]
        }
        
        return stats
    
    def get_frequency(self, column: str, limit: int = 10):
        """頻度分布を取得"""
        if not self.data or column not in self.columns:
            return None
        
        values = [row[column] for row in self.data]
        counter = Counter(values)
        
        return counter.most_common(limit)
    
    def filter_data(self, column: str, condition: str, value: str):
        """データをフィルタリング"""
        if not self.data or column not in self.columns:
            return []
        
        filtered = []
        for row in self.data:
            row_value = row[column]
            
            if condition == "==":
                if row_value == value:
                    filtered.append(row)
            elif condition == "!=":
                if row_value != value:
                    filtered.append(row)
            elif condition == ">":
                try:
                    if float(row_value) > float(value):
                        filtered.append(row)
                except ValueError:
                    pass
            elif condition == "<":
                try:
                    if float(row_value) < float(value):
                        filtered.append(row)
                except ValueError:
                    pass
            elif condition == ">=":
                try:
                    if float(row_value) >= float(value):
                        filtered.append(row)
                except ValueError:
                    pass
            elif condition == "<=":
                try:
                    if float(row_value) <= float(value):
                        filtered.append(row)
                except ValueError:
                    pass
        
        return filtered
    
    def show_data_preview(self, num_rows: int = 5):
        """データのプレビューを表示"""
        if not self.data:
            print("データがありません")
            return
        
        print(f"\n=== データプレビュー (最初の{num_rows}件) ===")
        print(" | ".join(self.columns))
        print("-" * 60)
        
        for i, row in enumerate(self.data[:num_rows], 1):
            values = [str(row[col])[:15] for col in self.columns]
            print(" | ".join(values))
        
        print(f"\n合計: {len(self.data)}件")
    
    def show_column_info(self):
        """列の情報を表示"""
        if not self.data:
            print("データがありません")
            return
        
        print("\n=== 列の情報 ===")
        print(f"列数: {len(self.columns)}")
        print("\n列一覧:")
        for i, col in enumerate(self.columns, 1):
            print(f"  {i}. {col}")
    
    def show_text_chart(self, column: str, max_width: int = 50):
        """テキストベースのチャートを表示"""
        frequency = self.get_frequency(column)
        if not frequency:
            print("データがありません")
            return
        
        max_count = frequency[0][1] if frequency else 1
        
        print(f"\n=== {column} の分布 ===")
        for value, count in frequency:
            bar_length = int((count / max_count) * max_width)
            bar = "█" * bar_length
            percentage = (count / len(self.data) * 100) if self.data else 0
            print(f"{value[:20]:<20} | {bar} {count} ({percentage:.1f}%)")
```

## Step 3：メインプログラムの作成

`main.py` を作成します：

```python
# main.py
from analyzer import DataAnalyzer

def main():
    """メイン関数"""
    analyzer = DataAnalyzer()
    
    while True:
        print("\n=== データ分析ツール ===")
        print("1. データを読み込む")
        print("2. データのプレビュー")
        print("3. 列の情報を表示")
        print("4. 基本統計情報")
        print("5. 頻度分布")
        print("6. テキストチャート")
        print("7. データのフィルタリング")
        print("8. 終了")
        
        choice = input("\n選択してください (1-8): ").strip()
        
        if choice == "1":
            filename = input("CSVファイル名: ").strip() or "data.csv"
            analyzer.load_csv(filename)
        elif choice == "2":
            analyzer.show_data_preview()
        elif choice == "3":
            analyzer.show_column_info()
        elif choice == "4":
            show_basic_stats(analyzer)
        elif choice == "5":
            show_frequency(analyzer)
        elif choice == "6":
            show_text_chart(analyzer)
        elif choice == "7":
            filter_data_interactive(analyzer)
        elif choice == "8":
            print("ツールを終了します")
            break
        else:
            print("無効な選択です")

def show_basic_stats(analyzer):
    """基本統計情報を表示"""
    if not analyzer.data:
        print("データを読み込んでください")
        return
    
    analyzer.show_column_info()
    column = input("\n統計情報を表示する列: ").strip()
    
    stats = analyzer.get_basic_stats(column)
    if stats:
        print(f"\n=== {column} の基本統計情報 ===")
        for key, value in stats.items():
            if isinstance(value, float):
                print(f"{key}: {value:,.2f}")
            else:
                print(f"{key}: {value:,}")
    else:
        print("数値データではないか、列が存在しません")

def show_frequency(analyzer):
    """頻度分布を表示"""
    if not analyzer.data:
        print("データを読み込んでください")
        return
    
    analyzer.show_column_info()
    column = input("\n頻度分布を表示する列: ").strip()
    limit = input("表示件数 (Enterで10件): ").strip()
    limit = int(limit) if limit.isdigit() else 10
    
    frequency = analyzer.get_frequency(column, limit)
    if frequency:
        print(f"\n=== {column} の頻度分布 ===")
        for i, (value, count) in enumerate(frequency, 1):
            percentage = (count / len(analyzer.data) * 100) if analyzer.data else 0
            print(f"{i}. {value}: {count}件 ({percentage:.1f}%)")
    else:
        print("データがありません")

def show_text_chart(analyzer):
    """テキストチャートを表示"""
    if not analyzer.data:
        print("データを読み込んでください")
        return
    
    analyzer.show_column_info()
    column = input("\nチャートを表示する列: ").strip()
    analyzer.show_text_chart(column)

def filter_data_interactive(analyzer):
    """対話的にデータをフィルタリング"""
    if not analyzer.data:
        print("データを読み込んでください")
        return
    
    analyzer.show_column_info()
    column = input("\nフィルタリングする列: ").strip()
    
    print("\n条件:")
    print("  ==  等しい")
    print("  !=  等しくない")
    print("  >   より大きい")
    print("  <   より小さい")
    print("  >=  以上")
    print("  <=  以下")
    
    condition = input("条件: ").strip()
    value = input("値: ").strip()
    
    filtered = analyzer.filter_data(column, condition, value)
    
    print(f"\nフィルタリング結果: {len(filtered)}件")
    if filtered:
        print("\n最初の5件:")
        for i, row in enumerate(filtered[:5], 1):
            print(f"{i}. {row}")

if __name__ == "__main__":
    main()
```
