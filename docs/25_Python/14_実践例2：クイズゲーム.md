## Step 1：問題クラスの作成

`quiz.py` を作成します：

```python
# quiz.py
import json
import random
from typing import List, Dict

class Question:
    """問題を表すクラス"""
    def __init__(self, question, options, correct_answer, explanation=""):
        self.question = question
        self.options = options
        self.correct_answer = correct_answer
        self.explanation = explanation
    
    def display(self):
        """問題を表示"""
        print(f"\n{self.question}")
        for i, option in enumerate(self.options, 1):
            print(f"{i}. {option}")
    
    def check_answer(self, user_answer):
        """答えをチェック"""
        try:
            selected_option = self.options[int(user_answer) - 1]
            is_correct = selected_option == self.correct_answer
            return is_correct, selected_option
        except (ValueError, IndexError):
            return False, None

class QuizGame:
    """クイズゲームを管理するクラス"""
    def __init__(self):
        self.questions: List[Question] = []
        self.score = 0
        self.total_questions = 0
    
    def load_questions_from_json(self, filename="questions.json"):
        """JSONファイルから問題を読み込む"""
        try:
            with open(filename, "r", encoding="utf-8") as file:
                data = json.load(file)
                self.questions = []
                for item in data:
                    question = Question(
                        item["question"],
                        item["options"],
                        item["correct_answer"],
                        item.get("explanation", "")
                    )
                    self.questions.append(question)
            print(f"問題を {len(self.questions)} 問読み込みました")
        except FileNotFoundError:
            print(f"{filename} が見つかりません")
    
    def add_question(self, question, options, correct_answer, explanation=""):
        """問題を手動で追加"""
        q = Question(question, options, correct_answer, explanation)
        self.questions.append(q)
    
    def start_quiz(self, num_questions=None):
        """クイズを開始"""
        if not self.questions:
            print("問題がありません")
            return
        
        # 問題をシャッフル
        questions_to_ask = self.questions.copy()
        random.shuffle(questions_to_ask)
        
        # 問題数を決定
        if num_questions:
            questions_to_ask = questions_to_ask[:num_questions]
        
        self.score = 0
        self.total_questions = len(questions_to_ask)
        
        print(f"\n=== クイズゲーム開始 ===")
        print(f"問題数: {self.total_questions}問\n")
        
        for i, question in enumerate(questions_to_ask, 1):
            print(f"【問題 {i}/{self.total_questions}】")
            question.display()
            
            user_answer = input("\n答えを選択してください (1-4): ").strip()
            is_correct, selected_option = question.check_answer(user_answer)
            
            if is_correct:
                print("✅ 正解！")
                self.score += 1
            else:
                print(f"❌ 不正解...")
                print(f"正解は: {question.correct_answer}")
            
            if question.explanation:
                print(f"解説: {question.explanation}")
        
        self.show_results()
    
    def show_results(self):
        """結果を表示"""
        percentage = (self.score / self.total_questions * 100) if self.total_questions > 0 else 0
        
        print("\n" + "=" * 30)
        print("=== 結果 ===")
        print(f"正解数: {self.score}/{self.total_questions}")
        print(f"正解率: {percentage:.1f}%")
        
        if percentage == 100:
            print("🎉 完璧です！")
        elif percentage >= 80:
            print("👍 素晴らしい！")
        elif percentage >= 60:
            print("👌 よくできました！")
        else:
            print("💪 もう少し練習しましょう！")
        
        print("=" * 30)
```

## Step 3：メインプログラムの作成

`main.py` を作成します：

```python
# main.py
from quiz import QuizGame

def main():
    """メイン関数"""
    game = QuizGame()
    
    # 問題を読み込む
    game.load_questions_from_json()
    
    # デモ問題を追加（オプション）
    if not game.questions:
        add_demo_questions(game)
    
    while True:
        print("\n=== クイズゲーム ===")
        print("1. クイズを開始")
        print("2. 問題を追加")
        print("3. 問題一覧を表示")
        print("4. 終了")
        
        choice = input("\n選択してください (1-4): ").strip()
        
        if choice == "1":
            num = input("問題数 (Enterで全問): ").strip()
            num_questions = int(num) if num.isdigit() else None
            game.start_quiz(num_questions)
        elif choice == "2":
            add_question_interactive(game)
        elif choice == "3":
            show_questions(game)
        elif choice == "4":
            print("ゲームを終了します")
            break
        else:
            print("無効な選択です")

def add_demo_questions(game):
    """デモ問題を追加"""
    game.add_question(
        "Pythonの創始者は？",
        ["Guido van Rossum", "Bjarne Stroustrup", "James Gosling", "Linus Torvalds"],
        "Guido van Rossum",
        "Guido van Rossumが1991年にPythonを開発しました。"
    )
    game.add_question(
        "Pythonでリストを並び替えるメソッドは？",
        ["sort()", "order()", "arrange()", "sort_list()"],
        "sort()",
        "sort()メソッドは、リストを昇順に並び替えます。"
    )

def add_question_interactive(game):
    """対話的に問題を追加"""
    print("\n【問題を追加】")
    question = input("問題文: ").strip()
    
    options = []
    for i in range(4):
        option = input(f"選択肢 {i+1}: ").strip()
        options.append(option)
    
    correct = input("正解の選択肢番号 (1-4): ").strip()
    try:
        correct_answer = options[int(correct) - 1]
    except (ValueError, IndexError):
        print("無効な選択肢番号です")
        return
    
    explanation = input("解説 (任意): ").strip()
    
    game.add_question(question, options, correct_answer, explanation)
    print("問題を追加しました")

def show_questions(game):
    """問題一覧を表示"""
    if not game.questions:
        print("問題がありません")
        return
    
    print(f"\n=== 問題一覧 ({len(game.questions)}問) ===")
    for i, q in enumerate(game.questions, 1):
        print(f"\n{i}. {q.question}")
        for j, option in enumerate(q.options, 1):
            marker = "✓" if option == q.correct_answer else " "
            print(f"   {marker} {j}. {option}")

if __name__ == "__main__":
    main()
```
