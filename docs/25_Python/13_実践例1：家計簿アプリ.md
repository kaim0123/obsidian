## Step 1：データクラスの作成

`budget.py` を作成します：

```python
# budget.py
import csv
from datetime import datetime
from typing import List, Dict

class Transaction:
    """取引を表すクラス"""
    def __init__(self, date, category, amount, description, type):
        self.date = date
        self.category = category
        self.amount = amount
        self.description = description
        self.type = type  # "income" または "expense"
    
    def __str__(self):
        return f"{self.date} | {self.category} | {self.amount}円 | {self.description}"

class BudgetManager:
    """家計簿を管理するクラス"""
    def __init__(self):
        self.transactions: List[Transaction] = []
    
    def add_transaction(self, date, category, amount, description, type):
        """取引を追加"""
        transaction = Transaction(date, category, amount, description, type)
        self.transactions.append(transaction)
        print(f"取引を追加しました: {transaction}")
    
    def get_balance(self):
        """現在の残高を計算"""
        total_income = sum(t.amount for t in self.transactions if t.type == "income")
        total_expense = sum(t.amount for t in self.transactions if t.type == "expense")
        return total_income - total_expense
    
    def get_category_summary(self):
        """カテゴリー別の集計"""
        income_by_category = {}
        expense_by_category = {}
        
        for transaction in self.transactions:
            if transaction.type == "income":
                if transaction.category not in income_by_category:
                    income_by_category[transaction.category] = 0
                income_by_category[transaction.category] += transaction.amount
            else:
                if transaction.category not in expense_by_category:
                    expense_by_category[transaction.category] = 0
                expense_by_category[transaction.category] += transaction.amount
        
        return income_by_category, expense_by_category
    
    def show_summary(self):
        """集計を表示"""
        balance = self.get_balance()
        income_by_category, expense_by_category = self.get_category_summary()
        
        print("\n=== 家計簿サマリー ===")
        print(f"現在の残高: {balance:,}円")
        
        print("\n【収入】")
        total_income = sum(income_by_category.values())
        for category, amount in income_by_category.items():
            percentage = (amount / total_income * 100) if total_income > 0 else 0
            print(f"  {category}: {amount:,}円 ({percentage:.1f}%)")
        
        print("\n【支出】")
        total_expense = sum(expense_by_category.values())
        for category, amount in expense_by_category.items():
            percentage = (amount / total_expense * 100) if total_expense > 0 else 0
            print(f"  {category}: {amount:,}円 ({percentage:.1f}%)")
        
        print("=" * 30)
    
    def save_to_csv(self, filename="data.csv"):
        """CSVファイルに保存"""
        with open(filename, "w", encoding="utf-8", newline="") as file:
            writer = csv.writer(file)
            writer.writerow(["日付", "カテゴリー", "金額", "説明", "タイプ"])
            for transaction in self.transactions:
                writer.writerow([
                    transaction.date,
                    transaction.category,
                    transaction.amount,
                    transaction.description,
                    transaction.type
                ])
        print(f"データを {filename} に保存しました")
    
    def load_from_csv(self, filename="data.csv"):
        """CSVファイルから読み込む"""
        try:
            with open(filename, "r", encoding="utf-8") as file:
                reader = csv.DictReader(file)
                self.transactions = []
                for row in reader:
                    transaction = Transaction(
                        row["日付"],
                        row["カテゴリー"],
                        int(row["金額"]),
                        row["説明"],
                        row["タイプ"]
                    )
                    self.transactions.append(transaction)
            print(f"データを {filename} から読み込みました（{len(self.transactions)}件）")
        except FileNotFoundError:
            print(f"{filename} が見つかりません。新規作成します。")
```

## Step 2：メインプログラムの作成

`main.py` を作成します：

```python
# main.py
from budget import BudgetManager
from datetime import datetime

def main():
    """メイン関数"""
    manager = BudgetManager()
    
    # 既存のデータを読み込む
    manager.load_from_csv()
    
    while True:
        print("\n=== 家計簿アプリ ===")
        print("1. 収入を記録")
        print("2. 支出を記録")
        print("3. 集計を表示")
        print("4. データを保存")
        print("5. 終了")
        
        choice = input("\n選択してください (1-5): ")
        
        if choice == "1":
            add_income(manager)
        elif choice == "2":
            add_expense(manager)
        elif choice == "3":
            manager.show_summary()
        elif choice == "4":
            manager.save_to_csv()
        elif choice == "5":
            print("アプリを終了します")
            break
        else:
            print("無効な選択です")

def add_income(manager):
    """収入を追加"""
    print("\n【収入を記録】")
    date = input("日付 (YYYY-MM-DD、Enterで今日): ").strip()
    if not date:
        date = datetime.now().strftime("%Y-%m-%d")
    
    category = input("カテゴリー (給与、ボーナスなど): ").strip()
    amount = int(input("金額: "))
    description = input("説明 (任意): ").strip()
    
    manager.add_transaction(date, category, amount, description, "income")

def add_expense(manager):
    """支出を追加"""
    print("\n【支出を記録】")
    date = input("日付 (YYYY-MM-DD、Enterで今日): ").strip()
    if not date:
        date = datetime.now().strftime("%Y-%m-%d")
    
    category = input("カテゴリー (食費、交通費、娯楽など): ").strip()
    amount = int(input("金額: "))
    description = input("説明 (任意): ").strip()
    
    manager.add_transaction(date, category, amount, description, "expense")

if __name__ == "__main__":
    main()
```
