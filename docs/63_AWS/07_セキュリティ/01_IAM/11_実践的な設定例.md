# 実践的な設定例

## 概要

このドキュメントでは、実際のビジネスシナリオに基づいたIAMの設定例を提供します。各例は段階的な手順で説明され、実際の環境で即座に使用できるようになっています。

## シナリオ1: 小規模スタートアップ向けの基本設定

### 要件
- 開発者3名、管理者1名
- 開発環境と本番環境の分離
- 基本的なセキュリティ要件

### 実装手順

#### ステップ1: 管理者ユーザーの作成
```bash

    --user-name admin-user \
    --tags Key=Role,Value=Administrator Key=Environment,Value=All

# 管理者グループの作成

# 管理者ポリシーのアタッチ

    --group-name administrators \
    --policy-arn arn:aws:iam::aws:policy/AdministratorAccess

    --group-name administrators \
    --user-name admin-user
```

```python
import boto3
import json

def create_tenant_resources(tenant_id, tenant_name):
    iam = boto3.client('iam')
    s3 = boto3.client('s3')
    
    # テナント用グループの作成
    group_name = f"tenant-{tenant_id}"
    iam.create_group(GroupName=group_name)
    
    # テナント用ポリシーの作成
    policy_document = {
        "Version": "2012-10-17",
        "Statement": [
            {
                "Effect": "Allow",
                "Action": "*",
                "Resource": "*",
                "Condition": {
                    "StringEquals": {
                        "aws:ResourceTag/Tenant": tenant_id
                    }
                }
            }
        ]
    }
    
    policy_name = f"Tenant{tenant_id}Policy"
    iam.create_policy(
        PolicyName=policy_name,
        PolicyDocument=json.dumps(policy_document)
    )
    
    # ポリシーをグループにアタッチ
    policy_arn = f"arn:aws:iam::{boto3.client('sts').get_caller_identity()['Account']}:policy/{policy_name}"
    iam.attach_group_policy(
        GroupName=group_name,
        PolicyArn=policy_arn
    )
    
    # テナント用S3バケットの作成
    bucket_name = f"tenant-{tenant_id}-bucket"
    s3.create_bucket(Bucket=bucket_name)
    
    # バケットにテナントタグを設定
    s3.put_bucket_tagging(
        Bucket=bucket_name,
        Tagging={
            'TagSet': [
                {'Key': 'Tenant', 'Value': tenant_id},
                {'Key': 'TenantName', 'Value': tenant_name}
            ]
        }
    )
    
    return {
        'group_name': group_name,
        'policy_name': policy_name,
        'bucket_name': bucket_name
    }
```

## シナリオ7: 災害復旧とバックアップ

### 要件
- IAM設定のバックアップ
- 災害復旧手順
- 設定の復元

### 実装手順

#### ステップ1: IAM設定のバックアップ
```python
import boto3
import json
from datetime import datetime

def backup_iam_configuration():
    iam = boto3.client('iam')
    backup_data = {
        'timestamp': datetime.now().isoformat(),
        'users': [],
        'groups': [],
        'roles': [],
        'policies': []
    }
    
    # ユーザーのバックアップ
    users = iam.list_users()
    for user in users['Users']:
        user_data = {
            'user': user,
            'policies': iam.list_attached_user_policies(UserName=user['UserName']),
            'groups': iam.get_groups_for_user(UserName=user['UserName'])
        }
        backup_data['users'].append(user_data)
    
    # グループのバックアップ
    groups = iam.list_groups()
    for group in groups['Groups']:
        group_data = {
            'group': group,
            'policies': iam.list_attached_group_policies(GroupName=group['GroupName']),
            'users': iam.get_group(GroupName=group['GroupName'])
        }
        backup_data['groups'].append(group_data)
    
    # ロールのバックアップ
    roles = iam.list_roles()
    for role in roles['Roles']:
        role_data = {
            'role': role,
            'policies': iam.list_attached_role_policies(RoleName=role['RoleName']),
            'trust_policy': iam.get_role(RoleName=role['RoleName'])
        }
        backup_data['roles'].append(role_data)
    
    # カスタムポリシーのバックアップ
    policies = iam.list_policies(Scope='Local')
    for policy in policies['Policies']:
        policy_data = {
            'policy': policy,
            'policy_document': iam.get_policy_version(
                PolicyArn=policy['Arn'],
                VersionId=policy['DefaultVersionId']
            )
        }
        backup_data['policies'].append(policy_data)
    
    # バックアップファイルの保存
    with open(f'iam-backup-{datetime.now().strftime("%Y%m%d-%H%M%S")}.json', 'w') as f:
        json.dump(backup_data, f, indent=2, default=str)
    
    return backup_data
```

#### ステップ2: 災害復旧手順
```python
def restore_iam_configuration(backup_file):
    iam = boto3.client('iam')
    
    with open(backup_file, 'r') as f:
        backup_data = json.load(f)
    
    # ポリシーの復元
    for policy_data in backup_data['policies']:
        policy = policy_data['policy']
        policy_document = policy_data['policy_document']['PolicyVersion']['Document']
        
        try:
            iam.create_policy(
                PolicyName=policy['PolicyName'],
                PolicyDocument=json.dumps(policy_document),
                Description=policy.get('Description', '')
            )
            print(f"Policy restored: {policy['PolicyName']}")
        except iam.exceptions.EntityAlreadyExistsException:
            print(f"Policy already exists: {policy['PolicyName']}")
    
    # グループの復元
    for group_data in backup_data['groups']:
        group = group_data['group']
        
        try:
            iam.create_group(GroupName=group['GroupName'])
            print(f"Group restored: {group['GroupName']}")
        except iam.exceptions.EntityAlreadyExistsException:
            print(f"Group already exists: {group['GroupName']}")
    
    # ユーザーの復元
    for user_data in backup_data['users']:
        user = user_data['user']
        
        try:
            iam.create_user(UserName=user['UserName'])
            print(f"User restored: {user['UserName']}")
        except iam.exceptions.EntityAlreadyExistsException:
            print(f"User already exists: {user['UserName']}")
    
    print("IAM configuration restoration completed")
```

```python
    return {
        'statusCode': 200,
        'body': json.dumps(f'Processed {len(suspicious_events)} suspicious events')
    }

def is_suspicious_event(event):
    # 異常なアクセスパターンの判定ロジック
    source_ip = event.get('SourceIPAddress', '')
    user_agent = event.get('UserAgent', '')
    
    # 例: 海外からのアクセス、異常な時間帯のアクセスなど
    suspicious_ips = ['0.0.0.0/0']  # 実際の設定に置き換え
    suspicious_user_agents = ['curl', 'wget']  # 実際の設定に置き換え
    
    return (source_ip in suspicious_ips or 
            any(agent in user_agent.lower() for agent in suspicious_user_agents))
```
