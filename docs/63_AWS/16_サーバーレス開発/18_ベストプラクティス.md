### Lambda関数の構造
```python
# functions/api/users/app.py

import json
import os
from typing import Dict, Any
from shared.utils import logger, create_response
from shared.validators import validate_user_input
from shared.models import User

# グローバルスコープ（再利用される）
import boto3
dynamodb = boto3.resource('dynamodb')
users_table = dynamodb.Table(os.environ['USERS_TABLE'])

def lambda_handler(event: Dict[str, Any], context: Any) -> Dict[str, Any]:
    """
    メインハンドラー：薄く保ち、ビジネスロジックは分離
    """
    try:
        # リクエストのパース
        http_method = event['requestContext']['http']['method']
        
        # ルーティング
        if http_method == 'GET':
            return handle_get(event)
        elif http_method == 'POST':
            return handle_post(event)
        else:
            return create_response(405, {'error': 'Method not allowed'})
    
    except Exception as e:
        logger.error(f"Unhandled error: {str(e)}", exc_info=True)
        return create_response(500, {'error': 'Internal server error'})

def handle_get(event: Dict[str, Any]) -> Dict[str, Any]:
    """GET リクエストの処理"""
    user_id = event['pathParameters'].get('id')
    
    if user_id:
        user = get_user_by_id(user_id)
        return create_response(200, user.to_dict())
    else:
        users = list_users()
        return create_response(200, {'users': [u.to_dict() for u in users]})

def handle_post(event: Dict[str, Any]) -> Dict[str, Any]:
    """POST リクエストの処理"""
    body = json.loads(event['body'])
    
    # バリデーション
    errors = validate_user_input(body)
    if errors:
        return create_response(400, {'errors': errors})
    
    # ビジネスロジック
    user = create_user(body)
    return create_response(201, user.to_dict())

# ビジネスロジック（テスト可能）
def get_user_by_id(user_id: str) -> User:
    """ユーザーをIDで取得"""
    response = users_table.get_item(Key={'userId': user_id})
    if 'Item' not in response:
        raise ValueError(f'User {user_id} not found')
    return User.from_dict(response['Item'])

def list_users(limit: int = 20) -> list:
    """ユーザー一覧を取得"""
    response = users_table.scan(Limit=limit)
    return [User.from_dict(item) for item in response['Items']]

def create_user(user_data: Dict[str, Any]) -> User:
    """ユーザーを作成"""
    user = User.from_dict(user_data)
    users_table.put_item(Item=user.to_dict())
    return user
```

```python
# functions/shared/utils.py

import json
import logging
from typing import Dict, Any
from decimal import Decimal

# ロガー設定
logger = logging.getLogger()
logger.setLevel(logging.INFO)

class DecimalEncoder(json.JSONEncoder):
    """DynamoDB Decimal型のシリアライズ"""
    def default(self, obj):
        if isinstance(obj, Decimal):
            return float(obj)
        return super(DecimalEncoder, self).default(obj)

def create_response(status_code: int, body: Any, headers: Dict = None) -> Dict[str, Any]:
    """標準的なAPIレスポンスを生成"""
    response = {
        'statusCode': status_code,
        'headers': {
            'Content-Type': 'application/json',
            'Access-Control-Allow-Origin': '*',
            **(headers or {})
        },
        'body': json.dumps(body, cls=DecimalEncoder)
    }
    return response

def get_user_from_event(event: Dict[str, Any]) -> Dict[str, Any]:
    """イベントから認証ユーザー情報を取得"""
    authorizer = event.get('requestContext', {}).get('authorizer', {})
    jwt_claims = authorizer.get('jwt', {}).get('claims', {})
    
    return {
        'userId': jwt_claims.get('sub'),
        'email': jwt_claims.get('email'),
        'name': jwt_claims.get('name')
    }
```

```python
# functions/shared/models.py

from dataclasses import dataclass, asdict
from datetime import datetime
from typing import Optional
import uuid

@dataclass
class User:
    userId: str
    email: str
    name: str
    createdAt: int
    updatedAt: Optional[int] = None
    
    @classmethod
    def from_dict(cls, data: dict) -> 'User':
        """辞書からインスタンスを生成"""
        return cls(**{k: v for k, v in data.items() if k in cls.__annotations__})
    
    def to_dict(self) -> dict:
        """辞書に変換"""
        return {k: v for k, v in asdict(self).items() if v is not None}
    
    @classmethod
    def create_new(cls, email: str, name: str) -> 'User':
        """新しいユーザーを生成"""
        now = int(datetime.now().timestamp())
        return cls(
            userId=str(uuid.uuid4()),
            email=email,
            name=name,
            createdAt=now
        )
```

```python
# functions/shared/exceptions.py

class AppError(Exception):
    """アプリケーションエラーのベースクラス"""
    def __init__(self, message: str, status_code: int = 500, details: dict = None):
        self.message = message
        self.status_code = status_code
        self.details = details or {}
        super().__init__(self.message)

class ValidationError(AppError):
    """バリデーションエラー"""
    def __init__(self, message: str, details: dict = None):
        super().__init__(message, 400, details)

class NotFoundError(AppError):
    """リソースが見つからない"""
    def __init__(self, resource: str, identifier: str):
        super().__init__(f'{resource} not found: {identifier}', 404)

class UnauthorizedError(AppError):
    """認証エラー"""
    def __init__(self, message: str = 'Unauthorized'):
        super().__init__(message, 401)

class ForbiddenError(AppError):
    """認可エラー"""
    def __init__(self, message: str = 'Forbidden'):
        super().__init__(message, 403)

class ConflictError(AppError):
    """競合エラー"""
    def __init__(self, message: str):
        super().__init__(message, 409)
```

### エラーハンドリングデコレーター
```python
# functions/shared/decorators.py

import functools
from typing import Callable
from .utils import create_response, logger
from .exceptions import AppError

def handle_errors(func: Callable) -> Callable:
    """エラーハンドリングデコレーター"""
    @functools.wraps(func)
    def wrapper(event, context):
        try:
            return func(event, context)
        
        except AppError as e:
            logger.warning(f"Application error: {e.message}", extra={'details': e.details})
            return create_response(e.status_code, {
                'error': e.message,
                'details': e.details
            })
        
        except Exception as e:
            logger.error(f"Unexpected error: {str(e)}", exc_info=True)
            return create_response(500, {
                'error': 'Internal server error'
            })
    
    return wrapper
```

```python
# functions/shared/logger.py

import json
import logging
import os
from datetime import datetime
from typing import Any, Dict

class StructuredLogger:
    """構造化ログを出力"""
    
    def __init__(self, service_name: str = None):
        self.logger = logging.getLogger()
        self.logger.setLevel(os.environ.get('LOG_LEVEL', 'INFO'))
        self.service_name = service_name or os.environ.get('AWS_LAMBDA_FUNCTION_NAME', 'unknown')
    
    def _log(self, level: str, message: str, **kwargs):
        """ログを出力"""
        log_entry = {
            'timestamp': datetime.utcnow().isoformat(),
            'level': level,
            'service': self.service_name,
            'message': message,
            **kwargs
        }
        
        # CloudWatch Logs Insightsでクエリしやすい形式
        self.logger.log(
            getattr(logging, level),
            json.dumps(log_entry, default=str)
        )
    
    def info(self, message: str, **kwargs):
        self._log('INFO', message, **kwargs)
    
    def warning(self, message: str, **kwargs):
        self._log('WARNING', message, **kwargs)
    
    def error(self, message: str, **kwargs):
        self._log('ERROR', message, **kwargs)
    
    def debug(self, message: str, **kwargs):
        self._log('DEBUG', message, **kwargs)

# 使用例
logger = StructuredLogger('user-service')

def lambda_handler(event, context):
    logger.info('Processing request', 
                user_id='123',
                action='create_order',
                request_id=context.request_id)
    
    try:
        # 処理...
        logger.info('Order created successfully',
                   order_id='456',
                   amount=1000)
    except Exception as e:
        logger.error('Failed to create order',
                    error=str(e),
                    user_id='123')
```

### 入力バリデーション
```python
from pydantic import BaseModel, EmailStr, validator, constr
from typing import Optional

class CreateUserRequest(BaseModel):
    """ユーザー作成リクエスト"""
    email: EmailStr
    name: constr(min_length=1, max_length=100)
    age: Optional[int] = None
    
    @validator('name')
    def name_must_not_contain_special_chars(cls, v):
        if not v.replace(' ', '').isalnum():
            raise ValueError('Name must contain only alphanumeric characters')
        return v.strip()
    
    @validator('age')
    def age_must_be_reasonable(cls, v):
        if v is not None and (v < 0 or v > 150):
            raise ValueError('Age must be between 0 and 150')
        return v

# 使用例
def lambda_handler(event, context):
    try:
        body = json.loads(event['body'])
        request = CreateUserRequest(**body)
        
        # 検証済みデータを使用
        user = create_user(request.dict())
        
    except ValueError as e:
        return create_response(400, {'error': str(e)})
```
