import json
import pytest
from decimal import Decimal
from lib.utils import DecimalEncoder, create_response, get_tenant_id


class TestDecimalEncoder:
    def test_encode_decimal(self):
        data = {'value': Decimal('10.5')}
        result = json.dumps(data, cls=DecimalEncoder)
        assert result == '{"value": 10.5}'


class TestCreateResponse:
    def test_success_response(self):
        response = create_response(200, {'message': 'success'})
        
        assert response['statusCode'] == 200
        assert response['headers']['Content-Type'] == 'application/json'
        assert 'Access-Control-Allow-Origin' in response['headers']
        
        body = json.loads(response['body'])
        assert body['message'] == 'success'
    
    def test_error_response(self):
        response = create_response(400, {'error': 'Bad request'})
        assert response['statusCode'] == 400


class TestGetTenantId:
    def test_get_tenant_id_success(self):
        event = {
            'requestContext': {
                'authorizer': {
                    'jwt': {
                        'claims': {
                            'custom:tenantId': 'tenant123'
                        }
                    }
                }
            }
        }
        
        tenant_id = get_tenant_id(event)
        assert tenant_id == 'tenant123'
    
    def test_get_tenant_id_missing(self):
        event = {
            'requestContext': {
                'authorizer': {
                    'jwt': {
                        'claims': {}
                    }
                }
            }
        }
        
        with pytest.raises(ValueError):
            get_tenant_id(event)

import pytest
import requests
import json
from datetime import datetime


class TestTenantAPI:
    """テナントAPI統合テスト"""
    
    @pytest.fixture
    def api_url(self):
        """API URLを環境変数から取得"""
        import os
        return os.environ.get('API_URL', 'http://localhost:3000')
    
    @pytest.fixture
    def auth_token(self, api_url):
        """認証トークンを取得"""
        response = requests.post(
            f'{api_url}/auth/signin',
            json={
                'email': 'admin@example.com',
                'password': 'TestPassword123!'
            }
        )
        assert response.status_code == 200
        return response.json()['idToken']
    
    def test_create_tenant(self, api_url, auth_token):
        """テナント作成テスト"""
        response = requests.post(
            f'{api_url}/tenants',
            headers={'Authorization': f'Bearer {auth_token}'},
            json={
                'tenantName': 'テスト株式会社',
                'plan': 'standard',
                'maxUsers': 50
            }
        )
        
        assert response.status_code == 201
        data = response.json()
        assert 'tenantId' in data
        assert data['tenantName'] == 'テスト株式会社'
        
        return data['tenantId']
    
    def test_get_tenant(self, api_url, auth_token):
        """テナント取得テスト"""
        # テナント作成
        tenant_id = self.test_create_tenant(api_url, auth_token)
        
        # テナント取得
        response = requests.get(
            f'{api_url}/tenants/{tenant_id}',
            headers={'Authorization': f'Bearer {auth_token}'}
        )
        
        assert response.status_code == 200
        data = response.json()
        assert data['tenantId'] == tenant_id

```python
# tests/integration/test_user_api.py

import pytest
import requests


class TestUserAPI:
    """ユーザーAPI統合テスト"""
    
    @pytest.fixture
    def api_url(self):
        import os
        return os.environ.get('API_URL', 'http://localhost:3000')
    
    @pytest.fixture
    def admin_token(self, api_url):
        response = requests.post(
            f'{api_url}/auth/signin',
            json={
                'email': 'admin@example.com',
                'password': 'TestPassword123!'
            }
        )
        return response.json()['idToken']
    
    def test_create_user(self, api_url, admin_token):
        """ユーザー作成テスト"""
        response = requests.post(
            f'{api_url}/users',
            headers={'Authorization': f'Bearer {admin_token}'},
            json={
                'email': f'test{datetime.now().timestamp()}@example.com',
                'name': 'テストユーザー',
                'role': 'user'
            }
        )
        
        assert response.status_code == 201
        data = response.json()
        assert 'userId' in data
        assert data['email'] == f'test{datetime.now().timestamp()}@example.com'
        
        return data['userId']
    
    def test_list_users(self, api_url, admin_token):
        """ユーザー一覧取得テスト"""
        response = requests.get(
            f'{api_url}/users',
            headers={'Authorization': f'Bearer {admin_token}'}
        )
        
        assert response.status_code == 200
        data = response.json()
        assert 'users' in data
        assert 'count' in data
        assert isinstance(data['users'], list)
    
    def test_update_user(self, api_url, admin_token):
        """ユーザー更新テスト"""
        # ユーザー作成
        user_id = self.test_create_user(api_url, admin_token)
        
        # ユーザー更新
        response = requests.put(
            f'{api_url}/users/{user_id}',
            headers={'Authorization': f'Bearer {admin_token}'},
            json={
                'name': '更新されたユーザー',
                'profile': {
                    'department': '開発部'
                }
            }
        )
        
        assert response.status_code == 200
        data = response.json()
        assert data['name'] == '更新されたユーザー'
        assert data['profile']['department'] == '開発部'
    
    def test_delete_user(self, api_url, admin_token):
        """ユーザー削除テスト"""
        # ユーザー作成
        user_id = self.test_create_user(api_url, admin_token)
        
        # ユーザー削除
        response = requests.delete(
            f'{api_url}/users/{user_id}',
            headers={'Authorization': f'Bearer {admin_token}'}
        )
        
        assert response.status_code == 204
        
        # 削除確認
        response = requests.get(
            f'{api_url}/users/{user_id}',
            headers={'Authorization': f'Bearer {admin_token}'}
        )
        assert response.status_code == 404
```

## パフォーマンステスト

### Locustによる負荷テスト

```python
# locustfile.py

from locust import HttpUser, task, between
import json


class TenantUserAPIUser(HttpUser):
    wait_time = between(1, 3)
    
    def on_start(self):
        """テスト開始時に認証"""
        response = self.client.post('/auth/signin', json={
            'email': 'admin@example.com',
            'password': 'TestPassword123!'
        })
        self.token = response.json()['idToken']
        self.headers = {'Authorization': f'Bearer {self.token}'}
    
    @task(3)
    def list_users(self):
        """ユーザー一覧取得（高頻度）"""
        self.client.get('/users', headers=self.headers)
    
    @task(1)
    def get_tenant(self):
        """テナント取得"""
        self.client.get('/tenants/tenant-id', headers=self.headers)
    
    @task(1)
    def create_user(self):
        """ユーザー作成（低頻度）"""
        self.client.post('/users', 
            headers=self.headers,
            json={
                'email': f'test{self.environment.runner.user_count}@example.com',
                'name': 'Test User',
                'role': 'user'
            }
        )
```

実行：

```bash
# 10ユーザー、1秒あたり2ユーザー追加
locust -f locustfile.py --host=${API_URL} --users 10 --spawn-rate 2
```
