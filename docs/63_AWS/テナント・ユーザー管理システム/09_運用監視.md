# 運用監視

## CloudWatchによる監視

### 1. ログ管理

#### Lambda関数のログ

```
ロググループ: /aws/lambda/{FunctionName}

例:
- /aws/lambda/CreateUser-dev
- /aws/lambda/GetUser-dev
- /aws/lambda/ListUsers-dev
```

#### ログの確認

```bash
# 最新ログの確認
aws logs tail /aws/lambda/CreateUser-dev --follow

# 特定期間のログ検索
aws logs filter-log-events \
  --log-group-name /aws/lambda/CreateUser-dev \
  --start-time 1234567890000 \
  --filter-pattern "ERROR"
```

#### CloudWatch Logs Insights クエリ

```
# エラーログの抽出
fields @timestamp, @message
| filter @message like /ERROR/
| sort @timestamp desc
| limit 100

# レスポンス時間の分析
fields @timestamp, @duration
| filter @type = "REPORT"
| stats avg(@duration), max(@duration), min(@duration)

# ユーザー作成の成功率
fields @timestamp, @message
| filter @message like /User created/
| stats count() as success_count by bin(5m)
```

### 3. カスタムメトリクス

```python
# layers/common/python/lib/metrics.py

import boto3
from datetime import datetime

cloudwatch = boto3.client('cloudwatch')

def put_metric(
    metric_name: str,
    value: float,
    unit: str = 'Count',
    dimensions: dict = None
):
    """カスタムメトリクス送信"""
    metric_data = {
        'MetricName': metric_name,
        'Value': value,
        'Unit': unit,
        'Timestamp': datetime.utcnow()
    }
    
    if dimensions:
        metric_data['Dimensions'] = [
            {'Name': k, 'Value': v}
            for k, v in dimensions.items()
        ]
    
    cloudwatch.put_metric_data(
        Namespace='TenantUserManagement',
        MetricData=[metric_data]
    )


# 使用例
def lambda_handler(event, context):
    try:
        # ビジネスロジック
        result = process_request(event)
        
        # 成功メトリクス
        put_metric(
            'UserCreationSuccess',
            1,
            dimensions={
                'Environment': os.environ['ENVIRONMENT'],
                'TenantId': tenant_id
            }
        )
        
        return result
        
    except Exception as e:
        # 失敗メトリクス
        put_metric(
            'UserCreationFailure',
            1,
            dimensions={
                'Environment': os.environ['ENVIRONMENT'],
                'ErrorType': type(e).__name__
            }
        )
        raise
```

### 1. CloudWatch Alarms

#### Lambda エラーアラート

```yaml
# SAM template.yaml に追加

Resources:
  # SNS トピック（アラート通知用）
  AlertTopic:
    Type: AWS::SNS::Topic
    Properties:
      TopicName: !Sub tenant-user-alerts-${Environment}
      Subscription:
        - Endpoint: admin@example.com
          Protocol: email
  
  # Lambda エラーアラーム
  CreateUserErrorAlarm:
    Type: AWS::CloudWatch::Alarm
    Properties:
      AlarmName: !Sub CreateUser-Errors-${Environment}
      AlarmDescription: CreateUser function errors
      MetricName: Errors
      Namespace: AWS/Lambda
      Statistic: Sum
      Period: 300  # 5分
      EvaluationPeriods: 1
      Threshold: 5  # 5分間に5回以上エラー
      ComparisonOperator: GreaterThanThreshold
      Dimensions:
        - Name: FunctionName
          Value: !Ref CreateUserFunction
      AlarmActions:
        - !Ref AlertTopic
      TreatMissingData: notBreaching
  
  # Lambda スロットルアラーム
  CreateUserThrottleAlarm:
    Type: AWS::CloudWatch::Alarm
    Properties:
      AlarmName: !Sub CreateUser-Throttles-${Environment}
      AlarmDescription: CreateUser function throttles
      MetricName: Throttles
      Namespace: AWS/Lambda
      Statistic: Sum
      Period: 60
      EvaluationPeriods: 1
      Threshold: 1
      ComparisonOperator: GreaterThanThreshold
      Dimensions:
        - Name: FunctionName
          Value: !Ref CreateUserFunction
      AlarmActions:
        - !Ref AlertTopic
  
  # API Gateway 5xxエラーアラーム
  ApiGateway5xxAlarm:
    Type: AWS::CloudWatch::Alarm
    Properties:
      AlarmName: !Sub API-5xxErrors-${Environment}
      AlarmDescription: API Gateway 5xx errors
      MetricName: 5XXError
      Namespace: AWS/ApiGateway
      Statistic: Sum
      Period: 300
      EvaluationPeriods: 1
      Threshold: 10
      ComparisonOperator: GreaterThanThreshold
      Dimensions:
        - Name: ApiName
          Value: !Ref HttpApi
      AlarmActions:
        - !Ref AlertTopic
  
  # DynamoDB スロットルアラーム
  DynamoDBThrottleAlarm:
    Type: AWS::CloudWatch::Alarm
    Properties:
      AlarmName: !Sub DynamoDB-Throttles-${Environment}
      AlarmDescription: DynamoDB throttling events
      MetricName: UserErrors
      Namespace: AWS/DynamoDB
      Statistic: Sum
      Period: 300
      EvaluationPeriods: 1
      Threshold: 5
      ComparisonOperator: GreaterThanThreshold
      Dimensions:
        - Name: TableName
          Value: !Ref TenantUserTable
      AlarmActions:
        - !Ref AlertTopic
```

from aws_xray_sdk.core import xray_recorder
from aws_xray_sdk.core import patch_all

# AWS SDKにパッチを適用
patch_all()


def lambda_handler(event, context):
    # サブセグメント作成
    with xray_recorder.capture('business_logic'):
        result = process_business_logic()
    
    # 注釈追加

### 2. DynamoDB パフォーマンス最適化

```python
# バッチ読み取り
def batch_get_users(user_ids: list):
    """複数ユーザーを効率的に取得"""
    from boto3.dynamodb.conditions import Key
    
    # 最大100件まで一度に取得
    keys = [
        {'PK': f'TENANT#{tenant_id}', 'SK': f'USER#{uid}'}
        for uid in user_ids[:100]
    ]
    
    response = dynamodb.batch_get_item(
        RequestItems={
            table.table_name: {'Keys': keys}
        }
    )
    
    return response['Responses'][table.table_name]


# クエリの最適化（GSI使用）
def list_users_optimized(tenant_id: str):
    """GSIを使用した効率的なクエリ"""
    response = table.query(
        IndexName='TenantIndex',
        KeyConditionExpression=Key('tenantId').eq(tenant_id),
        ProjectionExpression='userId, #n, email, #r',
        ExpressionAttributeNames={
            '#n': 'name',
            '#r': 'role'
        }
    )
    return response['Items']
```

# Lambda関数での異常検知

def detect_anomaly(event, context):
    """異常なアクセスパターンを検知"""
    
    # 短時間での大量リクエスト
    request_count = get_request_count_last_minute(user_id)
    if request_count > 100:
        alert_security_team(f'Suspicious activity from user {user_id}')
    
    # 異常な時間帯のアクセス
    from datetime import datetime
    hour = datetime.now().hour
    if hour < 6 or hour > 23:
        log_suspicious_access(user_id, 'unusual_time')
    
    # 複数テナントへのアクセス試行
    if attempting_cross_tenant_access(event):
        alert_security_team(f'Cross-tenant access attempt by {user_id}')
